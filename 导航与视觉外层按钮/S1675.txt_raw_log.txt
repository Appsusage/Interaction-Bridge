function positionFloatDockItems() {
  if (!floatMenu || !mainFloat || !floatMenu.classList.contains("open")) return;
  const items = Array.from(floatMenu.querySelectorAll(".float-item"));
  if (!items.length) return;
  const size = 44;
  const center = 29;
  const viewportW = window.innerWidth || document.documentElement.clientWidth;
  const viewportH = window.innerHeight || document.documentElement.clientHeight;
  const anchorRight = window.matchMedia("(max-width: 600px)").matches ? 22 : 28;
  const anchorBottom = window.matchMedia("(max-width: 600px)").matches ? 112 : 140;
  const cx = viewportW - anchorRight - center;
  const cy = viewportH - anchorBottom - center;
  const minSpacingRadius = items.length > 1
    ? (size + 4) / (2 * Math.sin(Math.PI / (2 * (items.length - 1))))
    : 0;
  const radius = Math.max(58, minSpacingRadius, Math.min(94, viewportW * 0.22, viewportH * 0.19));
  const arcAngles = items.length === 1
    ? [180]
    : items.map((_, i) => 90 + (180 * i / (items.length - 1)));
  items.forEach((item, i) => {
    const angle = arcAngles[i] * Math.PI / 180;
    let x = center + Math.cos(angle) * radius - size / 2;
    let y = center + Math.sin(angle) * radius - size / 2;
    // Keep each circular button inside the visible viewport.
    const minX = center - cx;
    const maxX = viewportW - size - cx + center;
    const minY = center - cy;
    const maxY = viewportH - size - cy + center;
    x = Math.max(minX, Math.min(x, maxX));
    y = Math.max(minY, Math.min(y, maxY));
    item.style.transform = `translate(${x}px, ${y}px)`;
  });
}
