import { chromium } from "playwright";
const browser = await chromium.launch({ executablePath:"/repl/tools/bin/chromium", headless:true, args:["--no-sandbox","--disable-dev-shm-usage"] });
const page = await browser.newPage({ viewport:{width:393,height:852} });
const errors=[];
page.on("pageerror",err=>errors.push("page:"+err.message));
page.on("console",msg=>{ if(msg.type()==="error") errors.push("console:"+msg.text().slice(0,240)); });
try {
  const response=await page.goto("http://127.0.0.1:3000/navigation/",{waitUntil:"domcontentloaded",timeout:15000});
  await page.waitForTimeout(2800);
  console.log("RENDER_RESULT "+JSON.stringify({
    http:response?.status(),
    title:await page.title(),
    statusText:await page.locator("#status").innerText(),
    mapExists:await page.locator(".leaflet-container").count(),
    tileLoaded:await page.locator(".leaflet-tile-loaded").count(),
    visibleButtons:await page.locator("button:visible").allTextContents(),
    errors
  }));
} finally { await browser.close(); }
'
