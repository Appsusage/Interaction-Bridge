set -euo pipefail
cd /home/runner/workspace/ai-agent
test -s capabilities/navigation/data/regions/selangor-pilot.osm.pbf
test ! -e capabilities/navigation/data/regions/selangor.sqlite
region_stage_dir=$(mktemp -d capabilities/navigation/data/regions/.import-stage.XXXXXX)
region_stage="$region_stage_dir/selangor.sqlite"
nix-shell -p osmium-tool --run 'osmium export -f geojsonseq -u type_id capabilities/navigation/data/regions/selangor-pilot.osm.pbf' | node capabilities/navigation/local/import_osm.js "$region_stage"

rmdir "$region_stage_dir" || true
wc -c capabilities/navigation/data/regions/selangor.sqlite
