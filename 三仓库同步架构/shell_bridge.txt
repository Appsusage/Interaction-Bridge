set -euo pipefail
src=/home/runner/workspace/ai-agent
test -d "$src"
test ! -d "$src/.git"
stage="$(mktemp -d /home/runner/workspace/gen-bootstrap.XXXXXXXX)"
clone="$stage/repo"
printf 'GEN_STAGE_CREATED=%s\n' "$stage"
git -c 'credential.helper=!gh auth git-credential' clone --quiet https://github.com/Appsusage/Gen.git "$clone"
test -d "$clone/.git"
cp -a "$src/." "$clone/"
git -C "$clone" lfs install --local >/dev/null
lfs_count=0
while IFS= read -r -d '' item; do
  rel="${item#"$clone/"}"
  git -C "$clone" lfs track -- "$rel" >/dev/null
  lfs_count=$((lfs_count+1))
done < <(find "$clone" -type f -size +90M -not -path "$clone/.git/*" -print0)
git -C "$clone" add -A -f -- .
git -C "$clone" config user.name ai-agent-sync
git -C "$clone" config user.email ai-agent-sync@users.noreply.github.com
git -C "$clone" commit -qm 'Gen: initial complete ai-agent working snapshot'
tracked="$(git -C "$clone" ls-files -z | tr -cd '\0' | wc -c)"
lfs="$(git -C "$clone" lfs ls-files --name-only | wc -l)"
test "$lfs" -ge "$lfs_count"
printf 'GEN_STAGED_FILES=%s\n' "$tracked"
printf 'GEN_LFS_FILES=%s\n' "$lfs"
printf 'GEN_STAGE_COMMIT=%s\n' "$(git -C "$clone" rev-parse HEAD)"
printf 'GEN_STAGE_READY=%s\n' "$stage"
