#!/bin/bash
set -euo pipefail
cd /home/runner/workspace/ai-agent
node --input-type=module -e '
import { stat, open, readFile } from "node:fs/promises";
const runtime="/home/runner/workspace/.ai-agent-tri-repo-runtime";
const log=runtime+"/supervisor.log";
const st=await stat(log);
const size=Math.min(st.size,8*1024*1024);
const f=await open(log,"r");let raw;
try {const b=Buffer.alloc(size);const n=await f.read(b,0,size,st.size-size);raw=b.toString("utf8",0,n.bytesRead);} finally {await f.close();}
const rows=raw.split("\n").filter(x=>x.startsWith("[tri-repo] "));
const counts={};const problems=[];
for(const line of rows){const m=line.match(/^\[tri-repo\] (Gen|Project-Delivery|Migration) (\{.*\})$/);if(!m)continue;let obj;try{obj=JSON.parse(m[2]);}catch{continue;}const key=m[1]+":"+obj.status;counts[key]=(counts[key]||0)+1;if(/error|fail|defer|block|quarant|incomplete|missing|conflict/i.test(obj.status)||obj.error)problems.push({worker:m[1],at:obj.at,status:obj.status,error:obj.error??null,blockedCount:obj.blockedCount??null,deferredCount:obj.deferredCount??null});}
const status=JSON.parse(await readFile(runtime+"/status.json","utf8"));
const gen=JSON.parse(await readFile("/home/runner/workspace/.gen-sync-state.json","utf8"));
const bl=JSON.parse(await readFile("backend/sync/.project-delivery-blacklist.json","utf8"));
const now=Date.now();
const deferred=Object.entries(gen.deferred??{}).map(([p,v])=>({path:p,reason:v.reason,retryInMinutes:Math.max(0,Math.ceil((v.until-now)/60000))}));
const blacklist=Object.entries(bl.files??{}).map(([p,v])=>({path:p,reason:typeof v==="object"?v.reason:String(v)}));
console.log("LOG_AUDIT",JSON.stringify({logSizeBytes:st.size,parsedEntries:rows.length,counts,recentEntries:rows.slice(-14),recentProblems:problems.slice(-14),currentWorkers:status.workers,genDeferredCount:deferred.length,genDeferred:deferred.slice(0,12),deliveryBlacklistCount:blacklist.length,deliveryBlacklist:blacklist.slice(0,12)}));
'
printf 'PROCESS\n'
node backend/sync/tri_repo_supervisor.js status
printf 'GEN_DIVERGENCE_HEAD_REMOTE\n'
git -C /home/runner/workspace/.gen-sync-cache rev-list --left-right --count HEAD...origin/main
