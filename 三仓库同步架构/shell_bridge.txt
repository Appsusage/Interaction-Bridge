#!/bin/bash
set -euo pipefail
R=/home/runner/workspace/ai-agent
C=/home/runner/workspace/.gen-sync-cache
B=/home/runner/workspace/.ai-agent-tri-repo-runtime
git -C "$C" merge-base --is-ancestor a3a5a001bc5b1ecf0e696f26b2dbedb8c9085922 HEAD
P1=backend/sync/migration_repository_sync.js
P2=backend/sync/gen_bidirectional_sync.js
P3=backend/sync/tests/gen_bidirectional_sync.test.js
P4=backend/sync/migration_key_custody.js
P5=backend/sync/tests/migration_key_custody.test.js
for P in "$P1" "$P2" "$P3" "$P4" "$P5"; do test -f "$C/$P"; done
cmp -s "$R/$P1" <(git -C "$C" show 8f00fc993bab2a04c31ee4c39e2a7b59a8250598:"$P1")
cmp -s "$R/$P2" <(git -C "$C" show 57e28079088b6e62459f3f3ba3140b19df17a7ab:"$P2")
cmp -s "$R/$P3" <(git -C "$C" show 57e28079088b6e62459f3f3ba3140b19df17a7ab:"$P3")
test ! -e "$R/$P4"
test ! -e "$R/$P5"
STAMP="$(date +%Y%m%d%H%M%S)"
D="$B/priority-code-backup-$STAMP"
mkdir -p "$D/backend/sync/tests"
for P in "$P1" "$P2" "$P3"; do cp -p "$R/$P" "$D/$P"; done
for P in "$P4" "$P5" "$P1" "$P2" "$P3"; do
  T="$R/$P.priority-$STAMP.tmp"
  cp -p "$C/$P" "$T"
  mv "$T" "$R/$P"
done
cd "$R"
node --check "$P4"
node --check "$P1"
node --check "$P2"
node "$P5"
node backend/sync/tests/gen_bidirectional_sync.test.js
printf 'PRIORITY_CODE_INSTALLED backup=%s\n' "$D"
