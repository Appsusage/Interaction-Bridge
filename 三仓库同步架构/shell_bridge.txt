#!/bin/bash
set -uo pipefail
C=/home/runner/workspace/.gen-sync-cache
printf 'GEN DIRTY COUNT\n'
git -C "$C" status --porcelain --untracked-files=all | wc -l
printf 'GEN DIRTY TOP\n'
git -C "$C" status --porcelain --untracked-files=all | head -n 35
printf 'GEN TRACKED DIFF\n'
git -C "$C" diff --name-status | head -n 35
printf 'GEN STAGED DIFF\n'
git -C "$C" diff --cached --name-status | head -n 35
printf 'GEN STATUS SUMMARY\n'
git -C "$C" status -sb -uno
printf 'SUPERVISOR PROCESS\n'
ps -p 156 -o pid=,etime=,args=
printf 'SUPERVISOR STDOUT DEST\n'
readlink /proc/156/fd/1 || true
