#!/bin/bash
set -u
C=/home/runner/workspace/.gen-sync-cache
echo 'CACHE STATUS:'
git -C "$C" status --short --untracked-files=all | head -n 10
echo 'LOCK FILE:'
stat -c 'size=%s age_seconds=%Y mtime=%y' "$C/.git/index.lock" 2>&1 || true
echo 'LOCK OWNER CHECK:'
if command -v fuser >/dev/null 2>&1; then fuser "$C/.git/index.lock" 2>&1 || true; fi
echo 'RUNNING GIT/SYNC PROCESSES:'
ps -eo pid,etime,args | rg 'git (add|commit|restore|push|fetch|merge|lfs)|tri_repo_supervisor|gen_bidirectional_sync' | head -n 15 || true
echo 'REPLIT MODULE VERSION:'
wc -c /home/runner/workspace/ai-agent/backend/sync/gen_bidirectional_sync.js
echo 'TARGET BACKUP:'
stat -c '%s bytes backed up' /home/runner/workspace/.gen-sync-backups/tri-repo-recovery-20261027/.chatgpt-profile/Default/DawnGraphiteCache/data_1 2>&1 || true
