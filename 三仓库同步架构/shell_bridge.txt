#!/bin/bash
set -euo pipefail
cd /home/runner/workspace/ai-agent
node backend/sync/tri_repo_supervisor.js stop
READY=0
for I in $(seq 1 25); do
  S="$(node backend/sync/tri_repo_supervisor.js status)"
  if printf '%s' "$S" | grep -q '"status":"stopped"'; then READY=1; break; fi
  sleep 2
done
if [ "$READY" -ne 1 ]; then printf 'BATCH_RELOAD_TIMEOUT\n'; exit 4; fi
node backend/sync/tri_repo_supervisor.js start
node backend/sync/tri_repo_supervisor.js status
