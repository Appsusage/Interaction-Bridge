function check(b,p,n){if(!Number.isSafeInteger(p)||p<0||p+n>b.length)throw Error("Out of bounds");}
function U16(b,p){check(b,p,2);return b.readUInt16LE(p);}
function U32(b,p){check(b,p,4);return b.readUInt32LE(p);}
async function range(spec,limit=MAX_C){
  const r=await fetch(URL,{headers:{Range:spec},signal:AbortSignal.timeout(35000)});
  if(r.status!==206){await r.body?.cancel();throw Error("Range unsupported: "+r.status);}
  const m=/^bytes (\d+)-(\d+)\/(\d+)$/.exec(r.headers.get("content-range")||"");
  if(!m)throw Error("Bad Content-Range");
  const n=Number(m[2])-Number(m[1])+1;
  if(n<1||n>limit)throw Error("Range too large: "+n);
  const b=Buffer.from(await r.arrayBuffer());
  if(b.length!==n)throw Error("Incomplete Range");
  return {b,total:Number(m[3])};
}
async function getDexEntries(){
  const tail=await range("bytes=-65557",65557);let eo=-1;
  for(let p=tail.b.length-22;p>=0;p--){

  }

  }

}
async function readDex(entry){
  if(entry.compressed>MAX_C||entry.uncompressed>MAX_U||(entry.flags&1))throw Error("Oversized/encrypted DEX");
  const h=(await range("bytes="+entry.local+"-"+(entry.local+29),30)).b;
  if(U32(h,0)!==0x04034b50)throw Error("Bad ZIP local header");
  const off=entry.local+30+U16(h,26)+U16(h,28);
  const comp=(await range("bytes="+off+"-"+(off+entry.compressed-1),MAX_C)).b;
  const b=entry.method===0?comp:entry.method===8?inflateRawSync(comp,{maxOutputLength:MAX_U}):null;
  if(!b||b.length!==entry.uncompressed||b.toString("utf8",0,4)!=="dex\n")
    throw Error("Not valid decompressed DEX");
  return b;
}
function uleb(b,at){
  let v=0;
  for(let shift=0;shift<35;shift+=7){
    check(b,at,1);const x=b[at++];
    v+=(x&127)*2**shift;
    if(!(x&128))return [v,at];
  }
  throw Error("Bad ULEB");
}
function createDex(b){
  const strN=U32(b,0x38),strO=U32(b,0x3c),typeN=U32(b,0x40),typeO=U32(b,0x44);
  const methodN=U32(b,0x58),methodO=U32(b,0x5c),classN=U32(b,0x60),classO=U32(b,0x64);
  if(strN>3000000||typeN>65535||methodN>65535||classN>65535)
    throw Error("DEX indices too large");
  check(b,strO,4*strN);check(b,typeO,4*typeN);check(b,methodO,8*methodN);check(b,classO,32*classN);
  const cache=new Map(),typeCache=new Map(),methodCache=new Map();
  function str(i){
    if(i>=strN)return "";
    if(cache.has(i))return cache.get(i);
    const off=U32(b,strO+4*i);
    const [len,at]=uleb(b,off);
    if(len>512)return "";
    let end=at;
    while(end<b.length&&end-at<600&&b[end]!==0)end++;
    if(end===b.length||end-at>=600)return "";
    const value=b.toString("utf8",at,end);
    cache.set(i,value);return value;
  }
  function type(i){
    if(i>=typeN)return "";
    if(typeCache.has(i))return typeCache.get(i);
    const value=str(U32(b,typeO+4*i));
    typeCache.set(i,value);return value;
  }
  function method(i){
    if(i>=methodN)return null;
    if(methodCache.has(i))return methodCache.get(i);
    const off=methodO+8*i;
    const result={class:type(U16(b,off)),name:str(U32(b,off+4)),idx:i};
    methodCache.set(i,result);return result;
  }
  function classMethods(classData){
    if(!classData)return [];
    let p=classData;const sizes=[];
    for(let i=0;i<4;i++){const [n,next]=uleb(b,p);sizes.push(n);p=next;}
    if(sizes.some(n=>n>50000))throw Error("Malformed class_data sizes");
    for(let section=0;section<2;section++){
      for(let i=0;i<sizes[section];i++){p=uleb(b,p)[1];p=uleb(b,p)[1];}
    }
    const list=[];
    for(let section=2;section<4;section++){
      let mid=0;
      for(let i=0;i<sizes[section];i++){
        let delta;[delta,p]=uleb(b,p);mid+=delta;
        p=uleb(b,p)[1];
        let code;[code,p]=uleb(b,p);
        if(mid<methodN)list.push({mid,code});
      }
    }
    return list;
  }
  // Dalvik instruction widths are in 16-bit code units.
  function width(b,at,codeEnd){
    const code=U16(b,at);const op=code&255;
    if(op===0){
      const hi=code>>>8;
      if(hi===1)return 4+2*U16(b,at+2);
      if(hi===2)return 2+4*U16(b,at+2);
      if(hi===3)return 4+Math.ceil(U16(b,at+2)*U32(b,at+4)/2);
      return 1;
    }
    if([0x03,0x06,0x09,0x14,0x17,0x1b,0x24,0x25,0x26,0x2a,0x2b,0x2c,0xfc,0xfd].includes(op))return 3;

       (op>=0x90&&op<=0xaf)||(op>=0xd0&&op<=0xe2))return 2;
    if((op>=0x6e&&op<=0x72)||(op>=0x74&&op<=0x78))return 3;
    return 1;
  }
  function invokes(code){
    if(!code)return [];
    check(b,code,16);
    const insns=U32(b,code+12),start=code+16,end=start+2*insns;
    if(insns>100000||end>b.length)throw Error("Invalid DEX method code size");

      }

    }

  }
  function traceInstructions(code) {
    check(b,code,16);

    if(total>100000)throw Error("Oversized method");
    const start=code+16,end=start+2*total;
    if(end>b.length)throw Error("Truncated instructions");

      const n=width(b,p,end);
      if(n<1||p+n*2>end)break;
      const pc=(p-start)/2;
      if(op===0x12){const raw=opword>>>12;events.push({pc,op:"const/4",value:raw>=8?raw-16:raw});}
      else if(op===0x13||op===0x16){events.push({pc,op:op===0x13?"const/16":"const-wide/16",value:b.readInt16LE(p+2)});}
      else if(op===0x14){events.push({pc,op:"const/32",value:b.readInt32LE(p+2)});}
      else if(op===0x15){events.push({pc,op:"const/high16",value:b.readInt16LE(p+2)*65536});}
      else if(op===0x17){events.push({pc,op:"const-wide/32",value:b.readInt32LE(p+2)});}
      else if(op===0x23){events.push({pc,op:"new-array"});}
      else if(op===0x24||op===0x25){events.push({pc,op:"filled-new-array"});}
      else if(op===0x4f||op===0x4b){events.push({pc,op:op===0x4f?"aput-byte":"aput"});}
      else if((op>=0x6e&&op<=0x72)||(op>=0x74&&op<=0x78)||op===0xfa||op===0xfb){
        const ref=method(U16(b,p+2));
        events.push({pc,op:"invoke",target:ref?ref.class+"->"+ref.name:"UNKNOWN"});
      }
      p+=2*n;
    }
    return {codeUnits:total,events:events.slice(0,180)};
  }
  return {classN,classO,type,method,classMethods,invokes,traceInstructions,U32:(p)=>U32(b,p)};
}
