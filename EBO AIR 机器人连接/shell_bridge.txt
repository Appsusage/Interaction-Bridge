 platform:readiness.runtime.platform,arch:readiness.runtime.arch,
 libraryCount:Object.keys(readiness.nativeLibraries).length,
 loaderAbiVerified:readiness.prerequisites.nativeLoaderAbiVerified,
 actualRobotAuthenticated:status.p2p.authenticated,
