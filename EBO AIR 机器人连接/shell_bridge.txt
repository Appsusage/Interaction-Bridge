import fs from "node:fs";
const x=JSON.parse(fs.readFileSync(process.argv[2],"utf8"));
for(const c of x.targets){
 const events=c.instructions?.events||[];
 console.log("EBO_TRANSPORT_METHOD",JSON.stringify({owner:c.class,method:c.method,codeUnits:c.instructions?.codeUnits,
   interestingInvokes:events.filter(e=>e.op==='invoke').map(e=>e.target).filter(e=>/RDT|AVAPIs|Packet|tutk|Coroutine/i.test(e)).slice(0,36),
   instructions:events.filter(e=>e.op==="invoke"&&/RDT_Write|RDT_Create|avSendIOCtrl|Packet|m;->b/i.test(e.target||"") ||
      e.op.startsWith("const")).slice(0,85)}));
}
console.log('EBO_TRANSPORT_TOTAL_METHODS',x.targets.length);
NODE
