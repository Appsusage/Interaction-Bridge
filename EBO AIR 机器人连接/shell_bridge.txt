console.log("EBO_TUTK_HANDLER_COUNT",x.targets.length);
for(const z of x.targets){
 const events=(z.instructions?.events||[]);
 const matches=events.filter(v=>v.op==="invoke" && /RDT_Write|RDT_Create|avSendIOCtrl|camera\/tutk\/m;->b/.test(v.target||""));
 console.log("EBO_TUTK_HANDLER",JSON.stringify({class:z.class,method:z.method,codeUnits:z.instructions?.codeUnits,transportCalls:matches}));
}
