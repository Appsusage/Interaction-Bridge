set -eu
echo EBO_MAVLINK_CRC_INSPECT_START
tmp="$(mktemp -d)"
trap 'rm -rf "$tmp"' EXIT
gh api --hostname github.com "repos/Appsusage/Gen/contents/backend/devices/ebo/research/trace_rola_mavlink_crc.mjs" --jq '.content' | base64 -d > "$tmp/scan.mjs"
node --check "$tmp/scan.mjs"
node "$tmp/scan.mjs" > "$tmp/out.json"
node --input-type=module - "$tmp/out.json" <<'NODE'
import fs from 'node:fs';
const x=JSON.parse(fs.readFileSync(process.argv[2],'utf8'));
console.log('MAVLINK_CRC_METHOD_COUNT',x.targets.length);
for(const z of x.targets) {
 const y=(z.instructions?.events||[]);
 console.log('MAVLINK_CRC_METHOD',JSON.stringify({
  class:z.class,method:z.method,codeUnits:z.instructions?.codeUnits,
  relevant:y.filter(e=>e.op==='invoke'&& /crc|checksum|encode|pack|byte|Lu\/b;/.test(e.target||'')).slice(0,32),
  numeric:y.filter(e=>e.op.startsWith('const')).slice(0,45)
 }));
}
NODE
echo EBO_MAVLINK_CRC_INSPECT_COMPLETE
