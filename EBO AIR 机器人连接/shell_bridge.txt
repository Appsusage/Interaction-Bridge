set -eu
echo EBO_READONLY_OBSERVER_VERIFY_BEGIN
tmp="$(mktemp -d)"
trap 'rm -rf "$tmp"' EXIT
for rel in backend/devices/ebo/ebo_p2p_observer.js test/ebo_p2p_observer.test.mjs; do
 mkdir -p "$tmp/$(dirname "$rel")"
 gh api "repos/Appsusage/Gen/contents/$rel" --jq '.content' | base64 -d > "$tmp/$rel"
 test -s "$tmp/$rel"
done
node --check "$tmp/backend/devices/ebo/ebo_p2p_observer.js"
(cd "$tmp" && node --test test/ebo_p2p_observer.test.mjs)
node --input-type=module - "$tmp/backend/devices/ebo/ebo_p2p_observer.js" <<'NODE'
import {pathToFileURL} from 'node:url';
const mod=await import(pathToFileURL(process.argv[2]));
const cfg=mod.getEboP2PObserverConfig();
const report=await mod.probeEboP2PObserver();
console.log('EBO_NATIVE_OBSERVER_READ_ATTEMPT',JSON.stringify({
 observerConfigured:cfg.observerConfigured,
 observerVerified:report.observerVerified,
 observerState:report.nativeSidecarReportedState||null,
 readOnlyReplies:report.nativeSidecarReportedReadOnlyReplies||0,
 robotAuthenticated:report.robotAuthenticated,
 code:report.code,
 safeToSend:report.safeToSend
}));
NODE
echo EBO_READONLY_OBSERVER_VERIFY_COMPLETE
