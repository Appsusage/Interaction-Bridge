set -eu
cd /home/runner/workspace/ai-agent
echo EBO_PUBLIC_WSS_PROBE_START
if [ -z "${REPLIT_DEV_DOMAIN:-}" ]; then echo EBO_DOMAIN_UNSET; exit 2; fi
curl -sS -o /dev/null --max-time 10 -w 'EBO_PUBLIC_CLIENT_HTTP:%{http_code} CONTENT_TYPE:%{content_type}\n' "https://${REPLIT_DEV_DOMAIN}/ebo/bridge-client.mjs" || true
node --input-type=module <<'NODE'
import WebSocket from 'ws';
const domain=process.env.REPLIT_DEV_DOMAIN;
const ws=new WebSocket('wss://'+domain+'/ebo/bridge',{handshakeTimeout:6000,headers:{Origin:'https://'+domain}});
let done=false;
const timer=setTimeout(()=>{if(done)return;done=true;ws.terminate();console.log('EBO_PUBLIC_WSS_TIMEOUT');process.exitCode=3},9000);
ws.on('open',()=>{console.log('EBO_PUBLIC_WSS_UPGRADED');ws.send(JSON.stringify({type:'auth',token:'invalid-test-value'}));});
ws.on('close',code=>{if(done)return;done=true;clearTimeout(timer);console.log('EBO_PUBLIC_WSS_CLOSE',code); if(code!==1008) process.exitCode=4;});
ws.on('error',err=>{if(done)return;done=true;clearTimeout(timer);console.log('EBO_PUBLIC_WSS_ERROR',err.message.slice(0,150));process.exitCode=5;});
NODE
echo EBO_PUBLIC_WSS_PROBE_END
