cd /home/runner/workspace/ai-agent
printf '%s\n' '=== S422 authenticated HeartBeat start test ==='
server_pid="$(ps -eo pid,args | awk '$0 ~ /node backend\/server\.js/ && $0 !~ /awk/ {print $1; exit}')"
if [ -z "$server_pid" ] || [ ! -r "/proc/$server_pid/environ" ]; then
  echo 'server process=NOT_FOUND'
  exit 1
fi
token="$(tr '\0' '\n' <"/proc/$server_pid/environ" | sed -n 's/^HEARTBEAT_SUPERVISOR_TOKEN=//p' | head -n 1)"
if [ -z "$token" ]; then
  echo 'server token=MISSING'
  exit 1
fi
echo 'server token=SET'
curl -sS --max-time 15 -X POST -H "Authorization: Bearer $token" -H 'Content-Type: application/json' http://127.0.0.1:3000/heartbeat/start -o /tmp/s422-start.json -w 'start HTTP %{http_code}\n' || true
cat /tmp/s422-start.json 2>/dev/null || true
unset token
printf '\n'
curl -sS --max-time 8 -o /tmp/s422-heartbeat.json -w 'heartbeat HTTP %{http_code}\n' http://127.0.0.1:3000/heartbeat/health || true
cat /tmp/s422-heartbeat.json 2>/dev/null || true
printf '\n'
