const url = "http://127.0.0.1:3000/chat";
const payload = {
  message: "请实际使用系统已集成的 Playwright Browser 工具完成只读测试：打开 https://example.com，读取真实网页标题和正文，并用中文汇报。严格按顺序执行：1) 只创建一次 browser_new_session；2) 必须使用该动作真实返回的 sessionId 执行 browser_open；3) 使用同一个 sessionId 执行 browser_read；4) 任务完成后用同一个 sessionId 执行 browser_close_session。禁止重复创建会话，禁止在未读取真实结果前声称成功。如果任何一步失败，请如实报告错误，不要重试创建新会话。不要登录或提交表单。",
