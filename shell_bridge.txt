const groups = [
  { type: "shell_command", words: ["shell bridge", "inbox", "outbox", "执行命令", "运行命令", "终端命令"] },
  { type: "file_operation", words: ["创建文件", "读取文件", "复制文件", "移动文件", "删除文件", "文件夹"] },
  { type: "coding", words: ["代码", "bug", "报错", "debug", "javascript", "node.js", "函数", "测试", "编程"] },
  { type: "analysis", words: ["分析", "比较", "原因", "检查", "诊断", "架构"] },
  { type: "conversation", words: ["你好", "谢谢", "早安", "晚安", "再见"] }
];

export function classifyTask(input = "") {
  const text = String(input).toLowerCase();
  for (const group of groups) {
    if (group.words.some(word => text.includes(word.toLowerCase()))) {
      return { type: group.type, matchedBy: "keyword_rule" };
    }
  }
  return { type: "general", matchedBy: "default_rule" };
}

export function classifyLearningIntent(input = "") {
  const task = classifyTask(input).type;
  if (task === "coding" || task === "shell_command" || task === "file_operation" || task === "analysis") return "technical";
  if (task === "conversation") return "general";
  return "conversation";
}
EOF

import { classifyTask, classifyLearningIntent } from "../task_classifier/task_classifier.js";

export function analyzeInput(input = {}) {
  const message = String(input.message ?? "");
  const task = classifyTask(message);
  const learningIntent = classifyLearningIntent(message);
  const complexity = message.length > 1200 ? "high" : message.length > 300 ? "medium" : "low";

  return {
    message,
    taskType: task.type,
    matchedBy: task.matchedBy,
    intentForLearning: learningIntent,
    complexity,
    hasContext: Boolean(input.context),
    mode: input.mode || "chat"
  };
}

cat > "$BASE/rules/rule_engine.js" <<'EOF'
export function applyRules(analysis) {
  if (!analysis || typeof analysis !== "object") {
    return { action: "ask_for_clarification", needsModel: false, reason: "缺少有效分析结果" };
  }

  if (analysis.taskType === "shell_command") {
    return { action: "prepare_shell_task", needsModel: false, reason: "Shell 任务可由本地规则整理；执行前仍须检查命令与权限" };
  }

  if (analysis.taskType === "file_operation") {
    return { action: "prepare_file_task", needsModel: false, reason: "文件任务可先由本地规则分类和整理" };
  }

  if (analysis.taskType === "coding" || analysis.taskType === "analysis") {
    return { action: "check_experience_first", needsModel: false, reason: "先检索经验；经验不足时再决定是否请求外部协助" };
  }

  if (analysis.taskType === "conversation") {
    return { action: "local_classification", needsModel: false, reason: "简单寒暄可以由规则识别" };
  }

  return { action: "check_experience_then_escalate", needsModel: false, reason: "先查经验库；本模块不会自行调用任何 LLM" };
}
EOF

export function selectModel(analysis, ruleDecision) {
  if (ruleDecision?.needsModel !== true) {
    return {
      required: false,
      provider: null,
      route: ruleDecision?.action === "check_experience_then_escalate" ? "experience_or_external_ai_bridge" : "local",
      reason: "当前决策不要求调用模型"
    };
  }

  return {
    required: true,
    provider: null,
    route: "model_registry_required",
    taskType: analysis?.taskType || "general",
    reason: "只返回选择建议；不在本模块内发起模型调用"
  };
}

cat > "$BASE/experience/experience_store.js" <<'EOF'
import fs from "node:fs";
import path from "node:path";
import { fileURLToPath } from "node:url";

const here = path.dirname(fileURLToPath(import.meta.url));
export const experienceFile = path.join(here, "experience.json");

export function readExperiences() {
  try {
    const data = JSON.parse(fs.readFileSync(experienceFile, "utf8"));
    return Array.isArray(data) ? data : [];
  } catch (error) {
    if (error.code === "ENOENT") return [];
    throw error;
  }
}

export function findExperiences(query, limit = 5) {
  const terms = String(query || "").toLowerCase().split(/\s+/).filter(Boolean);
  if (!terms.length) return [];
  return readExperiences()
    .map(item => {
      const haystack = [item.pattern, item.taskType, item.lesson, ...(item.keywords || [])].join(" ").toLowerCase();
      const score = terms.reduce((n, term) => n + (haystack.includes(term) ? 1 : 0), 0);
      return { item, score };
    })
    .filter(entry => entry.score > 0)
    .sort((a, b) => b.score - a.score)
    .slice(0, limit)
    .map(entry => ({ ...entry.item, matchScore: entry.score }));
}

export function saveExperience(experience) {
  if (!experience || typeof experience !== "object") throw new TypeError("experience must be an object");
  const entries = readExperiences();
  const id = experience.id || `exp-${Date.now()}`;
  const next = { ...experience, id, updatedAt: new Date().toISOString() };
  const index = entries.findIndex(entry => entry.id === id);
  if (index >= 0) entries[index] = next;
  else entries.push(next);
  fs.writeFileSync(experienceFile, JSON.stringify(entries, null, 2) + "\n", "utf8");
  return next;
}
EOF

import { saveExperience } from "../experience/experience_store.js";

const usefulSources = new Set([
  "shell_command",
  "shell_raw_log",
  "codex_execution",
  "tool_execution",
  "external_ai_response",
  "prompt_master_decision",
  "execution_result",
  "user_operation"
]);

export function evaluateLearningCandidate(event = {}) {
  const source = String(event.source || "");
  const hasOutcome = event.success === true || event.success === false ||
    Number.isInteger(event.exitCode) || Boolean(event.outcome);
  const usefulText = Boolean(event.command || event.error || event.lesson || event.result);

  if (!usefulSources.has(source) || !hasOutcome || !usefulText) {
    return { accepted: false, reason: "缺少受支持的数据来源、执行结果或可分析内容" };
  }

  return {
    accepted: true,
    candidate: {
      id: event.id,
      source,
      taskType: event.taskType || "unknown",
      pattern: event.pattern || event.command || event.error || "execution_event",
      keywords: Array.isArray(event.keywords) ? event.keywords : [],
      lesson: event.lesson || event.result || event.error || "",
      successCount: event.success === true || event.exitCode === 0 ? 1 : 0,
      failureCount: event.success === false || (Number.isInteger(event.exitCode) && event.exitCode !== 0) ? 1 : 0,
      verified: false,
      evidence: {
        command: event.command || null,
        exitCode: Number.isInteger(event.exitCode) ? event.exitCode : null,
        outcome: event.outcome || null,
        timestamp: event.timestamp || new Date().toISOString()
      }
    }
  };
}

export function ingestExecutionEvent(event) {
  const evaluated = evaluateLearningCandidate(event);
  if (!evaluated.accepted) return evaluated;
  return {
    accepted: true,
    saved: saveExperience(evaluated.candidate),
    note: "已保存为待验证经验；不会自动升级为可信规则"
  };
}
EOF

cat > "$BASE/executor/executor.js" <<'EOF'
export function prepareShellTask(command, options = {}) {
  if (typeof command !== "string" || !command.trim()) {
    throw new TypeError("command must be a non-empty string");
  }

  return {
    filename: options.filename || `PM-${Date.now()}.txt`,
    target: "Shell Bridge inbox",
    content: command.trim() + "\n",
    requiresApproval: options.requiresApproval !== false,
    dispatched: false,
    note: "仅生成任务内容；本函数不会上传到 Dropbox，也不会直接执行命令"
  };
}

export function createExecutionPlan(analysis, ruleDecision, modelDecision) {
  return {
    analysis,
    decision: ruleDecision,
    model: modelDecision,
    executed: false,
    createdAt: new Date().toISOString()
  };
}
EOF

import { analyzeInput } from "./analyzer/analyzer.js";
import { applyRules } from "./rules/rule_engine.js";
import { selectModel } from "./model_selector/model_selector.js";
import { findExperiences } from "./experience/experience_store.js";
import { createExecutionPlan } from "./executor/executor.js";

export async function optimizePrompt(input = {}) {
  const analysis = analyzeInput(input);
  const experiences = findExperiences(input.message || "", 5);
  const ruleDecision = applyRules(analysis);
  const modelDecision = selectModel(analysis, ruleDecision);
  const plan = createExecutionPlan(analysis, ruleDecision, modelDecision);

  const masterResult = {
    type: analysis.mode === "typing_predict" ? "prediction" : analysis.mode === "autonomy" ? "autonomy" : "reply",
    intent_for_learning: analysis.intentForLearning,
    task: analysis.message,
    cleaned_context: [
      "Prompt Master 本地分析结果（未调用 LLM）",
      `任务类型：${analysis.taskType}`,
      `复杂度：${analysis.complexity}`,
      `决策：${ruleDecision.action}`,
      `理由：${ruleDecision.reason}`,
      `匹配经验：${experiences.length} 条`,
      input.context ? `已有上下文：\n${input.context}` : ""
    ].filter(Boolean).join("\n"),
    taskType: analysis.taskType,
    action: ruleDecision.action,
    needsModel: modelDecision.required,
    modelRoute: modelDecision.route,
    confidence: analysis.matchedBy === "keyword_rule" ? 0.65 : 0.25,
    matchedExperiences: experiences,
    plan
  };

  return {
    system: input.system || "",
    context: masterResult.cleaned_context,
    message: input.message || "",
    intent_for_learning: masterResult.intent_for_learning,
    task: masterResult.task,
    masterResult
  };
}

export { analyzeInput } from "./analyzer/analyzer.js";
export { applyRules } from "./rules/rule_engine.js";
export { classifyTask, classifyLearningIntent } from "./task_classifier/task_classifier.js";
export { selectModel } from "./model_selector/model_selector.js";
export { findExperiences, readExperiences, saveExperience } from "./experience/experience_store.js";
export { evaluateLearningCandidate, ingestExecutionEvent } from "./learning/learning_engine.js";
export { prepareShellTask, createExecutionPlan } from "./executor/executor.js";
EOF
