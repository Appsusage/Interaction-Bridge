import express from 'express';
import {createServer} from 'node:http';
import {readFileSync, writeFileSync, unlinkSync, existsSync} from 'node:fs';
import {join,dirname} from 'node:path';
import {fileURLToPath} from 'node:url';
import {installMultiTaskShellFloats} from './shell_floats.js';

const base=dirname(fileURLToPath(import.meta.url));
const pidFile=join(base,'.multi-task-server.pid');
const pidNumber=Number(existsSync(pidFile)?readFileSync(pidFile,'utf8').trim():0);
if(Number.isSafeInteger(pidNumber)&&pidNumber>0){
  let cmd='';
  try{cmd=readFileSync('/proc/'+pidNumber+'/cmdline','utf8');}catch{}
  if(cmd.includes('bridge/shell/multi-task/server.js'))throw Error('V2 worker already running pid '+pidNumber);
  unlinkSync(pidFile);
}
writeFileSync(pidFile,String(process.pid)+'\n',{flag:'wx'});
const app=express();
app.use(express.json({limit:'64kb'}));
const server=createServer(app);
const consoleState=installMultiTaskShellFloats(app,server);
const port=Number(process.env.MULTI_SHELL_PORT||3102);
app.get('/health',(_req,res)=>res.json({ok:true,mode:'multi-task',pid:process.pid,...consoleState.getStatus()}));
app.get('/shell-multi/console',(_req,res)=>res.sendFile(join(base,'console.html')));
server.on('error',error=>{console.error('[multi-shell] server error',error);process.exitCode=1;});
let stopping=false;
const origin='http://127.0.0.1:'+port;
const action=async verb=>{
  const response=await fetch(origin+'/shell-multi/'+verb,{method:'POST',headers:{Origin:origin}});
  const data=await response.json();
  if(!response.ok)throw Error(JSON.stringify(data));
  return data;
};
server.listen(port,'0.0.0.0',async()=>{
  try{
    console.log('[multi-shell] listening '+port+' pid='+process.pid);
    console.log('[multi-shell] startup '+JSON.stringify(await action('start')));
  }catch(error){console.error('[multi-shell] failed to start',error);process.exitCode=1;server.close();}
});
process.on('SIGTERM',async()=>{
  if(stopping)return;stopping=true;
  console.log('[multi-shell] stopping intake');
  try{await action('stop');}catch(error){console.error('[multi-shell] stop',error);}
  const drain=()=>{
    if(consoleState.getStatus().active){setTimeout(drain,1000);return;}
    server.close(()=>process.exit(0));
  };
  drain();
});
process.once('exit',()=>{try{if(readFileSync(pidFile,'utf8').trim()===String(process.pid))unlinkSync(pidFile);}catch{}});

 const key=info.id||info.folder+'/'+info.taskName;let entry=tasks.get(key);
 if(!entry){
