from pathlib import Path
p=Path("backend/extension/pet/limbo/pet.js")
s=p.read_text()
old='''    face.style.transform = "";
    orb.querySelectorAll(".limbo-pupil").forEach(p => { p.style.transform = ""; });
'''
new='''    face.style.transform = "";
    // Move the newborn eye's pupil in all four directions while keeping it inside the eye.
    const newbornPupil = orb.querySelector(".limbo-pet-newborn .limbo-eye.left .limbo-pupil");
    if (newbornPupil) {
      const pupilPosition = {
        left: [1, 4], right: [7, 4], up: [4, 1], down: [4, 8], forward: [4, 4]
      }[gaze] || [4, 4];
      orb.style.setProperty("--newborn-pupil-x", pupilPosition[0] + "px");
      orb.style.setProperty("--newborn-pupil-y", pupilPosition[1] + "px");
    }
    orb.querySelectorAll(".limbo-pupil").forEach(p => { p.style.transform = ""; });
'''
if old not in s:
    raise SystemExit("SET_GAZE_TARGET_NOT_FOUND; no JS change made")
if "Move the newborn eye's pupil in all four directions" not in s:
    s=s.replace(old,new,1)
    p.write_text(s)
    print("NEWBORN_PUPIL_GAZE_LOGIC_APPLIED")
else:
    print("NEWBORN_PUPIL_GAZE_LOGIC_ALREADY_PRESENT")
PY
