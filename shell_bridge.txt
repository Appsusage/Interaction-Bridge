cd /home/runner/workspace/ai-agent
node --input-type=module <<'EOF'
import { createBrowserSession, getBrowserPage, openPage, readPage, closeBrowserSession } from "./backend/extension/browser/browser_service.js";
let session;
try {
  session = await createBrowserSession();
  console.log("SESSION_CREATED", JSON.stringify(session));
  const page = getBrowserPage(session.sessionId);
  console.log("OPEN_RESULT", JSON.stringify(await openPage(page, "https://example.com")));
  const read = await readPage(page);
  console.log("READ_RESULT", JSON.stringify({ title: read.title, textLength: read.text.length, hasExample: read.text.includes("Example Domain") }));
} catch (error) {
  console.error("BROWSER_SERVICE_ERROR", error.stack || error);
  process.exitCode = 1;
} finally {
  if (session?.sessionId) {
    try { console.log("CLOSE_RESULT", JSON.stringify(await closeBrowserSession(session.sessionId))); }
    catch (error) { console.error("CLOSE_ERROR", error.stack || error); process.exitCode = 1; }
  }
}
EOF
