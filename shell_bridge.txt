#!/bin/bash
set +e
cd /home/runner/workspace/ai-agent || exit 1
echo "=== float visibility / show-entry source ==="
sed -n '1870,2030p' frontend/app.js
echo "=== float styles and dock toggle markup ==="
grep -n -E 'float-main|dock-toggle|dockToggle|float-menu|\.open|display:none' frontend/index.html | head -100
echo "=== headed browser: activate dock entry, then real drag if visible ==="
node --input-type=module <<'NODE'
import {createBrowserSession,getBrowserPage,closeBrowserSession} from './backend/extension/browser/browser_service.js';
let id;const out={};
const state=async p=>p.evaluate(()=>{const ids=['mainFloat','dockToggle','floatMenu'];return Object.fromEntries(ids.map(id=>{const e=document.getElementById(id);if(!e)return[id,null];const r=e.getBoundingClientRect();return[id,{cls:e.className,display:getComputedStyle(e).display,rect:{x:r.x,y:r.y,w:r.width,h:r.height},html:e.outerHTML.slice(0,300)}]}))});
try{
 const s=await createBrowserSession({headless:false});id=s.sessionId;const p=getBrowserPage(id);
 await p.setViewportSize({width:900,height:700});await p.goto('http://127.0.0.1:3000/',{waitUntil:'domcontentloaded',timeout:15000});await p.waitForTimeout(1000);
 out.before=await state(p);
 const dock=p.locator('#dockToggle');
 out.dockCount=await dock.count();
 if(await dock.count()){try{await dock.click({force:true,timeout:2500});out.dockClicked=true}catch(e){out.dockClickError=String(e)}}
 await p.waitForTimeout(700);out.afterDock=await state(p);
 const f=await p.locator('#mainFloat').boundingBox();
 if(f&&f.width>0&&f.height>0){
  const x=f.x+f.width/2,y=f.y+f.height/2;
  await p.mouse.move(x,y);await p.mouse.down();await p.waitForTimeout(450);await p.mouse.move(x+130,y+100,{steps:14});await p.waitForTimeout(400);await p.mouse.up();await p.waitForTimeout(1400);
  out.drag={from:{x,y},to:{x:x+130,y:y+100},after:await state(p),pets:await p.evaluate(()=>[...document.querySelectorAll('.limbo-pet')].map(e=>{const r=e.getBoundingClientRect();return {center:{x:r.x+r.width/2,y:r.y+r.height/2},gaze:e.dataset.gaze}}))};
  out.screenshot='backend/extension/pet/cursor/visible_drag_dock_S1261.png';await p.screenshot({path:out.screenshot});
 }else out.dragSkipped='mainFloat still hidden/zero-sized after dock activation';
 out.title=await p.title();out.url=p.url();
}catch(e){out.error=String(e?.stack||e)}
finally{try{if(id){await closeBrowserSession(id);out.closed=true}}catch(e){out.cleanupError=String(e)}}
console.log(JSON.stringify(out));
NODE
