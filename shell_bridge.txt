#!/bin/sh
set +e
cd /home/runner/workspace/ai-agent || exit 90
echo "=== current backend pid and start time ==="
ps -p 139 -o pid,ppid,lstart,args 2>&1
echo "=== port 3000 owner via /proc ==="
for f in /proc/net/tcp /proc/net/tcp6; do
  echo "--- $f ---"
  awk 'NR==1 || $2 ~ /:0BB8$/ {print}' "$f" 2>/dev/null
done
echo "=== node backend processes ==="
ps -eo pid,ppid,lstart,args | grep '[b]ackend/server.js'
echo "=== source timestamp and startup log ==="
stat -c '%y %n' backend/server.js
grep -n 'AI Server running' /tmp/ai-agent-pet-api-restart.log 2>/dev/null || true
echo "=== request checks ==="
for route in /browser/session /browser/cursor/status /browser/recognition/inspect; do
  code=$(curl -sS -o /tmp/pet_probe_body.txt -w '%{http_code}' --max-time 4 -X POST "http://127.0.0.1:3000$route" -H 'Content-Type: application/json' --data '{}' 2>&1)
  printf '%s http=%s body=' "$route" "$code"
  head -c 180 /tmp/pet_probe_body.txt 2>/dev/null; echo
done
echo "=== process environment hints ==="
tr '\0' '\n' </proc/139/environ 2>/dev/null | grep -E '^(PORT|PWD|REPL_ID|REPLIT_DEPLOYMENT)=' || true
echo DIAGNOSE_BACKEND_INSTANCE_COMPLETE
