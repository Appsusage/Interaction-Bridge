(d / "pet.js").write_text(r'''// Pet Limbo: blue living orb, release-position landing, gaze, expression and whitespace-aware placement.
(() => {
  const personality = {
    name: "宠宝", mood: "curious", gaze: "forward",
    lines: ["嘿嘿～", "看我看我！", "我在这里哦。", "发现你啦！", "让我看看。"]
  };
  let currentPet = null, idleTimer = null, behaviorTimer = null;
  const clamp = (v, min, max) => Math.max(min, Math.min(max, v));
  function occupiedAt(x, y) {
    const el = document.elementFromPoint(x, y);
    if (!el || el === document.documentElement || el === document.body) return false;
    if (el.closest(".limbo-pet, #mainFloat, .gesture-magnifier")) return false;
    const style = getComputedStyle(el);
    if (style.visibility === "hidden" || style.display === "none" || Number(style.opacity) < .08) return false;
    const tag = el.tagName.toLowerCase();
    return /^(p|span|a|button|input|textarea|select|img|svg|video|canvas|label|h1|h2|h3|li|pre|code|article|section|nav|header|footer)$/.test(tag)
      || (el.getBoundingClientRect().width > 90 && el.getBoundingClientRect().height > 28);
  }
  function findClearLanding(x, y) {
    const pad = 38;
    const maxX = Math.max(pad, innerWidth - pad), maxY = Math.max(pad, innerHeight - pad);
    const baseX = clamp(Number.isFinite(x) ? x : innerWidth / 2, pad, maxX);
    const baseY = clamp(Number.isFinite(y) ? y : innerHeight / 2, pad, maxY);
    if (!occupiedAt(baseX, baseY)) return { x: baseX, y: baseY };
    // Keep close to the user's release point, but prefer nearby visually empty space.
    const offsets = [
      [0, -54], [52, 0], [-52, 0], [0, 54], [42, -42], [-42, -42],
      [42, 42], [-42, 42], [0, -82], [78, 0], [-78, 0], [0, 82],
      [65, -65], [-65, -65], [65, 65], [-65, 65]
    ];
    for (const [dx, dy] of offsets) {
      const px = clamp(baseX + dx, pad, maxX), py = clamp(baseY + dy, pad, maxY);
      if (!occupiedAt(px, py)) return { x: px, y: py };
    }
    // If the whole area is busy, stay near release point and remain non-blocking.
    return { x: clamp(baseX + 44, pad, maxX), y: clamp(baseY + 36, pad, maxY) };
  }
  function randomBetween(a, b) { return a + Math.random() * (b - a); }
  function setGaze(orb, direction) {
    personality.gaze = direction;
    orb.dataset.gaze = direction;
    const pupils = orb.querySelectorAll(".limbo-pupil");
    const poses = {
      forward: [0, 0], left: [-2.4, 0], right: [2.4, 0],
      up: [0, -2], down: [0, 2]
    };
    const [x, y] = poses[direction] || poses.forward;
    pupils.forEach(p => { p.style.transform = `translate(${x}px,${y}px)`; });
    const face = orb.querySelector(".limbo-pet-face");
    if (face) face.style.transform = direction === "left" ? "rotate(-45deg)" : direction === "right" ? "rotate(45deg)" : "rotate(0deg)";
  }
  function animateEntrance(orb, dx, dy, source) {
    source.classList.remove("pet-source-recoil");
    void source.offsetWidth;
    source.classList.add("pet-source-recoil");
    const frames = [
      { transform: "translate(0px,0px) scaleX(.82) scaleY(1.22) rotate(-3deg)", offset: 0 },
      { transform: `translate(${dx * .16}px,${dy * .16}px) scaleX(.9) scaleY(1.14) rotate(3deg)`, offset: .16 },
      { transform: `translate(${dx * .62}px,${dy * .62}px) scaleX(.84) scaleY(1.18) rotate(-2deg)`, offset: .58 },
      { transform: `translate(${dx}px,${dy}px) scaleX(1.2) scaleY(.76) rotate(2deg)`, offset: .72 },
      { transform: `translate(${dx - 5}px,${dy - 13}px) scaleX(.94) scaleY(1.1) rotate(-2deg)`, offset: .84 },
      { transform: `translate(${dx}px,${dy}px) scaleX(1.06) scaleY(.94) rotate(1deg)`, offset: .93 },
      { transform: `translate(${dx}px,${dy}px) scaleX(1) scaleY(1) rotate(0deg)`, offset: 1 }
    ];
    const animation = orb.animate(frames, { duration: clamp(360 + Math.hypot(dx, dy) * 1.15, 560, 1100), easing: "cubic-bezier(.2,.75,.35,1)", fill: "forwards" });
    animation.onfinish = () => {
      if (!orb.isConnected) return;
      orb.style.left = (orb._landing.x - 29) + "px";
      orb.style.top = (orb._landing.y - 29) + "px";
      orb.style.transform = "";
      orb.classList.add("limbo-pet-idle");
      animation.cancel();
    };
  }
  function spawnFrom(source, releaseX, releaseY) {
    if (!source) return;
    const rect = source.getBoundingClientRect();
    const landing = findClearLanding(releaseX, releaseY);
    const startX = rect.left + rect.width / 2, startY = rect.top + rect.height / 2;
    const orb = document.createElement("div");
    orb.className = "limbo-pet";
    orb.setAttribute("role", "button");
    orb.setAttribute("tabindex", "0");
    orb.setAttribute("aria-label", "宠宝；点击互动");
    orb.innerHTML = '<div class="limbo-pet-face"><i class="limbo-eye left"><b class="limbo-pupil"></b></i><i class="limbo-eye right"><b class="limbo-pupil"></b></i><i class="limbo-mouth"></i><i class="limbo-cheek left"></i><i class="limbo-cheek right"></i></div><span class="limbo-pet-word" aria-live="polite">嘿嘿～</span>';
    orb.style.left = (startX - 29) + "px";
    orb.style.top = (startY - 29) + "px";
    orb._landing = landing;
    orb.style.setProperty("--pet-glow", "1");
    document.body.appendChild(orb);
    if (currentPet && currentPet !== orb) currentPet.remove();
    currentPet = orb;
    if (idleTimer) clearTimeout(idleTimer);
    if (behaviorTimer) clearInterval(behaviorTimer);
    animateEntrance(orb, landing.x - startX, landing.y - startY, source);
    const word = orb.querySelector(".limbo-pet-word");
    function beCute() {
      const action = Math.floor(Math.random() * 8);
      orb.classList.remove("limbo-pet-wiggle", "limbo-pet-blush", "limbo-pet-bounce");
      if (action === 0) { setGaze(orb, "left"); }
      else if (action === 1) { setGaze(orb, "right"); }
      else if (action === 2) { setGaze(orb, "up"); }
      else if (action === 3) { setGaze(orb, "down"); }
      else if (action === 4) { setGaze(orb, "forward"); orb.classList.add("limbo-pet-blush"); }
      else if (action === 5) { orb.classList.add("limbo-pet-wiggle"); }
      else if (action === 6) { orb.classList.add("limbo-pet-bounce"); }
      else { setGaze(orb, "forward"); }
      personality.mood = action >= 4 ? "playful" : "curious";
    }
    function interact() {
      beCute();
      if (word) word.textContent = personality.lines[Math.floor(Math.random() * personality.lines.length)];
    }
    orb.addEventListener("click", interact);
    orb.addEventListener("keydown", e => { if (e.key === "Enter" || e.key === " ") { e.preventDefault(); interact(); } });
    idleTimer = setTimeout(() => {
      if (currentPet === orb && orb.isConnected) {
        behaviorTimer = setInterval(() => {
          if (currentPet !== orb || !orb.isConnected) return;
          beCute();
          // Drift only a little, and only when the destination is visually clear.
          const r = orb.getBoundingClientRect();
          const candidate = findClearLanding(r.left + r.width / 2 + randomBetween(-24, 24), r.top + r.height / 2 + randomBetween(-16, 16));
          if (Math.hypot(candidate.x - (r.left + r.width / 2), candidate.y - (r.top + r.height / 2)) < 36) {
            orb.style.left = (candidate.x - 29) + "px";
            orb.style.top = (candidate.y - 29) + "px";
          }
        }, 2800);
      }
    }, 900);
  }
  window.AICortexPetLimbo = Object.freeze({ spawnFrom, personality, findClearLanding });
})();
''')
