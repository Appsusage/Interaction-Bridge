export function formatExperienceContext(experiences, limit = 3) {
  if (!Array.isArray(experiences) || experiences.length === 0) return "";
  const safeLimit = Number.isInteger(limit) ? Math.max(0, Math.min(limit, 5)) : 3;
  return experiences.slice(0, safeLimit).map(item => {
    const taskType = String(item?.taskType || "general").slice(0, 60);
    const lesson = String(item?.lesson || item?.pattern || "").trim().slice(0, 240);
    if (!lesson) return null;
    const status = item?.verified === true ? "已验证经验" : "待验证经验（仅供参考，不自动作为可信规则）";
    return `- [${status}] 任务类型：${taskType}；经验：${lesson}`;
  }).filter(Boolean).join("\\n");
}
'''
if 'export function formatExperienceContext' not in s:
    if anchor not in s: raise SystemExit('experience_store anchor missing; no changes made')
