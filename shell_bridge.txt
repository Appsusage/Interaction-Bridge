let refreshPromise = null;

  if (cached) return cached;

    cached = JSON.parse(await fs.readFile(cachePath, "utf8"));

  } catch {

async function refreshToken() {
  if (refreshPromise) return refreshPromise;
  refreshPromise = (async () => {
    const clientId = process.env.DROPBOX_APP_KEY;
    const clientSecret = process.env.DROPBOX_APP_SECRET;
    const refreshTokenValue = process.env.DROPBOX_REFRESH_TOKEN;
    if (!clientId || !clientSecret || !refreshTokenValue) {
      throw new Error("Missing Dropbox refresh credentials");
    }
    const body = new URLSearchParams({
      grant_type: "refresh_token",
      refresh_token: refreshTokenValue,
      client_id: clientId,
      client_secret: clientSecret
    });

      method: "POST",
      headers: { "Content-Type": "application/x-www-form-urlencoded" },
      body
    });

      await response.arrayBuffer();
      throw new Error("Dropbox token refresh failed with HTTP " + response.status);
    }
    const data = await response.json();
    if (!data.access_token || !Number.isFinite(Number(data.expires_in))) {
      throw new Error("Dropbox token refresh returned an invalid response");

export async function getDropboxAccessToken({ forceRefresh = false } = {}) {
  if (forceRefresh) {
    cached = null;
    try { await fs.unlink(cachePath); } catch (error) {
      if (error.code !== "ENOENT") throw error;
    }

  }

}

export async function invalidateDropboxAccessToken() {
  cached = null;
  try { await fs.unlink(cachePath); } catch (error) {
    if (error.code !== "ENOENT") throw error;
  }
}

export async function dropboxFetch(url, options = {}) {
  const headers = { ...(options.headers || {}) };
  const token = await getDropboxAccessToken();
  headers.Authorization = "Bearer " + token;
  let response = await fetch(url, { ...options, headers });
  if (response.status !== 401) return response;
  await response.arrayBuffer();
  await invalidateDropboxAccessToken();
  const freshToken = await getDropboxAccessToken({ forceRefresh: true });
  const retryHeaders = { ...(options.headers || {}), Authorization: "Bearer " + freshToken };
  return fetch(url, { ...options, headers: retryHeaders });
}
