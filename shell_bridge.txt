cd /home/runner/workspace/ai-agent && set -eu
printf '%s\n' '=== OPEN CHATGPT IN THE RUNNING AI-AGENT BROWSER ==='
printf '%s\n' '--- server health ---'
if curl -fsS --max-time 5 http://127.0.0.1:3000/health >/tmp/chatgpt-health.txt; then
  cat /tmp/chatgpt-health.txt
else
  echo 'LOCAL_SERVER_NOT_RESPONDING'
  exit 2
fi
printf '%s\n' '--- auth key configuration names (values hidden) ---'
grep -nE 'requireBrowserKey|x-browser-key|BROWSER.*KEY|browser.*key' backend/server.js | head -n 60
printf '%s\n' '--- create authenticated browser session ---'
KEY="${BROWSER_API_KEY:-${X_BROWSER_KEY:-${BROWSER_KEY:-}}}"
if test -z "$KEY"; then
  echo 'BROWSER_KEY_NOT_AVAILABLE_IN_SHELL_ENV'
  exit 3
fi
SESSION_JSON=$(curl -fsS --max-time 20 -X POST http://127.0.0.1:3000/browser/session -H "content-type: application/json" -H "x-browser-key: $KEY" -d '{"displayPolicy":"foreground","importance":100}')
printf '%s\n' "$SESSION_JSON"
SESSION_ID=$(printf '%s' "$SESSION_JSON" | node -e 'let s="";process.stdin.on("data",d=>s+=d).on("end",()=>{try{process.stdout.write(JSON.parse(s).sessionId||"")}catch{}})')
test -n "$SESSION_ID"
printf 'session_id=%s\n' "$SESSION_ID"
printf '%s\n' '--- navigate to ChatGPT ---'
OPEN_JSON=$(curl -fsS --max-time 45 -X POST http://127.0.0.1:3000/browser/open -H "content-type: application/json" -H "x-browser-key: $KEY" -d "{\"sessionId\":\"$SESSION_ID\",\"url\":\"https://chatgpt.com/\",\"displayPolicy\":\"foreground\",\"importance\":100}")
printf '%s\n' "$OPEN_JSON"
printf '%s\n' '--- read visible page state ---'
READ_JSON=$(curl -fsS --max-time 30 -X POST http://127.0.0.1:3000/browser/read -H "content-type: application/json" -H "x-browser-key: $KEY" -d "{\"sessionId\":\"$SESSION_ID\"}")
printf '%s\n' "$READ_JSON" | node -e 'let s="";process.stdin.on("data",d=>s+=d).on("end",()=>{try{const x=JSON.parse(s);console.log("url="+x.url);console.log("title="+x.title);console.log("body_excerpt="+String(x.text||"").slice(0,1600));}catch(e){console.log(s)}})'
printf '%s\n' '=== CHATGPT PAGE OPENED; SESSION LEFT RUNNING IN AI-AGENT ==='
