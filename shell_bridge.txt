node --input-type=module <<'JS'
import fs from "node:fs/promises";
const names = ["DROPBOX_APP_KEY", "DROPBOX_APP_SECRET", "DROPBOX_REFRESH_TOKEN", "DROPBOX_ACCESS_TOKEN"];
for (const name of names) console.log(name + "_SET=" + Boolean(process.env[name]));
try {
  const s = await fs.stat("bridge/shell/.dropbox_token_cache.json");
  console.log("TOKEN_CACHE_EXISTS=true");
  console.log("TOKEN_CACHE_BYTES=" + s.size);
} catch (e) {
  console.log("TOKEN_CACHE_EXISTS=" + (e.code === "ENOENT" ? "false" : "unknown"));
}
if (process.env.DROPBOX_ACCESS_TOKEN) {
  const response = await fetch("https://api.dropboxapi.com/2/users/get_current_account", {
    method: "POST",
    headers: { Authorization: "Bearer " + process.env.DROPBOX_ACCESS_TOKEN, "Content-Type": "application/json" },
    body: "{}"
  });
  await response.arrayBuffer();
  console.log("ENV_ACCESS_TOKEN_HTTP=" + response.status);
} else {
  console.log("ENV_ACCESS_TOKEN_HTTP=SKIPPED");
}
JS
