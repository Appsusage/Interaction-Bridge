import { getBrowserPage } from "../../browser/browser_service.js";

const controls = new Map();

function requireSessionId(value) {
  const id = String(value || "").trim();
  if (!id) throw new Error("sessionId is required");
  return id;
}

function getControl(sessionId) {
  const control = controls.get(requireSessionId(sessionId));
  if (!control || !control.acquired) {
    throw new Error("AI cursor control has not been acquired for this browser session");
  }
  return control;
}

async function getViewport(page) {
  return page.evaluate(() => ({
    width: Math.max(1, window.innerWidth),
    height: Math.max(1, window.innerHeight),
    devicePixelRatio: window.devicePixelRatio || 1
  }));
}

function clampCoordinate(value, max, name) {
  const number = Number(value);
  if (!Number.isFinite(number)) throw new Error(name + " must be a finite number");
  return Math.max(0, Math.min(Math.max(0, max - 1), number));
}

async function dispatch(sessionId, events) {
  const control = getControl(sessionId);
  const page = getBrowserPage(sessionId);
  if (page.isClosed()) throw new Error("Browser page is closed");
  const viewport = await getViewport(page);
  const x = clampCoordinate(control.x, viewport.width, "x");
  const y = clampCoordinate(control.y, viewport.height, "y");
  for (const event of events) {
    await control.cdp.send("Input.dispatchMouseEvent", { ...event, x, y });
  }
  return { sessionId: String(sessionId), x, y, viewport, controlled: true };
}

export async function acquireControl(sessionId) {
  const id = requireSessionId(sessionId);
  const page = getBrowserPage(id);
  if (page.isClosed()) throw new Error("Browser page is closed");
  let control = controls.get(id);
  if (control?.acquired) return { acquired: true, sessionId: id, reused: true };
  const cdp = await page.context().newCDPSession(page);
  const viewport = await getViewport(page);
  control = { cdp, acquired: true, x: Math.floor(viewport.width / 2), y: Math.floor(viewport.height / 2) };
  controls.set(id, control);
  return { acquired: true, sessionId: id, x: control.x, y: control.y, viewport };
}

export async function moveTo(sessionId, x, y) {
  const control = getControl(sessionId);
  const page = getBrowserPage(sessionId);
  const viewport = await getViewport(page);
  control.x = clampCoordinate(x, viewport.width, "x");
  control.y = clampCoordinate(y, viewport.height, "y");
  await control.cdp.send("Input.dispatchMouseEvent", {
    type: "mouseMoved", x: control.x, y: control.y, button: "none"
  });
  return { sessionId: String(sessionId), x: control.x, y: control.y, viewport, moved: true };
}

export async function pointAt(sessionId, x, y) {
  return moveTo(sessionId, x, y);
}

export async function click(sessionId, options = {}) {
  const button = ["left", "middle", "right"].includes(options.button) ? options.button : "left";
  const clickCount = options.clickCount === 2 ? 2 : 1;
  return dispatch(sessionId, [
    { type: "mouseMoved", button: "none" },
    { type: "mousePressed", button, clickCount },
    { type: "mouseReleased", button, clickCount }
  ]).then(result => ({ ...result, clicked: true, button, clickCount }));
}

export async function doubleClick(sessionId) {
  return click(sessionId, { button: "left", clickCount: 2 });
}

export async function rightClick(sessionId) {
  return click(sessionId, { button: "right", clickCount: 1 });
}

export async function scroll(sessionId, deltaX = 0, deltaY = 600) {
  const control = getControl(sessionId);
  const dx = Math.max(-1600, Math.min(1600, Number(deltaX) || 0));
  const dy = Math.max(-2400, Math.min(2400, Number(deltaY) || 0));
  const result = await dispatch(sessionId, [
    { type: "mouseWheel", button: "none", deltaX: dx, deltaY: dy }
  ]);
  return { ...result, scrolled: true, deltaX: dx, deltaY: dy };
}

export async function releaseControl(sessionId) {
  const id = requireSessionId(sessionId);
  const control = controls.get(id);
  if (!control) return { released: true, sessionId: id, alreadyReleased: true };
  controls.delete(id);
  try { await control.cdp.detach(); } catch {}
  return { released: true, sessionId: id };
}

export async function cancel(sessionId) {
  return releaseControl(sessionId);
}

export function getControlStatus(sessionId) {
  const id = requireSessionId(sessionId);
  const control = controls.get(id);
  return { sessionId: id, acquired: !!control?.acquired, x: control?.x ?? null, y: control?.y ?? null };
}

export default Object.freeze({ acquireControl, moveTo, pointAt, click, doubleClick, rightClick, scroll, releaseControl, cancel, getControlStatus });

function finite(value, name) {
  const number = Number(value);
  if (!Number.isFinite(number)) throw new Error(name + " must be a finite number");
  return number;
}

export function mapImagePointToViewport(point, imageSize, viewportSize) {
  if (!point || !imageSize || !viewportSize) throw new Error("point, imageSize and viewportSize are required");
  const imageWidth = finite(imageSize.width, "imageSize.width");
  const imageHeight = finite(imageSize.height, "imageSize.height");
  const viewportWidth = finite(viewportSize.width, "viewportSize.width");
  const viewportHeight = finite(viewportSize.height, "viewportSize.height");
  if (imageWidth <= 0 || imageHeight <= 0 || viewportWidth <= 0 || viewportHeight <= 0) {
    throw new Error("Image and viewport dimensions must be positive");
  }
  const x = finite(point.x, "point.x");
  const y = finite(point.y, "point.y");
  return {
    x: Math.max(0, Math.min(viewportWidth - 1, x * viewportWidth / imageWidth)),
    y: Math.max(0, Math.min(viewportHeight - 1, y * viewportHeight / imageHeight))
  };
}

export function mapImageBoxToViewport(box, imageSize, viewportSize) {
  if (!box) throw new Error("box is required");
  const topLeft = mapImagePointToViewport({ x: box.x, y: box.y }, imageSize, viewportSize);
  const bottomRight = mapImagePointToViewport({
    x: Number(box.x) + Number(box.width),
    y: Number(box.y) + Number(box.height)
  }, imageSize, viewportSize);
  return {
    x: topLeft.x, y: topLeft.y,
    width: Math.max(0, bottomRight.x - topLeft.x),
    height: Math.max(0, bottomRight.y - topLeft.y),
    center: { x: (topLeft.x + bottomRight.x) / 2, y: (topLeft.y + bottomRight.y) / 2 }
  };
}

import { getBrowserPage } from "../../browser/browser_service.js";

const MAX_TEXT_LENGTH = 12000;
const MAX_CANDIDATES = 120;

function normaliseText(value) {
  return String(value || "").replace(/\\s+/g, " ").trim();
}

function isHumanVerification(text, url = "") {
  const sample = (String(text || "") + " " + String(url || "")).toLowerCase();
  return /captcha|recaptcha|hcaptcha|verify you are human|prove you are human|人机验证|确认您是人类|安全验证/.test(sample);
}

export async function captureFrame(sessionId, options = {}) {
  const page = getBrowserPage(String(sessionId || ""));
  if (page.isClosed()) throw new Error("Browser page is closed");
  const viewport = await page.evaluate(() => ({
    width: window.innerWidth,
    height: window.innerHeight,
    devicePixelRatio: window.devicePixelRatio || 1
  }));
  const image = await page.screenshot({
    type: "jpeg",
    quality: Math.max(30, Math.min(90, Number(options.quality) || 70)),
    fullPage: false,
    animations: "disabled"
  });
  return {
    sessionId: String(sessionId),
    url: page.url(),
    title: await page.title(),
    viewport,
    image: image.toString("base64"),
    mimeType: "image/jpeg",
    capturedAt: new Date().toISOString()
  };
}

export async function inspectPage(sessionId) {
  const page = getBrowserPage(String(sessionId || ""));
  if (page.isClosed()) throw new Error("Browser page is closed");
  const result = await page.evaluate(({ maxCandidates, maxTextLength }) => {
    const visible = el => {
      const style = getComputedStyle(el);
      const rect = el.getBoundingClientRect();
      return style.display !== "none" && style.visibility !== "hidden" &&
        Number(style.opacity) > 0.05 && rect.width > 0 && rect.height > 0 &&
        rect.bottom > 0 && rect.right > 0 && rect.top < innerHeight && rect.left < innerWidth;
    };
    const candidates = [...document.querySelectorAll(
      "button,a,input,textarea,select,[role=button],[role=link],[role=checkbox],[role=radio],[aria-label],[title],[contenteditable=true]"
    )].filter(visible).slice(0, maxCandidates).map((el, index) => {
      const r = el.getBoundingClientRect();
      const label = el.getAttribute("aria-label") || el.getAttribute("title") ||
        el.getAttribute("alt") || el.innerText || el.value || "";
      return {
        index, tag: el.tagName.toLowerCase(), role: el.getAttribute("role") || "",
        type: el.getAttribute("type") || "", text: String(label).replace(/\\s+/g, " ").trim().slice(0, 240),
        rect: { x: r.x, y: r.y, width: r.width, height: r.height },
        center: { x: r.x + r.width / 2, y: r.y + r.height / 2 },
        disabled: !!el.disabled,
        obscured: (() => {
          const top = document.elementFromPoint(r.x + r.width / 2, r.y + r.height / 2);
          return !!top && top !== el && !el.contains(top);
        })()
      };
    });
    return {
      viewport: { width: innerWidth, height: innerHeight, devicePixelRatio: devicePixelRatio || 1 },
      text: (document.body?.innerText || "").slice(0, maxTextLength),
      candidates
    };
  }, { maxCandidates: MAX_CANDIDATES, maxTextLength: MAX_TEXT_LENGTH });
  const challengeDetected = isHumanVerification(result.text, page.url());
  return {
    sessionId: String(sessionId), url: page.url(), title: await page.title(),
    ...result, challengeDetected,
    note: "这是 DOM/可访问性检查，不是对截图的语义视觉识别。截图视觉模型需通过 visionAdapter 接入。"
  };
}

export async function recognize(sessionId, options = {}) {
  const frame = await captureFrame(sessionId, options);
  const inspected = await inspectPage(sessionId);
  if (isHumanVerification(inspected.text, inspected.url)) {
    return {
      sessionId: String(sessionId), status: "human-verification-required",
      challengeDetected: true, frame, inspected,
      message: "检测到可能的人机验证；不会自动解答或代替真人通过。"
    };
  }
  if (typeof options.visionAdapter === "function") {
    const vision = await options.visionAdapter({
      imageBase64: frame.image, mimeType: frame.mimeType,
      viewport: frame.viewport, url: frame.url, title: frame.title
    });
    return {
      sessionId: String(sessionId), status: "recognized",
      challengeDetected: false, frame, inspected, vision
    };
  }
  return {
    sessionId: String(sessionId), status: "vision-adapter-required",
    challengeDetected: false, frame, inspected,
    message: "已取得当前视口截图和可见 DOM 候选元素，但没有配置视觉模型；不会伪造截图识别结果。"
  };
}

export async function verifyAfterAction(sessionId, before = {}) {
  const after = await inspectPage(sessionId);
  return {
    sessionId: String(sessionId),
    verified: before.url !== after.url || before.title !== after.title ||
      before.text !== after.text,
    changed: {
      url: before.url !== undefined && before.url !== after.url,
      title: before.title !== undefined && before.title !== after.title,
      text: before.text !== undefined && before.text !== after.text
    },
    after
  };
}

export { isHumanVerification };
export default Object.freeze({ captureFrame, inspectPage, recognize, verifyAfterAction, isHumanVerification });
