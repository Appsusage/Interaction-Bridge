import path from "node:path";

const STATE_FILE = path.join(os.tmpdir(), "ai-agent-shell-float-state.json");
const MAX_OUTPUT = 120000;

function readStateFile() {
  try {
    const state = JSON.parse(fs.readFileSync(STATE_FILE, "utf8"));
    if (state && typeof state === "object") return state;
  } catch {}
  return { seq: 0, event: "idle", active: false, taskName: "", status: "Waiting for Shell Bridge", command: "", output: "", updatedAt: null, finishedAt: null };
}

function writeState(state) {
  const temporary = STATE_FILE + "." + process.pid + ".tmp";
  fs.writeFileSync(temporary, JSON.stringify(state), "utf8");
  fs.renameSync(temporary, STATE_FILE);
}

export function readShellFloatState() {
  return readStateFile();
}

export function publishShellFloatState(patch = {}) {
  const current = readStateFile();
  const { append, ...fields } = patch;
  const event = fields.event || "update";
  const now = new Date().toISOString();
  const next = { ...current, ...fields, seq: Number(current.seq || 0) + 1, event, active: event !== "finish" && event !== "idle", updatedAt: now };
  if (event === "start") {
    next.output = "";
    next.command = fields.command || "";
    next.startedAt = now;
    next.finishedAt = null;
    next.active = true;
  }
  if (typeof append === "string" && append) {
    next.output = String(next.output || "") + append;
    if (next.output.length > MAX_OUTPUT) next.output = next.output.slice(-MAX_OUTPUT);
  }
  if (event === "finish") {
    next.active = false;
    next.finishedAt = Date.now();
    next.status = fields.status || "Outbox saved · closing in 10s";
  }
  if (event === "idle") next.active = false;
  writeState(next);
  return next;
}

export function appendShellFloatOutput(taskName, text) {
  if (!text) return readStateFile();
  const current = readStateFile();
  if (current.taskName !== taskName || !current.active || current.event === "finish") return current;
  return publishShellFloatState({ event: "update", taskName, status: "Processing command", append: String(text) });
}
