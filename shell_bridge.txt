#!/bin/bash
set -uo pipefail
cd /home/runner/workspace/ai-agent || exit 90
echo "=== S1267 MIGRATION MODULE RUNTIME FUNCTIONAL TEST ==="
echo "timestamp=$(date -u +%Y-%m-%dT%H:%M:%SZ)"
echo "project_root=$PWD"
echo
echo "=== MODULE SOURCE (migration folder only) ==="
for f in backend/migration/environment_analyzer.js backend/migration/dependency_manager.js backend/migration/config_manager.js backend/migration/health_checker.js backend/migration/migration_executor.js backend/migration/migration_manager.js; do
  echo "--- $f ---"
  sed -n '1,240p' "$f"
done
echo
echo "=== NON-MUTATING FUNCTIONAL TESTS ==="
node --input-type=module <<'NODE'
import { analyzeEnvironment } from './backend/migration/environment_analyzer.js';
import { checkDependencies } from './backend/migration/dependency_manager.js';
import { inspectConfig } from './backend/migration/config_manager.js';
import { runHealthChecks } from './backend/migration/health_checker.js';
import { validateMigrationPlan, executeMigrationPlan } from './backend/migration/migration_executor.js';
import { planMigration, verifyMigration } from './backend/migration/migration_manager.js';

const tests = [];
async function test(name, fn) {
  try {
    const value = await fn();
    tests.push({name, ok:true, result:value});
  } catch (error) {
    tests.push({name, ok:false, error:String(error?.stack || error)});
  }
}
await test('analyzeEnvironment()', () => analyzeEnvironment());
await test('checkDependencies()', () => checkDependencies());
await test('inspectConfig() (report presence only; do not print values)', async () => {
  const r = await inspectConfig();
  return JSON.stringify(r).replace(/(token|secret|password|api[_-]?key|credential)([^,}]*)([:,]\s*)("[^"]*"|[^,}]+)/ig, '$1$2$3"[REDACTED]"');
});
await test('runHealthChecks() default (record only; no server startup)', () => runHealthChecks({timeoutMs:1500}));
await test('validateMigrationPlan rejects invalid input', async () => {
  let rejected = false;
  try { validateMigrationPlan(null); } catch { rejected = true; }
  if (!rejected) throw new Error('Expected invalid plan to be rejected');
  return {invalidPlanRejected:true};
});
await test('executeMigrationPlan() default behavior must be non-mutating', () => executeMigrationPlan());
await test('planMigration() integrated plan', () => planMigration());
await test('verifyMigration() with localhost health checks only', () => verifyMigration({baseUrl:'http://127.0.0.1:3000', timeoutMs:1500}));
for (const t of tests) {
  console.log(JSON.stringify(t));
}
console.log('SUMMARY ' + JSON.stringify({passed:tests.filter(t=>t.ok).length,failed:tests.filter(t=>!t.ok).length,total:tests.length}));
NODE
TEST_EXIT=$?
echo
echo "=== SYNTAX / JSON VALIDATION ==="
for f in backend/migration/*.js; do node --check "$f" || TEST_EXIT=1; done
node -e "JSON.parse(require('fs').readFileSync('backend/migration/environment_requirements.json','utf8')); JSON.parse(require('fs').readFileSync('backend/migration/migration_history.json','utf8')); console.log('migration JSON valid')" || TEST_EXIT=1
echo
echo "=== SCOPE CHECK ==="
git status --short -- backend/migration 2>&1 || true
echo "=== S1267 FINISHED exit=$TEST_EXIT (runtime test only; no project files modified) ==="
exit "$TEST_EXIT"
