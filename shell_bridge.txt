set +e
BASE="http://127.0.0.1:3000"
KEY="${BROWSER_API_KEY:-}"
if [ -z "$KEY" ]; then
  echo "ERROR: BROWSER_API_KEY is not set in this Shell Bridge process environment."
  exit 2
fi
echo "=== OPEN HOMEPAGE ==="
curl --silent --show-error --max-time 25 -i -X POST "$BASE/browser/open" -H "Content-Type: application/json" -H "x-browser-key: $KEY" --data '{"sessionId":"1792235638727-vkfxa9dgov","url":"https://chatgpt.com/"}'
echo
echo "=== READ PAGE ==="
curl --silent --show-error --max-time 25 -i -X POST "$BASE/browser/read" -H "Content-Type: application/json" -H "x-browser-key: $KEY" --data '{"sessionId":"1792235638727-vkfxa9dgov"}'
echo
