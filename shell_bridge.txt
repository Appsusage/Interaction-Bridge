#!/bin/sh
set +e
cd /home/runner/workspace/ai-agent || exit 90
if [ -z "$BROWSER_API_KEY" ]; then echo "BROWSER_API_KEY missing in Shell Bridge environment"; exit 2; fi
node --input-type=module <<'NODE'
const base = "http://127.0.0.1:3000";
const key = process.env.BROWSER_API_KEY;
async function post(route, body={}) {
  const r = await fetch(base+route, {method:"POST",headers:{"content-type":"application/json","x-browser-key":key},body:JSON.stringify(body)});
  const raw = await r.text();
  let data; try { data=JSON.parse(raw); } catch { data={raw:raw.slice(0,200)}; }
  console.log(route, "HTTP", r.status, JSON.stringify(data, (k,v)=> k==="image" ? "[base64 omitted]" : v));
  if (!r.ok) throw new Error(route+" HTTP "+r.status+" "+raw.slice(0,300));
  return data;
}
let sid;
try {
  const session = await post("/browser/session", {headless:true});
  sid=session.sessionId;
  await post("/browser/open", {sessionId:sid,url:"https://example.com"});
  const before=await post("/browser/recognition/inspect",{sessionId:sid});
  const frame=await post("/browser/recognition/capture",{sessionId:sid,quality:40});
  console.log("frame_summary", JSON.stringify({mimeType:frame.mimeType,viewport:frame.viewport,imageBase64Length:frame.image?.length,challengeDetected:before.challengeDetected}));
  await post("/browser/cursor/acquire",{sessionId:sid});
  const vp=before.viewport;
  await post("/browser/cursor/move",{sessionId:sid,x:Math.max(0,vp.width-8),y:Math.max(0,vp.height-8)});
  await post("/browser/cursor/status",{sessionId:sid});
  await post("/browser/cursor/click",{sessionId:sid,button:"left",clickCount:1});
  await post("/browser/recognition/verify",{sessionId:sid,before:{url:before.url,title:before.title,text:before.text}});
  await post("/browser/cursor/release",{sessionId:sid});
  console.log("PET_LIVE_SMOKE_TEST_PASS");
} catch(e) {
  console.log("PET_LIVE_SMOKE_TEST_FAIL",e.message);
} finally {
  if(sid) { try { await post("/browser/close",{sessionId:sid}); } catch(e) { console.log("cleanup_note",e.message); } }
}
NODE
