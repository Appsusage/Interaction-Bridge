const key=process.env.BROWSER_API_KEY;
if(!key){console.log('BROWSER_API_KEY missing; cannot safely create session');process.exit(2)}
const base='http://127.0.0.1:3000';
const headers={'content-type':'application/json','x-browser-key':key};
let r=await fetch(base+'/browser/session',{method:'POST',headers,body:JSON.stringify({headless:true,displayPolicy:'foreground',importance:100,requiresUserAction:true})});

console.log('create_session_http='+r.status,JSON.stringify(j));
if(!r.ok||!j.sessionId) process.exit(3);
const id=j.sessionId;
r=await fetch(base+'/browser/open',{method:'POST',headers,body:JSON.stringify({sessionId:id,url:'https://chatgpt.com/',displayPolicy:'foreground',importance:100,requiresUserAction:true})});
j=await r.json().catch(()=>({}));
console.log('open_chatgpt_http='+r.status,JSON.stringify({sessionId:id,...j}));
if(!r.ok) process.exit(4);
r=await fetch(base+'/browser/read',{method:'POST',headers,body:JSON.stringify({sessionId:id})});
j=await r.json().catch(()=>({}));
console.log('read_http='+r.status,JSON.stringify({sessionId:id,url:j.url,title:j.title,loginVisible:/Log in|Sign up for free/i.test(j.text||''),text:(j.text||'').slice(0,1000)}));
NODE
