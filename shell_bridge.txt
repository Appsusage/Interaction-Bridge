 await w.collect();

 console.log("QUEUE_ITEMS",[...w.queued.values()].map(x=>({name:x.name,folder:x.folder,source:x.source?.kind})));
 if(w.queued.size>0){w.schedule();const started=Date.now();
 while((w.queued.size||w.running.size)&&Date.now()-started<26000){await new Promise(r=>setTimeout(r,1000));console.log("WORKER_PROGRESS","queued="+w.queued.size,"running="+w.running.size)}

}catch(e){console.log("WORKER_THIRD_TEST_ERROR",e.stack?.slice(0,700)||e.message);process.exitCode=1}
