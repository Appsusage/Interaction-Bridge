set -e
cd /home/runner/workspace/ai-agent
echo '=== BEFORE RESTART ==='
pgrep -af '^node backend/server.js$' || true
for pid in $(pgrep -f '^node backend/server.js$' || true); do kill -TERM "$pid" || true; done
for i in $(seq 1 10); do
  if ! pgrep -f '^node backend/server.js$' >/dev/null; then break; fi
  sleep 1
done
nohup npm start >/tmp/ai-agent-server.log 2>&1 </dev/null &
echo "START_PID=$!"
for i in $(seq 1 30); do
  sleep 1
  if curl -fsS --max-time 2 http://127.0.0.1:3000/health >/tmp/ai-health.txt 2>/dev/null; then
    echo "HEALTH_OK_AFTER=\${i}s"
    cat /tmp/ai-health.txt
    break
  fi
done
echo '=== START LOG ==='
tail -n 100 /tmp/ai-agent-server.log
echo '=== AUDIO PATCH ==='
grep -nE 'installRemoteAudioBridge|ignoreDefaultArgs|audio-offer|audio-answer|ensureAudioBridge|audio-error' backend/extension/browser/chatgpt_live.js backend/extension/browser/browser_live.js frontend/browser-live-viewer.js
echo '=== SYNTAX CHECK ==='
node --check backend/extension/browser/chatgpt_live.js
node --check backend/extension/browser/browser_live.js
node --check frontend/browser-live-viewer.js
