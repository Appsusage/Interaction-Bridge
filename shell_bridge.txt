cd /home/runner/workspace/ai-agent
printf '%s\n' '=== Console/server process check ==='
ps -eo pid,ppid,stat,args | grep -E '[b]ackend/server.js|[h]eartbeat/heartbeat.js|[h]eartbeat/supervisor-router.js' || true
printf '%s\n' '=== local health routes ==='
for route in /health /heartbeat/health; do
  code="$(curl -sS --max-time 12 -o /tmp/heartbeat-health-body -w '%{http_code}' "http://127.0.0.1:3000$route" || true)"
  echo "$route HTTP $code"
  cat /tmp/heartbeat-health-body 2>/dev/null || true
  printf '\n'
done
printf '%s\n' '=== public health routes ==='
domain="${REPLIT_DEV_DOMAIN:-}"
if [ -n "$domain" ]; then
  case "$domain" in http://*|https://*) base="$domain" ;; *) base="https://$domain" ;; esac
  for route in /health /heartbeat/health; do
    code="$(curl -sS --max-time 12 -o /tmp/public-health-body -w '%{http_code}' "$base$route" || true)"
    echo "$route HTTP $code"
    cat /tmp/public-health-body 2>/dev/null || true
    printf '\n'
  done
else
  echo 'REPLIT_DEV_DOMAIN missing'
fi
printf '%s\n' '=== secret presence only (values are never printed) ==='
for name in HEARTBEAT_SUPERVISOR_TOKEN REPLIT_DEV_DOMAIN GITHUB_TOKEN GH_TOKEN; do
  if printenv "$name" >/dev/null 2>&1; then echo "$name=SET"; else echo "$name=MISSING"; fi
done
printf '%s\n' '=== SiriusA workflow runs ==='
gh_token="${GITHUB_TOKEN:-${GH_TOKEN:-}}"
if [ -n "$gh_token" ]; then
  curl -sS -H "Authorization: Bearer $gh_token" -H 'Accept: application/vnd.github+json' 'https://api.github.com/repos/Appsusage/SiriusA/actions/workflows/supervisor.yml/runs?per_page=5' -o /tmp/actions-runs
  node -e 'const x=JSON.parse(require("fs").readFileSync("/tmp/actions-runs","utf8")); if(x.workflow_runs) console.log(JSON.stringify(x.workflow_runs.map(r=>({id:r.id,status:r.status,conclusion:r.conclusion,event:r.event,created_at:r.created_at,html_url:r.html_url})),null,2)); else console.log(JSON.stringify(x))'
  printf '%s\n' '=== SiriusA secret names only ==='
  curl -sS -H "Authorization: Bearer $gh_token" -H 'Accept: application/vnd.github+json' 'https://api.github.com/repos/Appsusage/SiriusA/actions/secrets?per_page=100' -o /tmp/actions-secrets
  node -e 'const x=JSON.parse(require("fs").readFileSync("/tmp/actions-secrets","utf8")); if(x.secrets) console.log(x.secrets.map(s=>s.name).join("\n") || "(no secrets)"); else console.log(JSON.stringify(x))'
else
  echo 'GITHUB_TOKEN/GH_TOKEN missing; cannot query GitHub Actions API from Replit'
fi
