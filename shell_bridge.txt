#!/bin/sh
set -eu
cd /home/runner/workspace/ai-agent

echo "=== PRECHECK ==="
pwd
test -d backend/extension/pet/cursor
test -f backend/extension/pet/cursor/CURSOR_MOUSE_SPEC.txt
echo "cursor specification exists"

echo "=== EXISTING PET / BROWSER CONTROL ENTRY POINTS ==="
find backend/extension/pet backend/extension/browser frontend -maxdepth 3 -type f \
  \( -name '*.js' -o -name '*.mjs' -o -name '*.cjs' -o -name '*.html' -o -name '*.json' \) \
  -print 2>/dev/null | sort | head -250

echo "=== RECOGNITION DIRECTORY STATUS ==="
if [ -e backend/extension/pet/recognition ]; then
  find backend/extension/pet/recognition -maxdepth 3 -type f -print
else
  echo "recognition directory does not exist yet"
fi

echo "=== PACKAGE / AUTOMATION CAPABILITY INDICATORS ==="
for f in package.json backend/package.json; do
  if [ -f "$f" ]; then
    echo "--- $f ---"
    grep -niE 'playwright|puppeteer|robotjs|nut-js|screenshot|sharp|opencv|vision|electron' "$f" || true
  fi
done

echo "=== PET MOVEMENT / POINTER KEYWORD LOCATIONS ==="
grep -RniE 'pointer|mouse|cursor|moveTo|move_to|setPointerCapture|dispatchEvent|requestAnimationFrame|setInterval|setTimeout|gaze|position:fixed' \
  backend/extension/pet/limbo/pet.js backend/extension/pet/limbo/pet.css backend/extension/pet/limbo/limbo.css \
  backend/extension/browser frontend/index.html 2>/dev/null | head -240 || true

echo "=== CHECK COMPLETE; NO APPLICATION FILES MODIFIED ==="
