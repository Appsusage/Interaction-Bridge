# S623 Fix duplicate-action guard: stop only when previous identical action failed
set -eu
cd /home/runner/workspace/ai-agent
python3 - <<'PY'
from pathlib import Path
p = Path("backend/server.js")
src = p.read_text()
old = '''    const executedActions = new Set();



while(action){

    const actionKey =
    JSON.stringify(action);


    // =======================
    // 防止重复 Action
    // =======================

    if(
        executedActions.has(actionKey)
    ){

        console.log(
            "检测到重复 action，停止:",
            action
        );

        finalReply =
            "检测到重复执行请求，已停止。";

        break;

    }


    // 记录已经执行过的 Action
    executedActions.add(actionKey);


    console.log(
        "执行AI Action:",
        action
    );


    result =
        await receiveAIInstruction(
            action
        );
'''
new = '''    // 记录每个 Action 最近一次的真实执行结果。
    // 相同 Action 上次失败时停止，避免无意义地反复重试；
    // 相同 Action 上次成功时允许再次执行，不把重复本身当成错误。
    const executedActions = new Map();



while(action){

    const actionKey =
    JSON.stringify(action);

    const previousResult = executedActions.get(actionKey);

    if(
        executedActions.has(actionKey) &&
        previousResult &&
        previousResult.success === false
    ){

        console.log(
            "检测到重复失败的 action，停止:",
            action,
            "上次执行结果:",
            previousResult
        );

        finalReply =
            "检测到相同 action 连续执行失败，已停止重试。";

        break;

    }


    console.log(
        "执行AI Action:",
        action
    );


    result =
        await receiveAIInstruction(
            action
        );

    // 保存本次真实执行结果；成功的重复 Action 可以继续执行。
    executedActions.set(actionKey, result);
'''
if old not in src:
    raise SystemExit("ERROR: exact original block not found; no file changed")
backup = Path("/tmp/server.js.before-duplicate-action-fix")
backup.write_text(src)
p.write_text(src.replace(old, new, 1))
print("PATCH_APPLIED: backend/server.js")
print("BACKUP: /tmp/server.js.before-duplicate-action-fix")
PY
printf '%s\n' '=== VERIFY SYNTAX ==='
node --check backend/server.js
printf '%s\n' '=== VERIFY CHANGED BLOCK ==='
nl -ba backend/server.js | sed -n '940,1010p'
printf '%s\n' '=== VERIFY DIFF IS LIMITED TO TARGET BLOCK ==='
git diff -- backend/server.js | sed -n '1,180p'
