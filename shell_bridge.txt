# Fix false-positive ChatGPT login detection; preserve exact backup; no unrelated changes
set -eu
cd /home/runner/workspace/ai-agent
f=backend/extension/browser/chatgpt_live.js
cp -p "$f" "$f.bak-login-detection-20261018"
python3 - <<'PY'
from pathlib import Path
p=Path('backend/extension/browser/chatgpt_live.js')
s=p.read_text()
old='''async function isLoggedIn(page) {
  const url = page.url();
  const text = await pageText(page);
  if (/auth\\.openai\\.com|\\/auth\\/login|\\/email-verification/i.test(url)) return false;
  if (/Log in or sign up|Continue with Google|Enter your email|Check your inbox|verification code/i.test(text)) return false;
  return /New chat|What’s on the agenda|What can I help with|Upgrade|Free/i.test(text);
}'''
new='''async function isLoggedIn(page) {
  const url = page.url();
  const text = await pageText(page);
  if (/auth\\.openai\\.com|\\/auth\\/login|\\/email-verification/i.test(url)) return false;

  // The signed-out ChatGPT homepage also contains "New chat" and other
  // navigation labels. Check explicit signed-out UI first to avoid treating
  // the public landing page as an authenticated session.
  if (/Log in to get answers|Log in or sign up|Sign up for free|Continue with Google|Continue with Microsoft|Enter your email|Check your inbox|verification code/i.test(text)) return false;

  return /What’s on the agenda|What can I help with|Upgrade|Free/i.test(text);
}'''
if old not in s:
    raise SystemExit('Expected isLoggedIn implementation not found; file unchanged')
p.write_text(s.replace(old,new,1))
print('PATCH_OK: signed-out homepage cues checked before logged-in cues')
PY
node --check "$f"
git diff --check -- "$f"
git diff -- "$f" | sed -n '1,120p'
