cd /home/runner/workspace/ai-agent || exit 1
node --input-type=module <<'JS'
import fs from 'node:fs';
const files=['backend/extension/pet/limbo/pet.css','backend/extension/pet/limbo/limbo.css'];
for(const p of files){
 if(!fs.existsSync(p)){console.log('SKIP missing '+p);continue;}
 let s=fs.readFileSync(p,'utf8'), edits=0;
 // Apply only to the left eye while gaze is left: another 5px left and 3px upward.
 s=s.replace(/([^{}]*data-gaze=["']left["'][^{}]*\.limbo-eye\.left[^{}]*)\{([^{}]*)\}/g,(whole,selector,body)=>{
   let next=body;
   if(/left\s*:\s*-3px\s*;/.test(next)) next=next.replace(/left\s*:\s*-3px\s*;/,'left:-8px;');
   else if(/left\s*:\s*-5px\s*;/.test(next)) next=next.replace(/left\s*:\s*-5px\s*;/,'left:-10px;');
   else console.log(p+': left offset did not match -3px/-5px; preserved current value');
   if(/top\s*:\s*4px\s*;/.test(next)) next=next.replace(/top\s*:\s*4px\s*;/,'top:1px;');
   else if(/top\s*:\s*\d+px\s*;/.test(next)) console.log(p+': top offset not 4px; preserved current value');
   if(next!==body){edits++;return selector+'{'+next+'}';}
   return whole;
 });
 if(edits) fs.writeFileSync(p,s);
 console.log(p+': left-gaze left-eye rules updated='+edits);
}
JS
