cd /home/runner/workspace/ai-agent && \
mv backend/heartbeat-supervisor.js backend/heartbeat-supervisor.js.broken-$(date +%Y%m%d-%H%M%S) && \
cat > backend/heartbeat-supervisor.sh <<'EOF'
#!/usr/bin/env bash
set -u
cd /home/runner/workspace/ai-agent
HB="backend/heartbeat.js"
PIDFILE="backend/heartbeat.pid"
LOG="backend/heartbeat-supervisor.log"

log() {
  printf '%s %s\n' "$(date -Is)" "$*" >> "$LOG"
}

start_heartbeat() {
  nohup node "$HB" >> backend/heartbeat.log 2>&1 &
  pid=$!
  printf '%s\n' "$pid" > "$PIDFILE"
  log "HEARTBEAT_STARTED pid=$pid"
}

log "SUPERVISOR_STARTED pid=$$"

while true; do
  if [ ! -s "$PIDFILE" ]; then
    start_heartbeat
  else
    pid=$(cat "$PIDFILE" 2>/dev/null || true)
    if [ -z "$pid" ] || [ ! -d "/proc/$pid" ]; then
      log "HEARTBEAT_MISSING old_pid=$pid"
      start_heartbeat
    fi
  fi
  sleep 2
done
EOF
chmod +x backend/heartbeat-supervisor.sh
bash -n backend/heartbeat-supervisor.sh
node --check backend/heartbeat.js
node --check backend/pid.js
printf '%s\n' 'SUPERVISOR_SH_OK'
