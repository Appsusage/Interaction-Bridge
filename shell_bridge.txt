needle='''window.setFloatMenuOpen = setFloatMenuOpen;'''
replacement='''window.setFloatMenuOpen = setFloatMenuOpen;

// Like the Console float, keep the quick-action dock as a direct body child.
// Re-append when another floating window is mounted later so equal z-index
// elements do not paint over the edge tab.
if (floatMenu && document.body) {
  const keepDockFront = () => {
    floatMenu.style.setProperty("z-index", "2147483647", "important");
    if (floatMenu.parentElement !== document.body || document.body.lastElementChild !== floatMenu) {
      document.body.appendChild(floatMenu);
    }
  };
  keepDockFront();
  const dockLayerObserver = new MutationObserver(records => {
    if (records.some(record => record.type === "childList" && record.addedNodes.length)) {
      keepDockFront();
    }
  });
  dockLayerObserver.observe(document.body, {childList: true});
}'''
assert s.count(needle)==1, 'dock insertion point mismatch'
if 'const keepDockFront = () =>' not in s: p.write_text(s.replace(needle,replacement))

s=p.read_text()
addition='''\n/* Console-equivalent fixed topmost layer for the quick-action dock. */\nbody > #floatMenu.float-menu, body > #floatMenu.float-menu.open {\n  position: fixed !important;\n  z-index: 2147483647 !important;\n}\n'''
if 'Console-equivalent fixed topmost layer' not in s: p.write_text(s.rstrip()+addition)
