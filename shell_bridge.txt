from pathlib import Path
import re
p = Path("frontend/shell-float.js")
s = p.read_text()
original = s
changes = []
# Widen and heighten the floating panel modestly, for desktop and compact/mobile rules.
repls = [
    (r'width:min\(260px,calc\(100vw - 24px\)\)', 'width:min(300px,calc(100vw - 24px))'),
    (r'height:min\(312px,34vh\);min-height:174px', 'height:min(360px,38vh);min-height:190px'),
    (r'width:calc\(50vw - 6px\)', 'width:calc(55vw - 6px)'),
    (r'height:min\(264px,32vh\);min-height:150px', 'height:min(300px,36vh);min-height:165px'),
]
for pattern, replacement in repls:
    s, n = re.subn(pattern, replacement, s)
    changes.append((pattern, n))
# Raise the float by reducing its top offset, but only for the known fixed-position top declaration.
s, n_top = re.subn(r'(?<![\w-])top:(\d+)px', lambda m: 'top:' + str(max(4, int(m.group(1)) - 8)) + 'px', s, count=1)
changes.append(("first top:px declaration reduced by 8px", n_top))
print("Matched edits:", changes)
if s == original:
    print("ERROR: no matching style values found; no file changed")
    raise SystemExit(2)
p.write_text(s)
print("Updated Shell Float dimensions and raised its first top offset.")
PY
