#!/bin/bash
set -uo pipefail
cd /home/runner/workspace/ai-agent || exit 90
echo "=== S1268 MIGRATION FUNCTIONAL EDGE-CASE TESTS ==="
echo "timestamp=$(date -u +%Y-%m-%dT%H:%M:%SZ)"
echo
echo "=== REAL APPLICATION HEALTH ROUTES ==="
for path in /health /heartbeat/health; do
  printf '\n--- GET %s ---\n' "$path"
  curl --max-time 3 -sS -o /tmp/migration_health_body.$$ -w 'http_status=%{http_code} total_seconds=%{time_total}\n' "http://127.0.0.1:3000$path" || true
  head -c 1200 /tmp/migration_health_body.$$ 2>/dev/null || true
  printf '\n'
done
rm -f /tmp/migration_health_body.$$
echo
echo "=== EDGE CASE TESTS (read-only) ==="
node --input-type=module <<'NODE'
import { checkDependencies } from './backend/migration/dependency_manager.js';
import { inspectConfig } from './backend/migration/config_manager.js';
import { runHealthChecks } from './backend/migration/health_checker.js';
import { validateMigrationPlan } from './backend/migration/migration_executor.js';

const results=[];
async function check(name,fn){try{const r=await fn();results.push({name,ok:true,result:r});}catch(e){results.push({name,ok:false,error:String(e)});}}
await check('dependency manager blocks missing manifest',async()=>{
 const r=await checkDependencies({projectRoot:'/tmp/migration-no-such-project-S1268'});
 if(r.status!=='blocked') throw new Error('Expected blocked status');
 return r;
});
await check('dependency manager reports parse error for invalid manifest',async()=>{
 const fs=await import('node:fs/promises'); const os=await import('node:os'); const path=await import('node:path');
 const root=await fs.mkdtemp(path.join(os.tmpdir(),'migration-s1268-'));
 try { await fs.writeFile(path.join(root,'package.json'),'{invalid'); let threw=false; try{await checkDependencies({projectRoot:root})}catch{threw=true}; if(!threw)throw new Error('Invalid JSON was not rejected'); return {invalidManifestRejected:true}; }
 finally { await fs.rm(root,{recursive:true,force:true}); }
});
await check('config inspection honors alternate projectRoot',async()=>{
 const r=await inspectConfig({projectRoot:'/tmp/migration-no-such-project-S1268'});
 if(r.secretFiles.some(f=>f.present)) throw new Error('Unexpected config file present in nonexistent temp path');
 return {checkedRoot:r.projectRoot,filesChecked:r.secretFiles.length};
});
await check('health checker rejects unreachable local port',async()=>{
 const r=await runHealthChecks({baseUrl:'http://127.0.0.1:1',timeoutMs:500});
 if(r.status==='passed') throw new Error('Unreachable port incorrectly passed');
 return r;
});
await check('health checker reports no-run state without URL',async()=>{
 const r=await runHealthChecks();
 if(r.checks[0]?.status!=='not_run') throw new Error('Expected not_run without URL');
 return r;
});
await check('plan validator accepts empty step list',async()=>{
 const r=validateMigrationPlan({steps:[]});
 if(!r.valid || r.stepCount!==0) throw new Error('Empty inspection plan should validate');
 return r;
});
await check('plan validator rejects unsupported step',async()=>{
 let rejected=false;try{validateMigrationPlan({steps:[{kind:'shell',command:'rm -rf /'}]})}catch{rejected=true}
 if(!rejected)throw new Error('Unsupported shell step was accepted');
 return {unsupportedStepRejected:true};
});
await check('plan validator rejects approval bypass for risky step',async()=>{
 let rejected=false;try{validateMigrationPlan({steps:[{kind:'deploy',requiresApproval:false}]})}catch{rejected=true}
 if(!rejected)throw new Error('Risky step disabled approval');
 return {approvalBypassRejected:true};
});
for(const r of results) console.log(JSON.stringify(r));
console.log('SUMMARY '+JSON.stringify({passed:results.filter(x=>x.ok).length,failed:results.filter(x=>!x.ok).length,total:results.length}));
NODE
TEST_EXIT=$?
echo
echo "=== MODULE SYNTAX AND JSON CHECK ==="
for f in backend/migration/*.js; do node --check "$f" || TEST_EXIT=1; done
node -e "JSON.parse(require('fs').readFileSync('backend/migration/environment_requirements.json','utf8')); JSON.parse(require('fs').readFileSync('backend/migration/migration_history.json','utf8')); console.log('migration JSON valid')" || TEST_EXIT=1
echo
echo "=== SCOPE STATUS ==="
git status --short -- backend/migration
echo "=== S1268 FINISHED exit=$TEST_EXIT (tests may create/remove temp files under /tmp only) ==="
exit "$TEST_EXIT"
