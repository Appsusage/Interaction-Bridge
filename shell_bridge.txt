cd /home/runner/workspace/ai-agent && set -eu
printf '%s\n' '=== ENABLE OPTIONAL HEADED MODE FOR THE AI BROWSER ==='
python3 - <<'PY'
from pathlib import Path
p=Path('backend/extension/browser/browser_service.js')
s=p.read_text()
old='''export async function launchBrowser() {
  const executablePath = process.env.PLAYWRIGHT_EXECUTABLE_PATH || "/repl/tools/bin/chromium";
  return chromium.launch({ headless: true, executablePath });
}'''
new='''export async function launchBrowser(options = {}) {
  const executablePath = process.env.PLAYWRIGHT_EXECUTABLE_PATH || "/repl/tools/bin/chromium";
  return chromium.launch({ headless: options.headless ?? true, executablePath });
}'''
if old not in s:
    raise SystemExit('launchBrowser source did not match; no edit made')
s=s.replace(old,new,1)
old2='''export async function createBrowserSession(options = {}) {
  const browser = await launchBrowser();'''
new2='''export async function createBrowserSession(options = {}) {
  const browser = await launchBrowser({ headless: options.headless ?? true });'''
if old2 not in s:
    raise SystemExit('createBrowserSession source did not match; no edit made')
s=s.replace(old2,new2,1)
p.write_text(s)
PY
node --check backend/extension/browser/browser_service.js
printf '%s\n' 'source_patch=verified'
printf '%s\n' '=== RESTART BACKEND SERVER ==='
PID=$(ps -eo pid,comm,args | awk '$2=="node" && $0 ~ /backend\/server.js/ {print $1; exit}')
test -n "$PID"
printf 'old_server_pid=%s\n' "$PID"
kill -TERM "$PID"
for n in $(seq 1 20); do
  if ! curl -fsS --max-time 1 http://127.0.0.1:3000/health >/dev/null 2>&1; then break; fi
  sleep 1
done
nohup node backend/server.js >/tmp/ai-agent-server-chatgpt.log 2>&1 &
NEWPID=$!
printf 'new_server_pid=%s\n' "$NEWPID"
for n in $(seq 1 20); do
  if curl -fsS --max-time 2 http://127.0.0.1:3000/health >/tmp/chatgpt-health-after-restart.txt 2>/dev/null; then break; fi
  sleep 1
done
cat /tmp/chatgpt-health-after-restart.txt
printf '%s\n' '=== CREATE HEADED BROWSER SESSION AND OPEN CHATGPT ==='
KEY="${BROWSER_API_KEY:-}"
test -n "$KEY"
SESSION_JSON=$(curl -fsS --max-time 20 -X POST http://127.0.0.1:3000/browser/session -H "content-type: application/json" -H "x-browser-key: $KEY" -d '{"headless":false,"displayPolicy":"foreground","importance":100}')
printf '%s\n' "$SESSION_JSON"
SESSION_ID=$(printf '%s' "$SESSION_JSON" | node -e 'let s="";process.stdin.on("data",d=>s+=d).on("end",()=>{try{process.stdout.write(JSON.parse(s).sessionId||"")}catch{}})')
test -n "$SESSION_ID"
printf 'headed_session_id=%s\n' "$SESSION_ID"
curl -fsS --max-time 45 -X POST http://127.0.0.1:3000/browser/open -H "content-type: application/json" -H "x-browser-key: $KEY" -d "{\"sessionId\":\"$SESSION_ID\",\"url\":\"https://chatgpt.com/\",\"displayPolicy\":\"foreground\",\"importance\":100}"
sleep 6
curl -fsS --max-time 25 -X POST http://127.0.0.1:3000/browser/read -H "content-type: application/json" -H "x-browser-key: $KEY" -d "{\"sessionId\":\"$SESSION_ID\"}" | node -e 'let s="";process.stdin.on("data",d=>s+=d).on("end",()=>{const x=JSON.parse(s);console.log("url="+x.url);console.log("title="+x.title);console.log("body_excerpt="+String(x.text||"").slice(0,1600))})'
printf '%s\n' '=== HEADED CHATGPT SESSION IS ACTIVE ==='
