#!/bin/bash
set +e
cd /home/runner/workspace/ai-agent || exit 1
echo "TASK=local_pet_cursor_follow_four_directions"
node --input-type=module <<'NODE'
import {createBrowserSession,getBrowserPage,closeBrowserSession} from './backend/extension/browser/browser_service.js';
import cursor from './backend/extension/pet/cursor/index.js';
const o={test:'headless local ai-agent page; invokes pet public API, then cursor integration',moves:[]};let id;
try{
 const s=await createBrowserSession({headless:true});id=s.sessionId;const p=getBrowserPage(id);
 await p.setViewportSize({width:1280,height:800});
 await p.goto('http://127.0.0.1:3000/',{waitUntil:'domcontentloaded',timeout:15000});await p.waitForTimeout(1200);
 o.page={title:await p.title(),url:p.url()};
 o.before=await p.evaluate(()=>({api:!!window.AICortexPetLimbo,viewport:{w:innerWidth,h:innerHeight},css:[...document.styleSheets].map(s=>s.href).filter(Boolean)}));
 if(!o.before.api)throw new Error('Pet API missing from app page');
 await p.evaluate(()=>window.AICortexPetLimbo.spawnFrom(document.body,640,400));
 await p.waitForTimeout(1100);
 o.spawn=await p.evaluate(()=>{const e=document.querySelector('.limbo-pet');if(!e)return null;const r=e.getBoundingClientRect();return {rect:{x:r.x,y:r.y,w:r.width,h:r.height},center:{x:r.left+r.width/2,y:r.top+r.height/2},gaze:e.dataset.gaze}});
 if(!o.spawn)throw new Error('Public spawnFrom did not create pet');
 o.acquire=await cursor.acquireControl(id);
 if(!o.acquire.acquired||o.acquire.petControl?.petPresent!==true)throw new Error('Cursor acquire did not confirm pet active: '+JSON.stringify(o.acquire));
 const targets=[{dir:'right',x:900,y:400},{dir:'down',x:640,y:600},{dir:'left',x:350,y:400},{dir:'up',x:640,y:200}];
 for(const t of targets){
  const response=await cursor.moveTo(id,t.x,t.y);
  const actual=await p.evaluate(()=>{const e=document.querySelector('.limbo-pet');if(!e)return null;const r=e.getBoundingClientRect();return {x:r.left+r.width/2,y:r.top+r.height/2,gaze:e.dataset.gaze}});
  o.moves.push({target:t,response,actual,positionMatches:!!actual&&Math.abs(actual.x-t.x)<2&&Math.abs(actual.y-t.y)<2});
 }
 o.pass=o.moves.length===4&&o.moves.every(m=>m.positionMatches);
}catch(e){o.error=String(e?.stack||e)}
finally{try{if(id){o.release=await cursor.releaseControl(id);await closeBrowserSession(id);o.sessionClosed=true;}}catch(e){o.cleanupError=String(e)}}
console.log(JSON.stringify(o));
NODE
