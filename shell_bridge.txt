#!/bin/sh
set +e
cd /home/runner/workspace/ai-agent || exit 90
echo '=== inspect browser and pet integration ==='
for f in backend/extension/browser/browser_service.js backend/extension/browser/browser_live.js backend/extension/pet/cursor/index.js backend/extension/pet/recognition/index.js backend/extension/pet/recognition/coordinate_mapper.js backend/extension/pet/limbo/pet.js package.json; do
  echo "===== $f ====="
  if [ -f "$f" ]; then wc -l "$f"; grep -n -E 'getBrowserPage|newCDPSession|dispatchMouseEvent|autonom|sessionId|export ' "$f" | head -n 100; else echo MISSING; fi
done
echo '=== package.json ==='
cat package.json 2>/dev/null
echo '=== server routes ==='
grep -n -E 'browser_live|browser_service|pet/limbo/pet|extension/pet' backend/server.js 2>/dev/null | head -n 100
echo '=== syntax checks ==='
node --check backend/extension/pet/cursor/index.js; echo cursor=$?
node --check backend/extension/pet/recognition/index.js; echo recognition=$?
node --check backend/extension/pet/recognition/coordinate_mapper.js; echo mapper=$?
echo INSPECTION_COMPLETE
