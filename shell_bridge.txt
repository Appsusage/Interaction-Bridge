import fs from 'node:fs';
const specs = [
  ['backend/server.js','ai_server'],
  ['bridge/shell/server.js','shell_bridge'],
  ['bridge/shell/github_adaptor.js','github_adaptor'],
  ['backend/heartbeat.js','heartbeat']
];
for (const pid of fs.readdirSync('/proc').filter(x => /^\d+$/.test(x))) {
  try {
    const args = fs.readFileSync('/proc/' + pid + '/cmdline','utf8').split('\0').filter(Boolean);
    if (!args.length) continue;
    const match = specs.find(([arg]) => args.slice(1).includes(arg));
    if (match) console.log(JSON.stringify({pid,args,cwd:fs.realpathSync('/proc/'+pid+'/cwd')}));
  } catch {}
}
__NODE_EOF__
