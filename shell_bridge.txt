set -e
cd /home/runner/workspace/ai-agent
node --input-type=module <<'RESTART_NEW_SERVER_ONLY'
import fs from 'node:fs';
const folder='./bridge/shell/multi-task/';
const oldPid=Number(fs.readFileSync(folder+'.multi-task-server.pid','utf8'));
const cmd=fs.readFileSync('/proc/'+oldPid+'/cmdline','utf8');
if(!cmd.includes('bridge/shell/multi-task/server.js'))throw Error('Refusing non-V2 process');
const supervisorPid=Number(fs.readFileSync(folder+'.multi-task-supervisor.pid','utf8'));
if(!fs.readFileSync('/proc/'+supervisorPid+'/cmdline','utf8').includes('bridge/shell/multi-task/supervisor.js'))throw Error('Supervisor not verified');
console.log('RESTART_V2_EXACT_PID',oldPid,'SUPERVISOR',supervisorPid);
process.kill(oldPid,'SIGTERM');
let fresh=null;
for(let i=0;i<80;i++){
 await new Promise(resolve=>setTimeout(resolve,500));
 try{
  const r=await fetch('http://127.0.0.1:3102/health');
  if(!r.ok)continue;
  const status=await r.json();
  if(status.running&&status.pid!==oldPid){fresh=status;break;}
 }catch{}
}
if(!fresh)throw Error('V2 restart verification failed; old legacy stays running');
console.log('V2_RESTART_SUCCEEDED',JSON.stringify(fresh));
RESTART_NEW_SERVER_ONLY
tail -n 12 bridge/shell/multi-task/server.log
