new_discover = '''function sameTargetFile(item, key) {
  return key.startsWith("file:") && (item.targetKey || targetKey(item)) === key;
}

function upsertTargetFile(registry, entry) {
  const key = entry.targetKey || targetKey(entry);
  if (!key.startsWith("file:")) return false;

  const matches = [];
  for (let i = 0; i < registry.processes.length; i++) {
    if (sameTargetFile(registry.processes[i], key)) matches.push(i);
  }

  if (matches.length === 0) {
    registry.processes.push(entry);
    return true;
  }

  const keepIndex = matches[0];
  const previous = registry.processes[keepIndex];
  registry.processes[keepIndex] = {
    ...entry,
    ...previous,
    pid: entry.pid,
    pidStartTime: entry.pidStartTime,
    command: entry.command,
    args: entry.args,
    cwd: entry.cwd,
    targetKey: key,
    status: "running",
    name: previous.name || entry.name
  };

  const duplicates = new Set(matches.slice(1));
  registry.processes = registry.processes.filter((item, index) => !duplicates.has(index));
  return true;
}

function discover() {
  const registry = readRegistry();
  const knownPids = new Set(registry.processes.map(item => Number(item.pid)));
  const active = activeProcesses();
  const activeIds = new Set(active.map(item => item.pid));

  for (const pid of [...candidates.keys()]) {
    if (!activeIds.has(pid) || knownPids.has(pid)) candidates.delete(pid);
  }

  for (const entry of active) {
    if (knownPids.has(entry.pid)) continue;
    const previous = candidates.get(entry.pid);
    const count = previous && previous.pidStartTime === entry.pidStartTime ? previous.count + 1 : 1;
    candidates.set(entry.pid, { pidStartTime: entry.pidStartTime, count, entry });
    if (count < CONFIRM_SCANS) continue;

    const latest = readRegistry();
    if (latest.processes.some(item => Number(item.pid) === entry.pid)) {
      candidates.delete(entry.pid);
      continue;
    }

    entry.name = uniqueName(entry, latest.processes);
    if (!upsertTargetFile(latest, entry)) latest.processes.push(entry);
    writeRegistry(latest);
    knownPids.add(entry.pid);
    candidates.delete(entry.pid);
  }
}'''

  if (recovering) return;

  try {
    const registry = readRegistry();
    for (const entry of [...registry.processes]) {
      const key = entry.targetKey || targetKey(entry);
      const fileTarget = key.startsWith("file:");

      if (sameProcess(entry)) {
        if (fileTarget) {
          const latest = readRegistry();
          if (upsertTargetFile(latest, entry)) writeRegistry(latest);
        }
        continue;
      }

      const active = findActiveTarget(key);
      if (active) {
        const latest = readRegistry();
        if (fileTarget) {
          upsertTargetFile(latest, active);
          writeRegistry(latest);
        } else {
          const index = latest.processes.findIndex(item =>
            item.name === entry.name && Number(item.pid) === Number(entry.pid)
          );
          if (index >= 0) {
            latest.processes[index] = {
              ...entry, pid: active.pid, pidStartTime: active.pidStartTime,
              command: active.command, args: active.args, cwd: active.cwd,
              targetKey: active.targetKey, status: "running"
            };
            writeRegistry(latest);
          }
        }
        continue;
      }

      const latest = readRegistry();
      const currentIndex = fileTarget
        ? latest.processes.findIndex(item => sameTargetFile(item, key))
        : latest.processes.findIndex(item =>
            item.name === entry.name && Number(item.pid) === Number(entry.pid)
          );
      if (currentIndex < 0) continue;

      if (latest.failedTargets.some(item => item.targetKey === key)) {

            item.name === entry.name && Number(item.pid) === Number(entry.pid)

          targetKey: key,
