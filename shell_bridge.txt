if 'id="browser-live-reopen"' not in s:
    anchor='''  document.body.appendChild(root);'''
    if anchor not in s: raise SystemExit("BLOCKED: viewer root insertion anchor not found")
    s=s.replace(anchor, '''  const reopen = document.createElement("button");
  reopen.id = "browser-live-reopen";
  reopen.type = "button";
  reopen.textContent = "浏览器";
  reopen.title = "重新打开浏览器查看器";
  reopen.hidden = true;
  document.body.appendChild(reopen);
  document.body.appendChild(root);''',1)
# Ensure explicit hide persists across background activity; the reopen pill is the explicit override.
s=s.replace('''showViewer(true);
        if (!selected)''','''showViewer();
        if (!selected)''',1)

hide='''  root.querySelector(".blv-hide").addEventListener("click", () => { manuallyHidden = true; root.classList.add("blv-hidden"); });'''

    manuallyHidden = true;
    root.classList.add("blv-hidden");
    reopen.hidden = false;
  });
  reopen.addEventListener("click", () => {
    manuallyHidden = false;
    reopen.hidden = true;
    showViewer(true);
  });''',1)

# New activity should not force the viewer open after a deliberate manual hide.
