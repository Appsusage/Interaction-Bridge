import fs from 'node:fs';
import {spawn} from 'node:child_process';
const base='./bridge/shell/multi-task/';
const pidPath=base+'.multi-task-supervisor.pid';
function cmdline(pid){try{return fs.readFileSync('/proc/'+pid+'/cmdline','utf8')}catch{return ''}}
const existing=fs.existsSync(pidPath)?Number(fs.readFileSync(pidPath,'utf8')):0;
if(!existing||!cmdline(existing).includes('bridge/shell/multi-task/supervisor.js')){
 const log=fs.openSync(base+'supervisor.log','a');
 const child=spawn(process.execPath,['bridge/shell/multi-task/supervisor.js'],{cwd:process.cwd(),detached:true,stdio:['ignore',log,log]});
 child.unref();fs.closeSync(log);
 console.log('V2_SUPERVISOR_RESTARTED',child.pid);
}else console.log('V2_SUPERVISOR_ALREADY_ONLINE',existing);
let worker=null;
for(let n=0;n<30;n++){
 await new Promise(r=>setTimeout(r,500));
 try{
  const response=await fetch('http://127.0.0.1:3102/health');
  const health=await response.json();
  if(response.ok&&health.ok&&health.running){worker=health;break;}
 }catch{}
}
if(!worker)throw Error('New worker failed health check');
console.log('V2_PRIMARY_HEALTH',JSON.stringify(worker));
RESTART_V2_PRIMARY
tail -n 18 bridge/shell/multi-task/server.log
