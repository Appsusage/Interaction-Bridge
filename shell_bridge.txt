set -e
cd /home/runner/workspace/ai-agent
node --input-type=module <<'HANDOVER_TO_MAIN_V2'
import fs from 'node:fs';
const base='./bridge/shell/multi-task/';
const owned=(pid,needle)=>{try{return fs.readFileSync('/proc/'+pid+'/cmdline','utf8').includes(needle)}catch{return false}};
const get=async(port,path)=>{const r=await fetch('http://127.0.0.1:'+port+path);if(!r.ok)throw Error('HTTP '+port+path+' '+r.status);return r.json()};
const before=await get(3102,'/health');
const main=await get(3000,'/shell-multi/status');
const sup=Number(fs.readFileSync(base+'.multi-task-supervisor.pid','utf8'));
const sid=Number(fs.readFileSync(base+'.multi-task-server.pid','utf8'));
if(!before.running||before.active||before.queued||before.pid!==sid)throw Error('Independent V2 not idle or identity changed');
if(!owned(sup,'bridge/shell/multi-task/supervisor.js')||!owned(sid,'bridge/shell/multi-task/server.js'))throw Error('Independent V2 PID verification failed');
console.log('PRE_HANDOVER',JSON.stringify({standalone:before,main,standaloneSupervisor:sup}));
const origin='http://127.0.0.1:3102';
const response=await fetch(origin+'/shell-multi/stop',{method:'POST',headers:{Origin:origin}});
const stop=await response.json();
if(!response.ok||stop.running||stop.active)throw Error('Independent worker did not drain: '+JSON.stringify(stop));
process.kill(sup,'SIGTERM');
for(let i=0;i<20&&owned(sup,'bridge/shell/multi-task/supervisor.js');i++)await new Promise(r=>setTimeout(r,500));
if(owned(sup,'bridge/shell/multi-task/supervisor.js'))throw Error('Independent supervisor still running; stop without escalation');
if(!owned(sid,'bridge/shell/multi-task/server.js'))throw Error('Standalone process changed before orderly stop');
process.kill(sid,'SIGTERM');
for(let i=0;i<20&&owned(sid,'bridge/shell/multi-task/server.js');i++)await new Promise(r=>setTimeout(r,500));
if(owned(sid,'bridge/shell/multi-task/server.js'))throw Error('Standalone process did not exit, refusing duplicate worker');
const mainOrigin='http://127.0.0.1:3000';
const started=await fetch(mainOrigin+'/shell-multi/start',{method:'POST',headers:{Origin:mainOrigin}});
const result=await started.json();
console.log('MAIN_V2_START_RESPONSE',JSON.stringify(result));
if(!started.ok||!result.running)throw Error('Main V2 startup failed');
const status=await get(3000,'/shell-multi/status');
if(!status.running)throw Error('Main V2 not running after startup');
console.log('V2_NOW_PRIMARY_IN_MAIN_SERVER',JSON.stringify(status));
HANDOVER_TO_MAIN_V2
