import fs from 'node:fs';
import { spawn } from 'node:child_process';
const target = "/home/runner/workspace/ai-agent/backend/heartbeat/supervisor-router.js";
let stopped = [];
for (const item of fs.readdirSync("/proc", {withFileTypes:true}).filter(x=>x.isDirectory() && /^\d+$/.test(x.name))) {
  const pid = Number(item.name);
  if (pid === process.pid) continue;
  try {
    const args = fs.readFileSync("/proc/" + pid + "/cmdline", "utf8").split("\0").filter(Boolean);
    if (args.includes(target)) {
      const stat = fs.readFileSync("/proc/" + pid + "/stat", "utf8");
      const close = stat.lastIndexOf(")");
      const start = close >= 0 ? stat.slice(close + 2).trim().split(/\s+/)[19] : null;
      if (start) {
        process.kill(pid, "SIGTERM");
        stopped.push(pid);
      }
    }
  } catch {}
}
await new Promise(r=>setTimeout(r,500));
const child = spawn(process.execPath, [target], {
  cwd: "/home/runner/workspace/ai-agent",
  detached: true,
  stdio: "ignore"
});
child.unref();
await new Promise(r=>setTimeout(r,1200));
const cmdline = fs.readFileSync("/proc/" + child.pid + "/cmdline", "utf8").split("\0").filter(Boolean);
if (!cmdline.includes(target)) {
  throw new Error("Updated supervisor-router.js process failed immediate verification; PID=" + child.pid);
}
console.log(JSON.stringify({stoppedOldRouterPids:stopped,newRouterPid:child.pid,verified:cmdline.includes(target)}));
