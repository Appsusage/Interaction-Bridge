cd /home/runner/workspace/ai-agent
node --input-type=module <<'MULTITASK_BOOT'
import fs from 'node:fs';
import { spawn } from 'node:child_process';
const marker='ai-agent-multitask-worker';
const argsOf=pid=>{
 try{return fs.readFileSync('/proc/'+pid+'/cmdline','utf8').split('\0').filter(Boolean);}
 catch{return [];}
};
const findWorker=()=>{
 const found=[];
 for(const entry of fs.readdirSync('/proc')){
  if(!/^\d+$/.test(entry))continue;
  const args=argsOf(entry);
  if(args.at(-1)===marker && args[0]?.includes('node'))found.push(Number(entry));
 }
 return found;
};
let existing=findWorker();
console.log('MULTITASK_BEFORE',JSON.stringify(existing));
if(existing.length){
 console.log('MULTITASK_ALREADY_RUNNING',JSON.stringify(existing));
}else{
 const logfile='/tmp/ai-agent-multitask-worker.log';
 const fd=fs.openSync(logfile,'a');
 const script=[
  "import {startShellCommand} from './bridge/shell/multi-task/shell_command.js';",
  "const worker=startShellCommand({pollMs:3000,maxConcurrent:1,logger:console});",
  "process.on('SIGTERM',()=>worker.stop());"
 ].join('\n');
 const child=spawn(process.execPath,['--input-type=module','-e',script,marker],{
  cwd:process.cwd(),env:process.env,detached:true,stdio:['ignore',fd,fd]
 });
 fs.closeSync(fd);
 child.unref();
 console.log('MULTITASK_STARTED_PID',child.pid);
 await new Promise(resolve=>setTimeout(resolve,4000));
 existing=findWorker();
 console.log('MULTITASK_AFTER',JSON.stringify(existing));
 console.log('MULTITASK_ALIVE',existing.includes(child.pid));
 console.log('MULTITASK_LOG',logfile);
 if(!existing.includes(child.pid))process.exitCode=1;
}
MULTITASK_BOOT
