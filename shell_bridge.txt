test("Cortex keeps explicit Shell commands on the local preparation path without dispatching them", async () => {
  const result = await optimizePrompt({
    message: "请通过 Shell Bridge 执行命令",
    shellCommand: "printf 'approved only later'"
  });

  assert.equal(result.masterResult.route, "local");

  assert.equal(result.masterResult.preparedShellTask.requiresApproval, true);
});

test("Cortex escalates file operations when no local executor is available", async () => {
  const result = await optimizePrompt({ message: "请创建文件 report.txt" });
  assert.equal(result.masterResult.needsModel, true);
  assert.equal(result.masterResult.route, "external_ai");
});
