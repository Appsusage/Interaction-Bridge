set -e
cd /home/runner/workspace/ai-agent
echo '=== RESTART FOR AUDIO DIAGNOSTIC ==='
ps -o pid,ppid,args -p 593,592 || true
kill -KILL 593 2>/dev/null || true
kill -TERM 592 2>/dev/null || true
for i in $(seq 1 10); do
  if ! curl -fsS --max-time 1 http://127.0.0.1:3000/health >/dev/null 2>&1; then echo "PORT_CLOSED"; break; fi
  sleep 1
done
if curl -fsS --max-time 1 http://127.0.0.1:3000/health >/dev/null 2>&1; then
  echo 'ERROR_PORT_STILL_OPEN'
  ps -eo pid,ppid,args | grep -E '[b]ackend/server.js'
  exit 3
fi
nohup npm start >/tmp/ai-agent-server.log 2>&1 </dev/null &
echo "START_PID=$!"
for i in $(seq 1 30); do
  sleep 1
  if curl -fsS --max-time 1 http://127.0.0.1:3000/health >/tmp/ai-health.txt 2>/dev/null; then
    echo "HEALTH_OK_AFTER=\${i}s"
    cat /tmp/ai-health.txt
    break
  fi
done
echo '=== PROCESSES ==='
ps -eo pid,ppid,etime,args | grep -E '[b]ackend/server.js|[s]upervisor-router.js'
echo '=== LOG ==='
tail -n 60 /tmp/ai-agent-server.log
