import test from 'node:test';
import assert from 'node:assert/strict';
import { executeMigrationPlan, validateMigrationPlan } from '../migration_executor.js';

test('default execution is a dry run and never invokes an adapter', async () => {
  let invoked = false;
  const result = await executeMigrationPlan(
    { id: 'dry-run', steps: [{ id: 'inspect-1', kind: 'inspect' }] },
    { adapter: async () => { invoked = true; } }
  );
  assert.equal(result.status, 'dry_run');
  assert.equal(invoked, false);
  assert.equal(result.steps.length, 1);
});

test('real execution requires explicit approval and authorization', async () => {
  let invoked = false;
  const plan = { id: 'approval', steps: [{ id: 'deploy-1', kind: 'deploy', requiresApproval: true }] };
  const noApproval = await executeMigrationPlan(plan, {
    dryRun: false, authorized: true, adapter: async () => { invoked = true; }
  });
  assert.equal(noApproval.status, 'approval_required');
  assert.equal(invoked, false);

  const noAuthorization = await executeMigrationPlan(plan, {
    dryRun: false, approved: true, adapter: async () => { invoked = true; }
  });
  assert.equal(noAuthorization.status, 'not_executed');
  assert.equal(invoked, false);
});

test('approved authorized adapter executes steps sequentially and records results', async () => {
  const order = [];
  const plan = { id: 'sequence', steps: [
    { id: 'inspect-1', kind: 'inspect' },
    { id: 'health-1', kind: 'health_check' }
  ] };
  const result = await executeMigrationPlan(plan, {
    dryRun: false, approved: true, authorized: true,
    adapter: async step => { order.push(step.id); return { ok: true }; }
  });
  assert.equal(result.status, 'completed');
  assert.deepEqual(order, ['inspect-1', 'health-1']);
  assert.equal(result.results.length, 2);
  assert.deepEqual(result.results.map(item => item.status), ['passed', 'passed']);
});

test('execution stops at first adapter failure and marks later steps skipped', async () => {
  const order = [];
  const plan = { id: 'stop-on-failure', steps: [
    { id: 'inspect-1', kind: 'inspect' },
    { id: 'backup-1', kind: 'backup', requiresApproval: true },
    { id: 'deploy-1', kind: 'deploy', requiresApproval: true }
  ] };
  const result = await executeMigrationPlan(plan, {
    dryRun: false, approved: true, authorized: true,
    adapter: async step => {
      order.push(step.id);
      if (step.id === 'backup-1') throw new Error('disk failed');
      return { ok: true };
    }
  });
  assert.equal(result.status, 'failed');
  assert.deepEqual(order, ['inspect-1', 'backup-1']);
  assert.equal(result.results[1].status, 'failed');
  assert.equal(result.results[2].status, 'skipped');
});

test('rejects arbitrary command payloads and unsafe approval bypass', () => {
  assert.throws(() => validateMigrationPlan({
    steps: [{ kind: 'inspect', command: 'unsafe command text' }]
  }), /unsupported field|command|executable/i);
  assert.throws(() => validateMigrationPlan({
    steps: [{ kind: 'deploy', requiresApproval: false }]
  }), /approval/i);
});

test('invalid plan fails closed before adapter invocation', async () => {
  let invoked = false;
  await assert.rejects(
    executeMigrationPlan({ steps: [{ kind: 'shell', command: 'unsafe command text' }] }, {
      dryRun: false, approved: true, authorized: true,
      adapter: async () => { invoked = true; }
    }),
    /unsupported migration step/i
  );
  assert.equal(invoked, false);
});

/**
 * Safe-by-default migration plan validator and execution coordinator.
 *
 * This module never runs shell commands itself. A caller must supply a trusted
 * adapter for an explicitly authorized environment. The default is dry-run.
 */
const ALLOWED_KINDS = new Set([
  'inspect',
  'backup',
  'install_dependency',
  'deploy',
  'health_check'
]);
const APPROVAL_REQUIRED_KINDS = new Set(['backup', 'install_dependency', 'deploy']);
const ALLOWED_STEP_FIELDS = new Set([
  'id', 'kind', 'description', 'requiresApproval', 'inputs', 'target', 'operation'
]);
const FORBIDDEN_EXECUTABLE_FIELDS = new Set([
  'command', 'cmd', 'shell', 'script', 'exec', 'executable', 'shellcommand'
]);

function assertNoExecutablePayload(value, path = 'step') {
  if (!value || typeof value !== 'object') return;
  for (const [key, nested] of Object.entries(value)) {
    const normalized = key.toLowerCase().replace(/[_-]/g, '');
    if (FORBIDDEN_EXECUTABLE_FIELDS.has(normalized)) {
      throw new Error(`Executable field "${key}" is not allowed in migration plans (${path}).`);
    }
    assertNoExecutablePayload(nested, `${path}.${key}`);
  }
}

export function validateMigrationPlan(plan) {
  if (!plan || typeof plan !== 'object' || Array.isArray(plan)) {
    throw new TypeError('Migration plan must be an object.');
  }
  if (!Array.isArray(plan.steps)) {
    throw new TypeError('Migration plan must contain a steps array.');
  }
  if (plan.id !== undefined && (typeof plan.id !== 'string' || !plan.id.trim())) {
    throw new TypeError('Migration plan id must be a non-empty string when provided.');
  }

  for (const [index, step] of plan.steps.entries()) {
    if (!step || typeof step !== 'object' || Array.isArray(step)) {
      throw new TypeError(`Migration step at index ${index} must be an object.`);
    }
    if (typeof step.kind !== 'string' || !ALLOWED_KINDS.has(step.kind)) {
      throw new Error(`Unsupported migration step at index ${index}.`);
    }
    for (const key of Object.keys(step)) {
      if (!ALLOWED_STEP_FIELDS.has(key)) {
        throw new Error(`Unsupported field "${key}" in migration step ${index}.`);
      }
    }
    assertNoExecutablePayload(step, `steps[${index}]`);
    if (step.id !== undefined && (typeof step.id !== 'string' || !step.id.trim())) {
      throw new TypeError(`Migration step id at index ${index} must be a non-empty string.`);
    }
    if (step.requiresApproval !== undefined && typeof step.requiresApproval !== 'boolean') {
      throw new TypeError(`requiresApproval at index ${index} must be boolean.`);
    }
    if (APPROVAL_REQUIRED_KINDS.has(step.kind) && step.requiresApproval === false) {
      throw new Error(`Risky step ${index} cannot disable approval.`);
    }
    if (step.inputs !== undefined &&
        (!step.inputs || typeof step.inputs !== 'object' || Array.isArray(step.inputs))) {
      throw new TypeError(`inputs at index ${index} must be an object when provided.`);
    }
  }

  return { valid: true, stepCount: plan.steps.length };
}

export async function executeMigrationPlan(plan, options = {}) {
  const validation = validateMigrationPlan(plan);
  const steps = plan.steps.map((step, index) => ({
    ...step,
    id: step.id ?? `step-${index + 1}`
  }));
  const createdAt = new Date().toISOString();

  if (options.dryRun !== false) {
    return {
      status: 'dry_run',
      planId: plan.id ?? null,
      createdAt,
      validation,
      steps: steps.map((step, index) => ({
        index,
        id: step.id,
        kind: step.kind,
        requiresApproval: APPROVAL_REQUIRED_KINDS.has(step.kind) || step.requiresApproval === true,
        executed: false
      })),
      reason: 'Dry-run is the default; no migration adapter was invoked.'
    };
  }

  if (options.approved !== true) {
    return {
      status: 'approval_required',
      planId: plan.id ?? null,
      createdAt,
      validation,
      executed: false,
      reason: 'Explicit approval is required before executing a migration plan.'
    };
  }
  if (options.authorized !== true) {
    return {
      status: 'not_executed',
      planId: plan.id ?? null,
      createdAt,
      validation,
      executed: false,
      reason: 'An explicitly authorized execution channel is required.'
    };
  }
  if (typeof options.adapter !== 'function') {
    return {
      status: 'not_executed',
      planId: plan.id ?? null,
      createdAt,
      validation,
      executed: false,
      reason: 'No trusted execution adapter is configured.'
    };
  }

  const results = [];
  for (let index = 0; index < steps.length; index += 1) {
    const step = steps[index];
    try {
      const value = await options.adapter(step, {
        planId: plan.id ?? null,
        index,
        total: steps.length
      });
      results.push({
        id: step.id,
        kind: step.kind,
        status: 'passed',
        result: value === undefined ? null : value
      });
    } catch (error) {
      const message = (error instanceof Error ? error.message : String(error))
        .replace(/[\r\n\t]+/g, ' ')
        .slice(0, 300);
      results.push({ id: step.id, kind: step.kind, status: 'failed', error: message });
      for (const remaining of steps.slice(index + 1)) {
        results.push({ id: remaining.id, kind: remaining.kind, status: 'skipped' });
      }
      return {
        status: 'failed',
        planId: plan.id ?? null,
        createdAt,
        completedAt: new Date().toISOString(),
        validation,
        executed: true,
        results
      };
    }
  }

  return {
    status: 'completed',
    planId: plan.id ?? null,
    createdAt,
    completedAt: new Date().toISOString(),
    validation,
    executed: true,
    results
  };
}
