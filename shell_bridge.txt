/** V2-only process supervisor. Does not modify, stop or monitor legacy files. */
import {spawn} from 'node:child_process';
import {existsSync,readFileSync,writeFileSync,unlinkSync,openSync,closeSync} from 'node:fs';
import {join,dirname} from 'node:path';
import {fileURLToPath} from 'node:url';
const dir=dirname(fileURLToPath(import.meta.url));
const selfPid=join(dir,'.multi-task-supervisor.pid');
const childPid=join(dir,'.multi-task-server.pid');
const childFile=join(dir,'server.js');
const log=join(dir,'server.log');
const delay=ms=>new Promise(resolve=>setTimeout(resolve,ms));
function owned(pid,target){
 if(!Number.isSafeInteger(pid)||pid<1)return false;
 try{
  const cmd=readFileSync('/proc/'+pid+'/cmdline','utf8');
  return cmd.includes('bridge/shell/multi-task/'+target);
 }catch{return false;}
}
if(existsSync(selfPid)){
 const previous=Number(readFileSync(selfPid,'utf8').trim());
 if(owned(previous,'supervisor.js'))throw Error('V2 supervisor already active pid='+previous);
 unlinkSync(selfPid);
}
writeFileSync(selfPid,String(process.pid)+'\n',{flag:'wx'});
let stopping=false;
process.once('SIGTERM',()=>{stopping=true;});
process.once('SIGINT',()=>{stopping=true;});
process.once('exit',()=>{try{if(readFileSync(selfPid,'utf8').trim()===String(process.pid))unlinkSync(selfPid);}catch{}});
console.log('[v2 supervisor] started pid='+process.pid);
while(!stopping){
 const current=Number(existsSync(childPid)?readFileSync(childPid,'utf8').trim():0);
 if(!owned(current,'server.js')){
  const fd=openSync(log,'a');
  try{
   const child=spawn(process.execPath,[childFile],{cwd:process.cwd(),detached:true,
      stdio:['ignore',fd,fd],env:process.env});
   child.unref();
   console.log('[v2 supervisor] started server pid='+child.pid);
  }finally{closeSync(fd);}
 }
 await delay(5000);
}
console.log('[v2 supervisor] stopped supervision, worker left unchanged');
V2_SUPERVISOR

import fs from 'node:fs';
const base='./bridge/shell/multi-task/';
const sup=Number(fs.readFileSync(base+'.multi-task-supervisor.pid','utf8'));
const srv=Number(fs.readFileSync(base+'.multi-task-server.pid','utf8'));
const read=pid=>fs.readFileSync('/proc/'+pid+'/cmdline','utf8');
if(!read(sup).includes('multi-task/supervisor.js')||!read(srv).includes('multi-task/server.js'))throw Error('V2 supervisor/server PID ownership mismatch');
const r=await fetch('http://127.0.0.1:3102/health');
const j=await r.json();
if(!j.running||j.pid!==srv)throw Error('V2 health mismatch');
console.log('V2_SUPERVISOR_CONFIRMED',{supervisor:sup,server:srv,running:j.running});
