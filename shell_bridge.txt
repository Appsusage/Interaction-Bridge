#!/bin/sh
set +e
cd /home/runner/workspace/ai-agent || exit 90
echo "=== processes listening on 3000 ==="
(ss -ltnp 2>/dev/null | grep ':3000' || true)
echo "=== node processes ==="
ps -eo pid,ppid,args | grep -E '[n]ode|[n]pm|[r]eplit' | head -n 60
echo "=== backend entry and package scripts ==="
node -e "const p=require('./package.json'); console.log(JSON.stringify({type:p.type,scripts:p.scripts},null,2))" 2>&1
echo "=== server route and startup markers ==="
grep -n -E 'browser/cursor/status|browser/recognition/inspect|app.listen|server.listen|PORT|requireBrowserKey' backend/server.js | tail -n 80
echo "=== HTTP responses ==="
for route in / /browser/cursor/status /browser/recognition/inspect; do
  if [ "$route" = "/" ]; then
    curl -sS -D - -o /tmp/pet_route_body.txt --max-time 4 http://127.0.0.1:3000/ | head -n 12
  else
    curl -sS -D - -o /tmp/pet_route_body.txt --max-time 4 -X POST "http://127.0.0.1:3000$route" -H 'Content-Type: application/json' --data '{}' | head -n 12
  fi
  echo "body[$route]:"
  head -c 500 /tmp/pet_route_body.txt 2>/dev/null; echo
done
echo "=== runtime probe complete ==="
echo RUNTIME_ROUTE_DIAGNOSTIC_COMPLETE
