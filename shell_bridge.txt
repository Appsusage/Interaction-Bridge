cd /home/runner/workspace/ai-agent
node --input-type=module <<'EOF'
import {
  createBrowserSession, getBrowserPage, openPage, readPage,
  clickElement, typeText, scrollPage, captureScreenshot, closeBrowserSession
} from "./backend/extension/browser/browser_service.js";
const session = await createBrowserSession();
try {
  const page = getBrowserPage(session.sessionId);
  const opened = await openPage(page, "https://example.com");
  const read = await readPage(page);
  const titleOk = opened.title === "Example Domain";
  const textOk = read.text.includes("Example Domain");
  await page.setContent('<input id="entry"><button id="go">Go</button><p id="result"></p>');
  await typeText(page, "#entry", "bridge-test");
  await page.locator("#go").evaluate(el => el.addEventListener("click", () => document.querySelector("#result").textContent = document.querySelector("#entry").value));
  await clickElement(page, "#go");
  const clickedText = await page.locator("#result").innerText();
  const clickOk = clickedText === "bridge-test";
  await scrollPage(page, 250);
  const screenshotPath = "/tmp/ai-agent-browser-smoke.png";
  const screenshot = await captureScreenshot(page, screenshotPath);
  console.log(JSON.stringify({titleOk,textOk,clickOk,screenshotSaved:screenshot.saved,screenshotPath,readStart:read.text.slice(0,100)}));
} finally {
  await closeBrowserSession(session.sessionId);
}
EOF
