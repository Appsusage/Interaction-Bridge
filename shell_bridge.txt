from pathlib import Path
p=Path("backend/extension/browser/browser_live.js")
s=p.read_text()
if "async function handleAudioCheck" not in s:
    marker="  async function handleAudioOffer(client, message) {"
    check='''  async function handleAudioCheck(client, message) {
    const sessionId = String(message.sessionId || "");
    if (!sessionId || client._browserLive?.sessionId !== sessionId) {
      throw new Error("请先连接当前浏览器会话");
    }
    const page = getBrowserPage(sessionId);
    const status = await page.evaluate(() => ({
      url: location.href,
      title: document.title,
      bridgeInstalled: !!window.__aiAudioBridge && typeof window.__aiAudioBridge.acceptOffer === "function",
      bridgeInitFlag: !!window.__aiAudioBridgeInstalled,
      getUserMediaAvailable: !!navigator.mediaDevices?.getUserMedia,
      audioElementCount: document.querySelectorAll("audio,video").length,
      secureContext: window.isSecureContext
    }));
    send(client, { type: "audio-check-result", sessionId, status });
  }

'''
    if marker not in s: raise SystemExit("audio offer marker missing")
    s=s.replace(marker,check+marker,1)
old='''      else if (message.type === "audio-offer") {
        handleAudioOffer(client, message).catch(error => send(client, { type: "audio-error", sessionId: message.sessionId, message: error.message }));
      } else if (message.type === "audio-close") {'''
new='''      else if (message.type === "audio-check") {
        handleAudioCheck(client, message).catch(error => send(client, { type: "audio-check-result", sessionId: message.sessionId, error: error.message }));
      } else if (message.type === "audio-offer") {
        handleAudioOffer(client, message).catch(error => send(client, { type: "audio-error", sessionId: message.sessionId, message: error.message }));
      } else if (message.type === "audio-close") {'''
if old in s:
    s=s.replace(old,new,1)
elif 'message.type === "audio-check"' not in s:
    raise SystemExit("audio message handler marker missing")
p.write_text(s)
PY
