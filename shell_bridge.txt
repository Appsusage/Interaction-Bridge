set -e
cd /home/runner/workspace/ai-agent
node --input-type=module <<'V2_SAFE_SMOKE'
import {ShellCommand} from './bridge/shell/multi-task/shell_command.js';
const events = [];
const worker = new ShellCommand({maxConcurrent:2,onEvent:e=>events.push(e.kind)});
const name = await worker.getNextTaskName();
const folder = 'Shell Bridge 多任务与浮窗测试';
const task = {name, folder, path:'/processing/'+folder+'/'+name, content:Buffer.from("printf 'V2 safe task running\\n'; printf 'V2 diagnostic stderr\\n' >&2"), source:null};
await worker.run(task);
console.log('V2_SMOKE_TASK',name,'EVENTS',events.join(','));
console.log('V2_SAFE_SMOKE_COMPLETED');
V2_SAFE_SMOKE
