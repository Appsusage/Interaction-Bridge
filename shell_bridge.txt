from pathlib import Path
import json, os

pid = 84464
print("=== PID 84464 DIAGNOSTIC ===")

p = Path("backend/pid.json")
try:
    data = json.loads(p.read_text())
    entries = data.get("processes", [])
    matches = [e for e in entries if int(e.get("pid", -1)) == pid]
    print("--- pid.json entry ---")
    print(json.dumps(matches, ensure_ascii=False, indent=2))
except Exception as e:
    print("pid.json read error:", repr(e))
    matches = []

hb = Path("backend/heartbeat.js")
try:
    text = hb.read_text()
    for fn in ["function procStartTime", "function procCommand", "function procCwd", "function isExactProcess"]:
        i = text.find(fn)
        print(f"--- {fn} ---")
        print(text[i:i+1800] if i >= 0 else "NOT FOUND")
except Exception as e:
    print("heartbeat.js read error:", repr(e))

proc = Path(f"/proc/{pid}")
print("--- /proc/84464 ---")
print("exists:", proc.exists())
if proc.exists():
    for name, fn in [
        ("comm", lambda: (proc/"comm").read_text().strip()),
        ("cmdline", lambda: (proc/"cmdline").read_bytes().replace(b"\\0", b" ").decode(errors="replace")),
        ("cwd", lambda: os.readlink(proc/"cwd")),
        ("starttime_ticks", lambda: (proc/"stat").read_text().split()[21]),
    ]:
        try:
            print(f"{name}:", fn())
        except Exception as e:
            print(f"{name} error:", repr(e))

print("--- feedback raw logs containing 84464 ---")
fb = Path("bridge/feedback")
found = False
if fb.exists():
    for f in sorted(fb.glob("raw_log*.txt")):
        try:
            s = f.read_text(errors="replace")
            if "84464" in s:
                found = True
                print(f"### {f}")
                print(s)
        except Exception as e:
            print(f"{f}: read error {e!r}")
if not found:
    print("NONE")
print("=== END ===")
