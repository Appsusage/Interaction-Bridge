// Main floating-ball press/drag gestures: up=settings, left=outer dock, down=pet,
// ordinary drag=magnifier. Pointer capture keeps release detection reliable on touch.
(() => {
  if (!mainFloat) return;
  const settingsPanel = document.getElementById("gestureSettingsPanel");
  const petPanel = document.getElementById("gesturePetPanel");
  const magnifier = document.getElementById("gestureMagnifier");
  const magnifierContent = document.getElementById("gestureMagnifierContent");
  const threshold = 42;
  const lensSize = 150;
  const zoom = 1.8;
  let gesture = null;
  let clonedPage = null;

  function closeGesturePanels() {
    if (settingsPanel) settingsPanel.hidden = true;
    if (petPanel) petPanel.hidden = true;
  }
  function showPanel(panel) {
    closeGesturePanels();
    if (panel) panel.hidden = false;
  }
  function clearMagnifier() {
    if (magnifier) magnifier.hidden = true;
    if (magnifierContent) magnifierContent.replaceChildren();
    clonedPage = null;
  }
  function showMagnifier(x, y) {
    if (!magnifier || !magnifierContent) return;
    magnifier.hidden = false;
    magnifier.style.left = (x - lensSize / 2) + "px";
    magnifier.style.top = (y - lensSize / 2) + "px";
    if (!clonedPage) {
      clonedPage = document.body.cloneNode(true);
      clonedPage.querySelectorAll("script,style,iframe,video,audio,canvas").forEach(node => node.remove());
      clonedPage.querySelectorAll("[id]").forEach(node => node.removeAttribute("id"));
      clonedPage.style.cssText = "position:absolute!important;left:0!important;top:0!important;width:" + window.innerWidth + "px!important;min-height:" + window.innerHeight + "px!important;max-width:none!important;margin:0!important;transform:scale(" + zoom + ")!important;transform-origin:top left!important;pointer-events:none!important;";
      magnifierContent.replaceChildren(clonedPage);
    }
    clonedPage.style.left = (-(x * zoom - lensSize / 2)) + "px";
    clonedPage.style.top = (-(y * zoom - lensSize / 2)) + "px";
  }
  function performGesture(type) {
    if (type === "left") {
      closeGesturePanels();
      setFloatMenuOpen(true);
    } else if (type === "up") {
      showPanel(settingsPanel);
    } else if (type === "down") {
      showPanel(petPanel);
    }
  }

  mainFloat.addEventListener("pointerdown", event => {
    if (event.button !== undefined && event.button !== 0) return;
    gesture = { id: event.pointerId, x: event.clientX, y: event.clientY, lastX: event.clientX, lastY: event.clientY, type: null, moved: false };
    try { mainFloat.setPointerCapture(event.pointerId); } catch {}
  });
  mainFloat.addEventListener("pointermove", event => {
    if (!gesture || gesture.id !== event.pointerId) return;
    const dx = event.clientX - gesture.x;
    const dy = event.clientY - gesture.y;
    gesture.lastX = event.clientX;
    gesture.lastY = event.clientY;
    if (Math.hypot(dx, dy) < 14) return;
    gesture.moved = true;
    if (Math.abs(dx) > Math.abs(dy) * 1.2 && dx < -threshold) gesture.type = "left";
    else if (Math.abs(dy) > Math.abs(dx) * 1.2 && dy < -threshold) gesture.type = "up";
    else if (Math.abs(dy) > Math.abs(dx) * 1.2 && dy > threshold) gesture.type = "down";
    else gesture.type = null;
    if (gesture.type) clearMagnifier();
    else showMagnifier(event.clientX, event.clientY);
  });
  function finishGesture(event) {
    if (!gesture || (event.pointerId !== undefined && gesture.id !== event.pointerId)) return;
    const completed = gesture;
    gesture = null;
    clearMagnifier();
    if (completed.moved && completed.type) {
      event.preventDefault();
      performGesture(completed.type);
    }
  }
  mainFloat.addEventListener("pointerup", finishGesture);
  mainFloat.addEventListener("pointercancel", finishGesture);
  mainFloat.addEventListener("lostpointercapture", event => {
    if (gesture && gesture.id === event.pointerId) finishGesture(event);
  });
  document.querySelectorAll("[data-gesture-close]").forEach(button => {
    button.addEventListener("click", closeGesturePanels);
  });
  [settingsPanel, petPanel].forEach(panel => {
    if (panel) panel.addEventListener("pointerdown", event => {
      if (event.target === panel) closeGesturePanels();
    });
  });
  document.addEventListener("keydown", event => {
    if (event.key === "Escape") closeGesturePanels();
  });
})();'''
