// 手动重新连接：实际检查服务健康状态，并通知实时模块重连
// =======================
async function reconnectServer() {
  const status = document.getElementById("serverStatus");
  const button = document.getElementById("reconnectFloatButton");
  if (button) button.disabled = true;
  if (status) {
    status.textContent = "● 正在重新连接…";
    status.style.color = "#ffaa00";
  }
  let lastError = "服务暂时无法连接";
  try {
    for (let attempt = 1; attempt <= 5; attempt++) {
      try {
        const response = await fetch("/health", { cache: "no-store" });
        if (response.ok) {
          let payload = {};
          try { payload = await response.json(); } catch {}
          if (!payload.status || payload.status === "ok") {
            if (status) {
              status.textContent = "● 已连接";
              status.style.color = "#00d99a";
            }
            window.dispatchEvent(new CustomEvent("ai-server-reconnect"));
            if (typeof window.reconnectBrowserLive === "function") window.reconnectBrowserLive();
            return true;
          }
          lastError = "服务已响应，但状态异常";
        } else {
          lastError = "服务返回 HTTP " + response.status;
        }
      } catch (error) {
        lastError = error && error.message ? error.message : "网络连接失败";
      }
      if (attempt < 5) await new Promise(resolve => setTimeout(resolve, 700));
    }
    if (status) {
      status.textContent = "● 连接失败：" + lastError;
      status.style.color = "#ff7777";
    }
    return false;
  } finally {
    if (button) button.disabled = false;
  }
}

'''

const floatMenu = document.getElementById("floatMenu");
const mainFloat = document.getElementById("mainFloat");
const dockToggle = document.getElementById("dockToggle");

function setFloatMenuOpen(open) {
  if (!floatMenu) return;
  floatMenu.classList.toggle("open", !!open);
  if (dockToggle) dockToggle.setAttribute("aria-expanded", String(!!open));
  if (dockToggle) dockToggle.setAttribute("aria-label", open ? "功能菜单已展开" : "展开功能菜单");
  if (mainFloat) mainFloat.setAttribute("aria-label", open ? "收起功能菜单" : "展开功能菜单");
}
if (floatMenu && mainFloat && dockToggle) {
  dockToggle.addEventListener("click", () => setFloatMenuOpen(true));
  mainFloat.addEventListener("click", () => setFloatMenuOpen(false));
  // Tap outside the expanded dock to dismiss it without affecting the chat.
  document.addEventListener("pointerdown", event => {
    if (floatMenu.classList.contains("open") && !floatMenu.contains(event.target)) setFloatMenuOpen(false);
  });
}
window.setFloatMenuOpen = setFloatMenuOpen;

// 语音输入：使用浏览器 Web Speech API，不自动发送识别内容
// =======================
function toggleVoice() {
  const button = document.getElementById("voiceFloatButton");
  const status = document.getElementById("serverStatus");
  const Recognition = window.SpeechRecognition || window.webkitSpeechRecognition;
  const current = window.__aiSpeechRecognition;
  if (current) {
    try { current.stop(); } catch {}
    window.__aiSpeechRecognition = null;
    if (button) button.classList.remove("voice-active");
    if (status) status.textContent = "语音输入已停止";
    return;
  }
  if (!Recognition) {
    if (status) {
      status.textContent = "当前浏览器不支持语音识别；请使用支持 Web Speech API 的浏览器";
      status.style.color = "#ff7777";
    }
    return;
  }
  const recognition = new Recognition();
  recognition.lang = (navigator.language && /^(zh|en)/i.test(navigator.language)) ? navigator.language : "zh-CN";
  recognition.continuous = false;
  recognition.interimResults = true;
  let finalText = "";
  recognition.onstart = () => {
    window.__aiSpeechRecognition = recognition;
    if (button) button.classList.add("voice-active");
    if (status) { status.textContent = "● 正在聆听…再次按麦克风可停止"; status.style.color = "#00b889"; }
  };
  recognition.onresult = event => {
    let interim = "";
    for (let i = event.resultIndex; i < event.results.length; i++) {
      const phrase = event.results[i][0].transcript || "";
      if (event.results[i].isFinal) finalText += phrase;
      else interim += phrase;
    }
    const input = document.getElementById("messageInput");
    if (input) {
      input.value = (finalText + interim).trim();
      input.dispatchEvent(new Event("input", { bubbles: true }));
      input.focus({ preventScroll: true });
      input.setSelectionRange(input.value.length, input.value.length);
    }
  };
  recognition.onerror = event => {
    const reason = event.error === "not-allowed" ? "麦克风权限被拒绝" :
      event.error === "no-speech" ? "没有检测到语音，请再试一次" :
      event.error === "audio-capture" ? "找不到可用麦克风" : "语音识别失败：" + event.error;
    if (status) { status.textContent = reason; status.style.color = "#ff7777"; }
  };
  recognition.onend = () => {
    if (window.__aiSpeechRecognition === recognition) window.__aiSpeechRecognition = null;
    if (button) button.classList.remove("voice-active");
    if (status && finalText.trim()) { status.textContent = "语音已转成文字，请检查后发送"; status.style.color = "#00d99a"; }
  };
  try { recognition.start(); }
  catch (error) {
    window.__aiSpeechRecognition = null;
    if (button) button.classList.remove("voice-active");
    if (status) { status.textContent = "无法启动语音：" + (error.message || error); status.style.color = "#ff7777"; }
  }
}

'''
s = s[:start] + new + s[end:]
