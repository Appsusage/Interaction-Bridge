cat > backend/heartbeat/supervisor-router.js <<'EOF'
import fs from "node:fs";
import path from "node:path";
import https from "node:https";
import { spawn } from "node:child_process";
import { fileURLToPath } from "node:url";

const HERE = path.dirname(fileURLToPath(import.meta.url));
const ROOT = path.resolve(HERE, "../..");
const SCRIPT_FILE = path.join(HERE, "heartbeat-supervisor.sh");
const SUPERVISOR_PID_FILE = path.join(HERE, "supervisor.pid");
const STATE_FILE = path.join(HERE, "supervisor-state.json");
const FAILURE_FILE = path.join(HERE, "feedback", "failure1.txt");
const CHECK_MS = 360000;
const REPOSITORIES = ["Appsusage/SiriusA", "Appsusage/SiriusB"];
let runningCycle = false;
let startingSupervisor = false;

function appendFailure(rawLog, label) {
  try {
    fs.mkdirSync(path.dirname(FAILURE_FILE), { recursive: true });
    fs.appendFileSync(
      FAILURE_FILE,
      "\n[" + new Date().toISOString() + "] " + label + "\n" +
      String(rawLog || "No raw error output was available.") + "\n",
      "utf8"
    );
  } catch {}
}

function readState() {
  try {
    const value = JSON.parse(fs.readFileSync(STATE_FILE, "utf8"));
    return { activeRepo: REPOSITORIES.includes(value.activeRepo) ? value.activeRepo : REPOSITORIES[0] };
  } catch {
    return { activeRepo: REPOSITORIES[0] };
  }
}

function writeState(state) {
  try {
    fs.writeFileSync(STATE_FILE + ".tmp", JSON.stringify(state, null, 2) + "\n", "utf8");
    fs.renameSync(STATE_FILE + ".tmp", STATE_FILE);
  } catch (error) {
    appendFailure(error.stack || error, "Unable to save Supervisor active-repository state");
  }
}

function githubRequest(method, repo, body) {
  const token = process.env.GITHUB_TOKEN || process.env.GH_TOKEN;
  if (!token) {
    return Promise.resolve({ ok: false, rawLog: "Missing GITHUB_TOKEN or GH_TOKEN environment variable." });
  }
  const requestBody = body ? JSON.stringify(body) : "";
  return new Promise(resolve => {
    const request = https.request({
      hostname: "api.github.com",
      path: "/repos/" + repo + "/contents/heartbeat-supervisor.sh",
      method,
      headers: {
        "User-Agent": "ai-agent-supervisor-router",
        "Authorization": "Bearer " + token,
        "Accept": "application/vnd.github+json",
        ...(body ? { "Content-Type": "application/json", "Content-Length": Buffer.byteLength(requestBody) } : {})
      },
      timeout: 12000
    }, response => {
      const chunks = [];
      response.setEncoding("utf8");
      response.on("data", chunk => chunks.push(chunk));
      response.on("end", () => {
        const raw = chunks.join("");
        let data = null;
        try { data = raw ? JSON.parse(raw) : null; } catch {}
        resolve({
          ok: response.statusCode >= 200 && response.statusCode < 300,
          statusCode: response.statusCode,
          data,
          rawLog: "GitHub " + method + " " + repo + " returned HTTP " + response.statusCode + "\n" + raw
        });
      });
    });
    request.on("error", error => resolve({ ok: false, rawLog: error.stack || String(error) }));
    request.on("timeout", () => request.destroy(new Error("GitHub request timed out")));
    if (body) request.write(requestBody);
    request.end();
  });
}

async function syncSupervisorSource(repo) {
  let current = await githubRequest("GET", repo);
  if (current.ok && current.data && typeof current.data.content === "string") {
    const remote = Buffer.from(current.data.content.replace(/\s/g, ""), "base64").toString("utf8");
    const local = fs.readFileSync(SCRIPT_FILE, "utf8");
    if (remote === local) return { ok: true };
  } else if (current.statusCode !== 404) {
    return { ok: false, rawLog: current.rawLog };
  }

  const body = {
    message: "Update HeartBeat supervisor source",
    content: fs.readFileSync(SCRIPT_FILE, "utf8").toString("base64")
  };
  if (current.ok && current.data && current.data.sha) body.sha = current.data.sha;
  const saved = await githubRequest("PUT", repo, body);
  if (!saved.ok) return { ok: false, rawLog: saved.rawLog };
  return { ok: true };
}

function supervisorIsRunning() {
  try {
    const pid = Number(fs.readFileSync(SUPERVISOR_PID_FILE, "utf8").trim());
    if (!Number.isInteger(pid) || pid <= 0 || !process.kill(pid, 0)) return false;
    const cmdline = fs.readFileSync("/proc/" + pid + "/cmdline", "utf8").replace(/\0/g, " ");
    return cmdline.includes(SCRIPT_FILE);
  } catch {
    return false;
  }
}

function ensureSupervisorRunning() {
  if (startingSupervisor || supervisorIsRunning() || !fs.existsSync(SCRIPT_FILE)) return;
  startingSupervisor = true;
  try {
    const child = spawn("bash", [SCRIPT_FILE], {
      cwd: ROOT,
      detached: true,
      stdio: "ignore"
    });
    child.once("error", error => appendFailure(error.stack || error, "Supervisor script launch failed"));
    child.unref();
  } catch (error) {
    appendFailure(error.stack || error, "Supervisor script launch exception");
  } finally {
    startingSupervisor = false;
  }
}

async function rotateRepositories() {
  const state = readState();
  const first = state.activeRepo;
  const second = REPOSITORIES.find(repo => repo !== first);
  for (const repo of [first, second]) {
    const result = await syncSupervisorSource(repo);
    if (result.ok) {
      if (state.activeRepo !== repo) writeState({ activeRepo: repo });
      return true;
    }
    appendFailure(result.rawLog, "Supervisor source upload/create failed in " + repo);
  }
  return false;
}

async function runCycle() {
  if (runningCycle) return;
  runningCycle = true;
  try {
    ensureSupervisorRunning();
    await rotateRepositories();
    ensureSupervisorRunning();
  } catch (error) {
    appendFailure(error.stack || error, "Supervisor router cycle failed");
  } finally {
    runningCycle = false;
  }
}

if (!fs.existsSync(STATE_FILE)) writeState({ activeRepo: REPOSITORIES[0] });
runCycle();
setInterval(() => { runCycle(); }, CHECK_MS);
EOF
