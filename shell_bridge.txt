#!/bin/sh
set +e
cd /home/runner/workspace/ai-agent || exit 90
node --input-type=module <<'NODE'
import { readFile, writeFile } from 'node:fs/promises';
const file = 'backend/server.js';
let s = await readFile(file, 'utf8');
const marker = '} from "./extension/browser/browser_service.js";';
const imports = ` } from "./extension/browser/browser_service.js";

import {
  acquireControl as acquirePetCursor,
  moveTo as movePetCursor,
  click as clickPetCursor,
  scroll as scrollPetCursor,
  releaseControl as releasePetCursor,
  getControlStatus as getPetCursorStatus
} from "./extension/pet/cursor/index.js";
import {
  captureFrame as capturePetRecognitionFrame,
  inspectPage as inspectPetRecognitionPage,
  verifyAfterAction as verifyPetRecognitionAction
} from "./extension/pet/recognition/index.js";`;
if (!s.includes('acquirePetCursor')) {
  if (s.split(marker).length !== 2) throw new Error('Browser service import marker not unique; aborting');
  s = s.replace(marker, imports);
}
const routeMarker = 'app.post("/browser/close", requireBrowserKey, async (req, res) => {';
const routes = `app.post("/browser/cursor/acquire", requireBrowserKey, async (req, res) => {
  try { res.json(await acquirePetCursor(req.body.sessionId)); }
  catch (error) { res.status(400).json({ error: error.message }); }
});
app.post("/browser/cursor/status", requireBrowserKey, (req, res) => {
  try { res.json(getPetCursorStatus(req.body.sessionId)); }
  catch (error) { res.status(400).json({ error: error.message }); }
});
app.post("/browser/cursor/move", requireBrowserKey, async (req, res) => {
  try { res.json(await movePetCursor(req.body.sessionId, req.body.x, req.body.y)); }
  catch (error) { res.status(400).json({ error: error.message }); }
});
app.post("/browser/cursor/click", requireBrowserKey, async (req, res) => {
  try { res.json(await clickPetCursor(req.body.sessionId, { button: req.body.button, clickCount: req.body.clickCount })); }
  catch (error) { res.status(400).json({ error: error.message }); }
});
app.post("/browser/cursor/scroll", requireBrowserKey, async (req, res) => {
  try { res.json(await scrollPetCursor(req.body.sessionId, req.body.deltaX, req.body.deltaY)); }
  catch (error) { res.status(400).json({ error: error.message }); }
});
app.post("/browser/cursor/release", requireBrowserKey, async (req, res) => {
  try { res.json(await releasePetCursor(req.body.sessionId)); }
  catch (error) { res.status(400).json({ error: error.message }); }
});
app.post("/browser/recognition/capture", requireBrowserKey, async (req, res) => {
  try { res.json(await capturePetRecognitionFrame(req.body.sessionId, req.body || {})); }
  catch (error) { res.status(400).json({ error: error.message }); }
});
app.post("/browser/recognition/inspect", requireBrowserKey, async (req, res) => {
  try { res.json(await inspectPetRecognitionPage(req.body.sessionId)); }
  catch (error) { res.status(400).json({ error: error.message }); }
});
app.post("/browser/recognition/verify", requireBrowserKey, async (req, res) => {
  try { res.json(await verifyPetRecognitionAction(req.body.sessionId, req.body.before || {})); }
  catch (error) { res.status(400).json({ error: error.message }); }
});

`;
if (!s.includes('app.post("/browser/cursor/acquire"')) {
  if (!s.includes(routeMarker)) throw new Error('Browser close route marker missing; aborting');
  s = s.replace(routeMarker, routes + routeMarker);
}
await writeFile(file, s);
console.log('Integrated cursor and recognition API routes in backend/server.js');
NODE
echo '=== syntax ==='
node --check backend/server.js; echo server_syntax=$?
node --check backend/extension/pet/cursor/index.js; echo cursor_syntax=$?
node --check backend/extension/pet/recognition/index.js; echo recognition_syntax=$?
node --check backend/extension/pet/recognition/coordinate_mapper.js; echo mapper_syntax=$?
echo '=== route registration lines ==='
grep -n -E '/browser/cursor/|/browser/recognition/' backend/server.js
echo '=== import smoke test ==='
node --input-type=module -e "Promise.all([import('./backend/extension/pet/cursor/index.js'),import('./backend/extension/pet/recognition/index.js')]).then(ms=>console.log('module imports ok:',ms.length)).catch(e=>{console.error(e.message);process.exitCode=1})"
echo '=== working tree ==='
git status --short
echo INTEGRATION_COMPLETE
