  if (options.approved !== true) {
    return { dispatched: false, completed: false, reason: "approval_required" };
  }
  const dispatched = await dispatchShellTask(task, { approved: true });
  const timeoutMs = Number.isFinite(options.timeoutMs) ? Math.max(0, options.timeoutMs) : 600000;
  const pollIntervalMs = Number.isFinite(options.pollIntervalMs) ? Math.max(250, options.pollIntervalMs) : 3000;
  const deadline = Date.now() + timeoutMs;
  while (Date.now() <= deadline) {
    const result = await collectShellTaskResult(dispatched.filename, {
      command: options.command || task.content,
      taskType: options.taskType || "shell_command"
    });
    if (result.completed) return { ...dispatched, ...result };
    await new Promise(resolve => setTimeout(resolve, pollIntervalMs));
  }
  return { ...dispatched, completed: false, reason: "result_timeout" };
}

'''
