cd /home/runner/workspace/ai-agent
node --input-type=module <<'EOF'
import { selectToolChain } from "./backend/extension/upgrade/tool_selector.js";
import { receiveAIInstruction } from "./backend/extension/nexus.js";
const types = ["browser_new_session","browser_open","browser_read","browser_click","browser_type","browser_scroll","browser_screenshot","browser_close_session"];
for (const type of types) {
  const chain = selectToolChain({ type }).chain;
  if (!Array.isArray(chain) || chain[0] !== type) throw new Error("route failed: " + type);
}
function findSession(value) {
  if (!value || typeof value !== "object") return null;
  if (typeof value.sessionId === "string") return value.sessionId;
  for (const child of Object.values(value)) { const found = findSession(child); if (found) return found; }
  return null;
}
let sessionId;
try {
  const created = await receiveAIInstruction({ type: "browser_new_session" });
  sessionId = findSession(created);
  console.log("CREATE", JSON.stringify({ success: created.success, hasSession: Boolean(sessionId), error: created.error }));
  if (!created.success || !sessionId) throw new Error("Nexus browser session action failed");
  const opened = await receiveAIInstruction({ type: "browser_open", sessionId, url: "https://example.com" });
  console.log("OPEN", JSON.stringify({ success: opened.success, title: opened.result?.results?.[0]?.result?.title, error: opened.error }));
  if (!opened.success) throw new Error("Nexus browser open action failed");
  const read = await receiveAIInstruction({ type: "browser_read", sessionId });
  const pageResult = read.result?.results?.[0]?.result;
  console.log("READ", JSON.stringify({ success: read.success, title: pageResult?.title, textLength: pageResult?.text?.length, error: read.error }));
  if (!read.success || !pageResult?.title || !(pageResult.text?.length > 0)) throw new Error("Nexus browser read action failed");
  const shot = await receiveAIInstruction({ type: "browser_screenshot", sessionId, path: "/tmp/ai-agent-nexus-browser-test.png" });
  console.log("SCREENSHOT", JSON.stringify({ success: shot.success, result: shot.result?.results?.[0]?.result, error: shot.error }));
  if (!shot.success) throw new Error("Nexus screenshot action failed");
} finally {
  if (sessionId) {
    const closed = await receiveAIInstruction({ type: "browser_close_session", sessionId });
    console.log("CLOSE", JSON.stringify({ success: closed.success, result: closed.result?.results?.[0]?.result, error: closed.error }));
    if (!closed.success) process.exitCode = 1;
  }
}
console.log("NEXUS_BROWSER_INTEGRATION_OK");
EOF
node --check backend/extension/nexus.js
node --check backend/server.js
node --check backend/extension/browser/browser_service.js
