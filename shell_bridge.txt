# S626 Read-only diagnose ChatGPT browser startup failure
set +e
cd /home/runner/workspace/ai-agent
printf '%s\n' '=== 1. BROWSER SERVICE FILE ==='
if [ -f backend/extension/browser/browser_service.js ]; then
  sed -n '1,260p' backend/extension/browser/browser_service.js
else
  echo 'MISSING: backend/extension/browser/browser_service.js'
fi
printf '\n%s\n' '=== 2. BROWSER-RELATED FILES ==='
find backend/extension/browser -maxdepth 3 -type f -print 2>/dev/null | sort
printf '\n%s\n' '=== 3. BROWSER LAUNCH / SESSION REFERENCES ==='
grep -RInE 'D-Bus|dbus|browser_new_session|browser_open|launch\(|executablePath|headless|sessionId|session_id' backend/extension/browser backend 2>/dev/null | head -n 260
printf '\n%s\n' '=== 4. PACKAGE DEPENDENCIES (NAMES/VERSIONS ONLY) ==='
node -e "const p=require('./package.json'); console.log(JSON.stringify({scripts:p.scripts,dependencies:Object.fromEntries(Object.entries(p.dependencies||{}).filter(([k])=>/playwright|puppeteer|chromium|selenium/i.test(k))),devDependencies:Object.fromEntries(Object.entries(p.devDependencies||{}).filter(([k])=>/playwright|puppeteer|chromium|selenium/i.test(k)))},null,2))" 2>&1
printf '\n%s\n' '=== 5. RUNTIME / BROWSER PROCESS CHECK ==='
printf 'DISPLAY=%s\n' "${DISPLAY-<unset>}"
printf 'WAYLAND_DISPLAY=%s\n' "${WAYLAND_DISPLAY-<unset>}"
for bin in chromium chromium-browser google-chrome google-chrome-stable firefox; do command -v "$bin" 2>/dev/null && "$bin" --version 2>/dev/null | head -n 1; done
ps -ef | grep -E '[c]hrom(e|ium)|[f]irefox|[p]laywright' | head -n 80
printf '\n%s\n' '=== 6. RECENT LOG FILE CANDIDATES ==='
find . -maxdepth 4 -type f \( -iname '*log*' -o -iname '*browser*' -o -iname '*error*' \) -printf '%TY-%Tm-%Td %TH:%TM %p\n' 2>/dev/null | sort -r | head -n 100
printf '\n%s\n' '=== END READ-ONLY DIAGNOSTICS ==='
