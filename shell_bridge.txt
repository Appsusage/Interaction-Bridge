import fs from 'node:fs';
const env = fs.readFileSync('/proc/102/environ','utf8').split('\0');
const key = env.find(x => x.startsWith('BROWSER_API_KEY='))?.slice('BROWSER_API_KEY='.length);
if (!key) { console.log('ERROR: server process has no BROWSER_API_KEY'); process.exit(2); }
const base='http://127.0.0.1:3000';
async function post(path, body) {
 const r=await fetch(base+path,{method:'POST',headers:{'content-type':'application/json','x-browser-key':key},body:JSON.stringify(body)});
 const t=await r.text();
 console.log(path+' HTTP '+r.status+' '+t.slice(0,6000));
 if(!r.ok) throw new Error(path+' failed');
 return JSON.parse(t);
}
const s=await post('/browser/session',{headless:false});
const sessionId=s.sessionId || s.id;
if(!sessionId) { console.log('SESSION RESPONSE KEYS:',Object.keys(s)); process.exit(3); }
console.log('SESSION_ID='+sessionId);
await post('/browser/open',{sessionId,url:'https://chatgpt.com/auth/login',waitUntil:'domcontentloaded',timeout:45000});
await new Promise(r=>setTimeout(r,5000));
await post('/browser/read',{sessionId});
