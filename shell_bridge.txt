cd /home/runner/workspace/ai-agent
printf '%s\n' '=== HEARTBEAT CLEAN RESTART ==='

# Kill every existing HeartBeat node instance, but do not kill this task shell.
for d in /proc/[0-9]*; do
  p=${d##*/}
  [ "$p" = "$$" ] && continue
  [ -r "$d/cmdline" ] || continue
  first=''
  IFS= read -r -d '' first < "$d/cmdline" 2>/dev/null || true
  case "$first" in
    */node)
      if grep -aq 'backend/heartbeat.js' "$d/cmdline" 2>/dev/null; then
        kill -KILL "$p" 2>/dev/null || true
      fi
      ;;
  esac
done

# Stop old Supervisor so it cannot race the cleanup.
if [ -s backend/heartbeat-supervisor.pid ]; then
  spid=$(cat backend/heartbeat-supervisor.pid 2>/dev/null || true)
  case "$spid" in ''|*[!0-9]*) ;; *) kill -KILL "$spid" 2>/dev/null || true ;; esac
fi
rm -f backend/heartbeat-supervisor.pid backend/heartbeat.pid

sleep 2

# Apply the recursion fix before restarting HeartBeat.
node --input-type=module <<'NODE'
import fs from 'node:fs';
const path='backend/heartbeat.js';
let s=fs.readFileSync(path,'utf8');
const old=s;
s=s.replace(/const REQUIRED_PROCESSES\s*=\s*\[[\s\S]*?\];/m, `const REQUIRED_PROCESSES = [
  { name:'ai_server', command:'node', args:['backend/server.js'] },
  { name:'shell_bridge', command:'node', args:['bridge/shell/server.js'] },
  { name:'github_adaptor', command:'node', args:['bridge/shell/github_adaptor.js'] }
];`);
if (s===old) throw new Error('REQUIRED_PROCESSES_PATCH_NOT_APPLIED');
fs.writeFileSync(path,s);
NODE
node --check backend/heartbeat.js

# Start exactly one external Supervisor; Supervisor starts one HeartBeat.
nohup bash backend/heartbeat-supervisor.sh >/dev/null 2>&1 &

sleep 4

printf '%s\n' '--- VERIFY HEARTBEAT ---'
ps -eo pid,ppid,args | grep 'backend/heartbeat.js' | grep -v grep || true
printf '%s\n' '--- VERIFY SUPERVISOR ---'
ps -eo pid,ppid,args | grep 'heartbeat-supervisor.sh' | grep -v grep || true
printf '%s\n' '--- VERIFY REQUIRED LIST ---'
grep -n -A6 'REQUIRED_PROCESSES' backend/heartbeat.js || true
printf '%s\n' '=== HEARTBEAT CLEAN RESTART DONE ==='
