import path from "node:path";
import { fileURLToPath } from "node:url";

const HERE = path.dirname(fileURLToPath(import.meta.url));
const CACHE_FILE = path.join(HERE, ".dropbox_token_cache.json");
const TOKEN_URL = "https://api.dropboxapi.com/oauth2/token";
const REFRESH_MARGIN_MS = 60000;

let cachedToken = null;
let cachedExpiresAt = 0;
let refreshPromise = null;

async function readCache() {
  try {
    const data = JSON.parse(await fs.readFile(CACHE_FILE, "utf8"));
    if (typeof data.access_token === "string" && data.access_token &&
        Number.isFinite(data.expires_at)) {
      cachedToken = data.access_token;
      cachedExpiresAt = data.expires_at;
    }
  } catch {
    // No usable cache: refresh from Dropbox.
  }
}

async function saveCache(accessToken, expiresAt) {
  const temp = CACHE_FILE + "." + process.pid + ".tmp";
  await fs.writeFile(temp, JSON.stringify({
    access_token: accessToken,
    expires_at: expiresAt
  }), { mode: 0o600 });
  await fs.rename(temp, CACHE_FILE);
}

export async function invalidateDropboxAccessToken() {
  cachedToken = null;
  cachedExpiresAt = 0;
  try {
    await fs.unlink(CACHE_FILE);
  } catch (error) {
    if (error.code !== "ENOENT") throw error;
  }
}

export async function getDropboxAccessToken({ forceRefresh = false } = {}) {
  if (!forceRefresh && !cachedToken) await readCache();

  if (!forceRefresh && cachedToken &&
      cachedExpiresAt > Date.now() + REFRESH_MARGIN_MS) {
    return cachedToken;
  }

  if (refreshPromise) return refreshPromise;

  refreshPromise = (async () => {
    const appKey = process.env.DROPBOX_APP_KEY?.trim();
    const appSecret = process.env.DROPBOX_APP_SECRET?.trim();
    const refreshToken = process.env.DROPBOX_REFRESH_TOKEN?.trim();

    if (!appKey || !appSecret || !refreshToken) {
      throw new Error("Dropbox Token Manager requires DROPBOX_APP_KEY, DROPBOX_APP_SECRET, and DROPBOX_REFRESH_TOKEN");
    }

    const response = await fetch(TOKEN_URL, {
      method: "POST",
      headers: { "Content-Type": "application/x-www-form-urlencoded" },
      body: new URLSearchParams({
        grant_type: "refresh_token",
        refresh_token: refreshToken,
        client_id: appKey,
        client_secret: appSecret
      })
    });

    const text = await response.text();
    if (!response.ok) {
      throw new Error("Dropbox OAuth refresh failed (" + response.status + "): " + text);
    }

    let result;
    try {
      result = JSON.parse(text);
    } catch {
      throw new Error("Dropbox OAuth refresh returned invalid JSON");
    }

    if (typeof result.access_token !== "string" || !result.access_token) {
      throw new Error("Dropbox OAuth refresh response has no access_token");
    }

    const expiresIn = Number(result.expires_in);
    if (!Number.isFinite(expiresIn) || expiresIn <= 0) {
      throw new Error("Dropbox OAuth refresh response has invalid expires_in");
    }

    cachedToken = result.access_token;
    cachedExpiresAt = Date.now() + expiresIn * 1000;
    await saveCache(cachedToken, cachedExpiresAt);
    return cachedToken;
  })();

  try {
    return await refreshPromise;
  } finally {
    refreshPromise = null;
  }
}

export async function dropboxFetch(url, options = {}) {
  const originalHeaders = options.headers || {};
  const firstToken = await getDropboxAccessToken();
  const firstResponse = await fetch(url, {
    ...options,
    headers: { ...originalHeaders, Authorization: "Bearer " + firstToken }
  });

  if (firstResponse.status !== 401) return firstResponse;

  await invalidateDropboxAccessToken();
  const refreshedToken = await getDropboxAccessToken({ forceRefresh: true });

  // Retry the original request exactly once after a 401.
  return await fetch(url, {
    ...options,
    headers: { ...originalHeaders, Authorization: "Bearer " + refreshedToken }
  });
}
