cd /home/runner/workspace/ai-agent && node <<'NODE'
const fs = require("fs");
const path = "backend/heartbeat.js";
let s = fs.readFileSync(path, "utf8");

const oldBlock = `async function sendFailedRestartFeedback(entry, rawLog) {
  const ts=new Date().toISOString().replace(/[:.]/g,"-");
  const safe=String(entry.name).replace(/[^A-Za-z0-9._-]+/g,"_");
  await githubPut(`feedback/restart-failure/${safe}/${ts}/raw_log.txt`,rawLog || "restart failed with empty raw log\\n");
}`;

const newBlock = `async function sendFailedRestartFeedback(entry, rawLog) {
  const feedbackDir = path.join(ROOT, "bridge", "feedback");
  fs.mkdirSync(feedbackDir, { recursive: true });

  let filePath = path.join(feedbackDir, "raw_log.txt");
  if (fs.existsSync(filePath)) {
    let index = 1;
    while (fs.existsSync(path.join(feedbackDir, \`raw_log${index}.txt\`))) index++;
    filePath = path.join(feedbackDir, \`raw_log${index}.txt\`);
  }

  fs.writeFileSync(
    filePath,
    rawLog || "restart failed with empty raw log\\n",
    "utf8"
  );
}`;

if (!s.includes(oldBlock)) {
  console.error("TARGET_BLOCK_NOT_FOUND");
  process.exit(2);
}

s = s.replace(oldBlock, newBlock);
fs.writeFileSync(path, s, "utf8");
NODE
node --check backend/heartbeat.js
printf '\n--- changed block ---\n'
sed -n '/async function sendFailedRestartFeedback/,/async function recover/p' backend/heartbeat.js
