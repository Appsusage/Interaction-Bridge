#!/usr/bin/env bash
set -euo pipefail
cd /home/runner/workspace/ai-agent
python3 - <<'PY'
from pathlib import Path
import re, sys
css_path = Path("frontend/backend-console-viewer.css")
js_path = Path("frontend/backend-console-viewer.js")
if not css_path.is_file() or not js_path.is_file():
    print("ERROR: Expected Console frontend files are missing; no files changed.")
    sys.exit(2)
css = css_path.read_text()
js = js_path.read_text()
# Discover existing Console selectors, then scope the overrides to selectors already defined by this module.
blocks = re.findall(r'([^{}]+)\{[^{}]*\}', css, flags=re.S)
root_terms = ("window", "panel", "viewer", "container", "frame", "floating", "root", "wrap")
footer_terms = ("footer", "bottom", "status", "controls", "toolbar")
root_selectors = []
footer_selectors = []
for block in blocks:
    selector = " ".join(block.split()).strip()
    low = selector.lower()
    if "console" not in low:
        continue
    # Ignore at-rules and selectors whose apparent class/id is not part of this Console module.
    if low.startswith("@"):
        continue
    if any(term in low for term in root_terms):
        root_selectors.append(selector)
    if any(term in low for term in footer_terms):
        footer_selectors.append(selector)
# Prefer selectors that also appear as IDs/classes in the module's actual markup.
def markup_overlap(selector):
    names = re.findall(r'[.#]([A-Za-z_][\w-]*)', selector)
    return any(name in js for name in names)
root_selectors = list(dict.fromkeys(s for s in root_selectors if markup_overlap(s)))
footer_selectors = list(dict.fromkeys(s for s in footer_selectors if markup_overlap(s)))
if not root_selectors:
    # Fail safely rather than styling unrelated UI if the Console root cannot be identified.
    print("ERROR: Could not confidently identify the Console outer-window selector; no files changed.")
    print("Console-related selectors found:", [s for s in dict.fromkeys(" ".join(b.split()) for b in blocks if "console" in b.lower())][:30])
    sys.exit(3)
if not footer_selectors:
    print("ERROR: Could not confidently identify the Console bottom-row selector; no files changed.")
    print("Console-related selectors found:", [s for s in dict.fromkeys(" ".join(b.split()) for b in blocks if "console" in b.lower())][:30])
    sys.exit(4)
marker = "/* ChatGPT update: Console glass surface + high-contrast bottom row */"
if marker in css:
    css = css[:css.index(marker)].rstrip() + "\n"
root = ",\n".join(root_selectors)
footer = ",\n".join(footer_selectors)
css += f"""

{marker}
{root} {{
  background-color: rgba(20, 30, 43, 0.68);
  -webkit-backdrop-filter: blur(16px) saturate(135%);
  backdrop-filter: blur(16px) saturate(135%);
  border-color: rgba(100, 125, 150, 0.78);
}}

{footer} {{
  color: #ffffff !important;
  text-shadow: 0 0 5px rgba(255, 255, 255, 0.45);
}}
"""
css_path.write_text(css)
print("UPDATED ONLY:", css_path)
print("Glass surface selectors:", root_selectors)
print("White-highlight bottom-row selectors:", footer_selectors)
print("Existing color rules were otherwise preserved.")
PY
node --check frontend/backend-console-viewer.js
git diff --check -- frontend/backend-console-viewer.css
git diff -- frontend/backend-console-viewer.css
