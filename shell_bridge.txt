node --input-type=module <<'RESTART_V2_TO_APPLY_GUARDS'
import fs from 'node:fs';
const base='./bridge/shell/multi-task/';
const p=Number(fs.readFileSync(base+'.multi-task-server.pid','utf8'));
const sup=Number(fs.readFileSync(base+'.multi-task-supervisor.pid','utf8'));
function verify(pid,marker){const argv=fs.readFileSync('/proc/'+pid+'/cmdline','utf8');if(!argv.includes(marker))throw Error('Refuse PID '+pid+': unexpected process');}
verify(p,'multi-task/server.js');verify(sup,'multi-task/supervisor.js');
const status=await (await fetch('http://127.0.0.1:3102/health')).json();
if(status.pid!==p||status.active!==0||status.queued!==0)throw Error('V2 worker not idle; refuse reload');
process.kill(p,'SIGTERM');
let fresh=null;
for(let i=0;i<70;i++){
 await new Promise(r=>setTimeout(r,500));
 try{
  const rsp=await fetch('http://127.0.0.1:3102/health');const x=await rsp.json();
  if(rsp.ok&&x.running&&x.pid!==p){fresh=x;break;}
 }catch{}
}
if(!fresh)throw Error('V2 supervisor did not restart server');
verify(fresh.pid,'multi-task/server.js');
console.log('V2_PATCH_ACTIVATED',JSON.stringify({oldServer:p,newServer:fresh.pid,supervisor:sup,status:fresh}));
RESTART_V2_TO_APPLY_GUARDS
