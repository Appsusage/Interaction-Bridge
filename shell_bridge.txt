node --input-type=module -e 'import {ShellCommand} from "./bridge/shell/multi-task/shell_command.js";
const w=new ShellCommand({maxConcurrent:2,logger:console});
console.log("TOPIC_SEQUENCE_REGRESSION_START");
try{
 await w.collect();
 console.log("QUEUED",JSON.stringify([...w.queued.values()].map(t=>({folder:t.folder,name:t.name,source:t.source?.kind}))));
 w.schedule();
 const deadline=Date.now()+25000;
 while(w.running.size && Date.now()<deadline){await new Promise(r=>setTimeout(r,500));}
 console.log("PENDING_AFTER_TEST",w.queued.size,"RUNNING_AFTER_TEST",w.running.size);
}catch(e){console.error("REGRESSION_ERROR",e.stack);process.exitCode=1}
console.log("TOPIC_SEQUENCE_REGRESSION_END");'
