# chatgpt_live persistent-profile diagnostics
set -eu
cd /home/runner/workspace/ai-agent
printf '\n=== workspace ===\n'
pwd
printf '\n=== browser module candidates ===\n'
find backend -type f \( -iname '*chatgpt*live*.js' -o -iname '*browser*live*.js' \) -print 2>/dev/null | head -40
printf '\n=== profile directory metadata (no cookie/token contents) ===\n'
for p in "${CHATGPT_PROFILE_DIR:-.chatgpt-profile}" ".chatgpt-profile"; do
  if [ -d "$p" ]; then
    printf 'PROFILE_DIR=%s\n' "$p"
    find "$p" -maxdepth 2 -type f -printf '%P %s bytes\n' 2>/dev/null | head -40
  fi
done
printf '\n=== relevant persistent-context/login references ===\n'
grep -RInE 'launchPersistentContext|CHATGPT_PROFILE_DIR|chatgpt\.com|auth\.openai\.com|storageState|userDataDir|headless' backend/extension/browser backend 2>/dev/null | head -100 || true
printf '\n=== syntax check candidates ===\n'
for f in $(find backend -type f -iname '*chatgpt*live*.js' 2>/dev/null); do node --check "$f" && echo "SYNTAX_OK $f"; done
