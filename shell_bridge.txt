import WebSocket from "ws";
const key = process.env.BROWSER_API_KEY;
const ws = new WebSocket("ws://127.0.0.1:3000/browser/live");
const timer = setTimeout(() => { console.log("RESULT timeout"); ws.close(); }, 5000);
ws.on("message", async raw => {
  let m; try { m=JSON.parse(raw.toString()); } catch { return; }
  if (m.type !== "sessions") return;
  clearTimeout(timer);
  console.log("SESSIONS", JSON.stringify(m.sessions));
  const target=(m.sessions||[]).find(s => /chatgpt\.com/i.test(s.url||""));
  if (target) {
    const r=await fetch("http://127.0.0.1:3000/browser/read", {
      method:"POST",headers:{"content-type":"application/json","x-browser-key":key},
      body:JSON.stringify({sessionId:target.sessionId})
    });
    const j=await r.json().catch(()=>({}));
    const t=String(j.text||"");
    console.log("CHATGPT_PAGE",JSON.stringify({
      sessionId:target.sessionId,url:j.url||target.url,title:j.title||"",
      loggedIn:/New chat|What’s on the agenda|What can I help with|Upgrade|Free/i.test(t)&&!/Log in or sign up|Continue with Google|Enter your email|Check your inbox|verification code/i.test(t),
      login:/Log in or sign up|Continue with Google|Enter your email/i.test(t),
      verification:/Check your inbox|verification code|Enter the code/i.test(t),
      security:/captcha|verify you are human|security check|unusual activity/i.test(t),
      text:t.slice(0,700)
    }));
  } else {
    console.log("CHATGPT_PAGE missing; no registered ChatGPT session");
  }
  ws.close();
});
NODE
