cd /home/runner/workspace/ai-agent && set -eu
printf '%s\n' '=== FORCE RELOAD SERVER MODULES ==='
oldpid=$(ps -eo pid,args | awk '$2 ~ /node$/ && $3=="backend/server.js" {print $1; exit}')
echo "old_server_pid=${oldpid:-none}"
if [ -n "${oldpid:-}" ]; then kill -KILL "$oldpid" 2>/dev/null || true; fi
for i in $(seq 1 15); do
  alive=$(ps -eo pid,args | awk '$2 ~ /node$/ && $3=="backend/server.js" {print $1; exit}')
  if [ -z "${alive:-}" ] || [ "$alive" != "$oldpid" ]; then break; fi
  sleep 1
done
alive=$(ps -eo pid,args | awk '$2 ~ /node$/ && $3=="backend/server.js" {print $1; exit}')
if [ -z "${alive:-}" ]; then
  echo 'starting_server_manually=true'
  nohup node backend/server.js >/tmp/ai-agent-chatgpt-login-restart.log 2>&1 &
  alive=$!
else
  echo "server_restarted_by_supervisor_pid=$alive"
fi
for i in $(seq 1 30); do
  if curl -fsS http://127.0.0.1:3000/health >/dev/null 2>&1; then break; fi
  sleep 1
done
printf 'active_server_pid=%s\n' "$alive"
curl -sS -o /dev/null -w 'health_http=%{http_code}\n' http://127.0.0.1:3000/health
printf '%s\n' '=== CREATE HEADED CHATGPT SESSION ==='
node --input-type=module <<'NODE'
import WebSocket from 'ws';
const key=process.env.BROWSER_API_KEY,base='http://127.0.0.1:3000';
if(!key){console.log('BROWSER_API_KEY missing');process.exit(2)}
const headers={'content-type':'application/json','x-browser-key':key};
let r=await fetch(base+'/browser/session',{method:'POST',headers,body:JSON.stringify({headless:false,displayPolicy:'foreground',importance:100,requiresUserAction:true})});
let j=await r.json().catch(()=>({}));
console.log('create_http='+r.status,JSON.stringify(j));
if(!r.ok||!j.sessionId)process.exit(3);
const id=j.sessionId;
r=await fetch(base+'/browser/open',{method:'POST',headers,body:JSON.stringify({sessionId:id,url:'https://chatgpt.com/',displayPolicy:'foreground',importance:100,requiresUserAction:true})});
j=await r.json().catch(()=>({}));
console.log('open_http='+r.status,JSON.stringify({sessionId:id,url:j.url,title:j.title,error:j.error}));
await new Promise(resolve=>setTimeout(resolve,3000));
r=await fetch(base+'/browser/read',{method:'POST',headers,body:JSON.stringify({sessionId:id})});
j=await r.json().catch(()=>({}));
console.log('read_http='+r.status,JSON.stringify({sessionId:id,url:j.url,title:j.title,challenge:/Just a moment|Verify you are human|security check/i.test(j.title+' '+j.text),loginVisible:/Log in|Sign up for free/i.test(j.text||''),text:(j.text||'').slice(0,700)}));
console.log('=== VERIFY LIVE KEYBOARD HANDLER WITHOUT INSERTING TEXT ===');
const ws=new WebSocket('ws://127.0.0.1:3000/browser/live');
let ack=false,done=false;
const timer=setTimeout(()=>{if(!done){done=true;console.log('user_text_handler_ack=false');ws.close()}},5000);
ws.on('open',()=>setTimeout(()=>ws.send(JSON.stringify({type:'attach',sessionId:id})),200));
ws.on('message',raw=>{
 let m;try{m=JSON.parse(raw.toString())}catch{return}
 if(m.type==='status'&&m.state==='streaming'&&!done){ws.send(JSON.stringify({type:'user-text',sessionId:id,text:''}))}
 if(m.type==='input-accepted'&&!done){ack=true;done=true;clearTimeout(timer);console.log('user_text_handler_ack=true');ws.close()}
});
ws.on('error',e=>{if(!done){done=true;clearTimeout(timer);console.log('websocket_error='+e.message)}});
await new Promise(resolve=>setTimeout(resolve,5500));
if(!done){done=true;clearTimeout(timer);console.log('user_text_handler_ack='+ack);ws.close()}
NODE
printf '%s\n' '=== VERIFY VIEWER INPUT BAR ==='
curl -fsS http://127.0.0.1:3000/browser-live-viewer.js | grep -q 'blv-user-text' && echo viewer_input_bar=served
printf '%s\n' '=== END ==='
