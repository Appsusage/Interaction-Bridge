import WebSocket from 'ws';
const key=process.env.BROWSER_API_KEY;
const ws=new WebSocket('ws://127.0.0.1:3000/browser/live');
const timer=setTimeout(()=>{console.log('WS_TIMEOUT');ws.close()},10000);
ws.on('message',async raw=>{
 const m=JSON.parse(raw.toString());
 if(m.type!=='sessions') return;
 console.log('SESSIONS',JSON.stringify(m.sessions));
 for(const s of m.sessions||[]){
   try{
     const r=await fetch('http://127.0.0.1:3000/browser/read',{method:'POST',headers:{'content-type':'application/json','x-browser-key':key},body:JSON.stringify({sessionId:s.sessionId})});
     console.log('PAGE',s.sessionId,'HTTP',r.status,(await r.text()).slice(0,4000));
   }catch(e){console.log('PAGE_READ_ERROR',e.message)}
 }
 clearTimeout(timer);ws.close();
});
ws.on('error',e=>{console.log('WS_ERROR',e.message);clearTimeout(timer)});
JS
