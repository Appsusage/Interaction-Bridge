import { WebSocketServer } from "ws";
import { StringDecoder } from "node:string_decoder";
const MAX_HISTORY_CHARS = 600000;
const history = [];
let historyChars = 0;
let attached = false;
export function attachBackendConsole(httpServer) {
  if (attached) return;
  attached = true;
  const wss = new WebSocketServer({ server: httpServer, path: "/backend-console/live" });
  function send(client, payload) { if (client.readyState === 1) client.send(JSON.stringify(payload)); }
  function record(stream, text) {
    if (!text) return;
    const item = { type: "chunk", stream, text };
    history.push(item); historyChars += text.length;
    while (historyChars > MAX_HISTORY_CHARS && history.length > 1) historyChars -= history.shift().text.length;
    const encoded = JSON.stringify(item);
    for (const client of wss.clients) if (client.readyState === 1) { try { client.send(encoded); } catch {} }
  }
  function intercept(streamName, stream) {
    const originalWrite = stream.write;
    const decoder = new StringDecoder("utf8");
    stream.write = function (chunk, encoding, callback) {
      const result = originalWrite.apply(this, arguments);
      try {
        const enc = typeof encoding === "string" ? encoding : "utf8";
        const buffer = typeof chunk === "string" ? Buffer.from(chunk, enc) :
          Buffer.isBuffer(chunk) ? chunk :
          ArrayBuffer.isView(chunk) ? Buffer.from(chunk.buffer, chunk.byteOffset, chunk.byteLength) : Buffer.from(String(chunk));
        record(streamName, decoder.write(buffer));
      } catch {}
      return result;
    };
  }
  intercept("stdout", process.stdout);
  intercept("stderr", process.stderr);
  wss.on("connection", (client, request) => {
    const origin = request.headers.origin, host = request.headers.host;
    if (origin && host) {
      try { if (new URL(origin).host !== host) { client.close(1008, "Origin not allowed"); return; } }
      catch { client.close(1008, "Invalid origin"); return; }
    }
    send(client, { type: "snapshot-start" });
    for (const item of history) send(client, item);
    send(client, { type: "snapshot-end" });
    send(client, { type: "status", state: "connected" });
  });
  return wss;
}

(() => {
  if (document.getElementById("backend-console-viewer")) return;
  const root = document.createElement("section");
  root.id = "backend-console-viewer";
  root.setAttribute("aria-label", "后端 Console 原始输出");
  root.innerHTML = '<header class="bcv-head"><span class="bcv-indicator"></span><strong>后端 Console</strong><span class="bcv-status">连接中…</span><button type="button" class="bcv-pause">暂停滚动</button><button type="button" class="bcv-clear">清空显示</button><button type="button" class="bcv-min" aria-label="最小化">−</button><button type="button" class="bcv-hide" aria-label="隐藏窗口">×</button></header><pre class="bcv-output" aria-live="off"></pre><footer class="bcv-foot">原始 stdout / stderr · 实时传输 · 清除显示不会清除后端日志</footer>';
  document.body.appendChild(root);
  const output = root.querySelector(".bcv-output"), status = root.querySelector(".bcv-status"), indicator = root.querySelector(".bcv-indicator"), pauseButton = root.querySelector(".bcv-pause");
  let ws, paused = false, minimized = false, reconnectTimer, drag = null, queued = [], totalChars = 0;
  const MAX_DISPLAY_CHARS = 600000;
  function addChunk(stream, text) {
    if (!text) return;
    if (paused) { queued.push({stream,text}); return; }
    const nearBottom = output.scrollHeight - output.scrollTop - output.clientHeight < 60;
    const node = document.createElement("span");
    node.className = stream === "stderr" ? "bcv-stderr" : "bcv-stdout";
    node.textContent = text;
    output.appendChild(node); totalChars += text.length;
    while (totalChars > MAX_DISPLAY_CHARS && output.firstChild) {
      const first = output.firstChild, length = first.textContent.length, excess = totalChars - MAX_DISPLAY_CHARS;
      if (length <= excess) { totalChars -= length; first.remove(); }
      else { first.textContent = first.textContent.slice(excess); totalChars = MAX_DISPLAY_CHARS; }
    }
    if (nearBottom) output.scrollTop = output.scrollHeight;
  }
  function connect() {
    clearTimeout(reconnectTimer);
    const protocol = location.protocol === "https:" ? "wss:" : "ws:";
    ws = new WebSocket(protocol + "//" + location.host + "/backend-console/live");
    ws.onopen = () => { status.textContent = "已连接"; indicator.classList.add("bcv-live"); };
    ws.onclose = () => { status.textContent = "连接中断，重连中…"; indicator.classList.remove("bcv-live"); reconnectTimer = setTimeout(connect, 1500); };
    ws.onerror = () => { try { ws.close(); } catch {} };
    ws.onmessage = event => {
      let m; try { m = JSON.parse(event.data); } catch { return; }
      if (m.type === "snapshot-start") { output.replaceChildren(); totalChars = 0; }
      else if (m.type === "snapshot-end") output.scrollTop = output.scrollHeight;
      else if (m.type === "chunk") addChunk(m.stream, m.text);
      else if (m.type === "status") status.textContent = "已连接 · 实时";
    };
  }
  pauseButton.addEventListener("click", () => {
    paused = !paused; pauseButton.textContent = paused ? "继续显示" : "暂停滚动";
    if (!paused) { const pending = queued; queued = []; for (const item of pending) addChunk(item.stream, item.text); }
  });
  root.querySelector(".bcv-clear").addEventListener("click", () => { output.replaceChildren(); totalChars = 0; queued = []; });
  root.querySelector(".bcv-min").addEventListener("click", () => { minimized = !minimized; root.classList.toggle("bcv-minimized", minimized); root.querySelector(".bcv-min").textContent = minimized ? "□" : "−"; });
  root.querySelector(".bcv-hide").addEventListener("click", () => root.classList.add("bcv-hidden"));
  const head = root.querySelector(".bcv-head");
  head.addEventListener("pointerdown", e => { if (e.target.closest("button")) return; const r = root.getBoundingClientRect(); drag = {x:e.clientX,y:e.clientY,left:r.left,top:r.top}; head.setPointerCapture(e.pointerId); });
  head.addEventListener("pointermove", e => { if (!drag) return; root.style.left = Math.max(0,Math.min(innerWidth-root.offsetWidth,drag.left+e.clientX-drag.x))+"px"; root.style.top = Math.max(0,Math.min(innerHeight-36,drag.top+e.clientY-drag.y))+"px"; root.style.right="auto"; root.style.bottom="auto"; });
  head.addEventListener("pointerup", () => { drag = null; });
  head.addEventListener("pointercancel", () => { drag = null; });
  window.toggleBackendConsoleViewer = () => { root.classList.toggle("bcv-hidden"); return !root.classList.contains("bcv-hidden"); };
  connect();
})();
