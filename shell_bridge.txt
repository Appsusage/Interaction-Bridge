      if (orb._newbornExploreTimer) clearTimeout(orb._newbornExploreTimer);
      const explore = () => {
        if (!orb.isConnected || !orb.classList.contains("limbo-pet-newborn") || orb.dataset.gaze !== "forward") return;
        const x = 2 + Math.floor(Math.random() * 7);
        const y = 2 + Math.floor(Math.random() * 12);
        orb.style.setProperty("--newborn-pupil-x", x + "px");
        orb.style.setProperty("--newborn-pupil-y", y + "px");

      };
      orb._newbornExploreTimer = setTimeout(explore, 450);

      clearTimeout(orb._newbornExploreTimer);
      orb._newbornExploreTimer = null;
    }
'''
