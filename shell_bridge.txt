# Independent gesture controller: no changes to the five quick-button elements or handlers.
gesture = r'''
// Independent floating-ball gestures. The existing five quick actions remain untouched.
(() => {
  const ball = document.getElementById("mainFloat");
  const settings = document.getElementById("aiAgentSettingsPanel");
  const outerRing = document.getElementById("futureOuterRing");
  if (!ball) return;
  const threshold = 48;
  let active = null, lens = null, lensClone = null, suppressClickUntil = 0;

  const closeSettings = () => { if (settings) settings.hidden = true; };
  const hideOuterRing = () => { if (outerRing) outerRing.hidden = true; };
  function clearLens() {
    if (lens) lens.hidden = true;
    if (lensClone) lensClone.remove();
    lensClone = null;
  }
  function showLens(x, y) {
    if (!lens) {
      lens = document.createElement("div");
      lens.className = "gesture-magnifier";
      lens.hidden = true;
      lens.setAttribute("aria-hidden", "true");
      lens.innerHTML = '<div class="gesture-magnifier-inner"></div>';
      document.body.appendChild(lens);
    }
    lens.hidden = false;
    lens.style.left = (x - 76) + "px";
    lens.style.top = (y - 76) + "px";
    if (!lensClone) {
      lensClone = document.body.cloneNode(true);
      lensClone.querySelectorAll("script,style,iframe,video,audio,canvas,#aiAgentSettingsPanel,#futureOuterRing,.gesture-magnifier,.limbo-pet").forEach(n => n.remove());
      lensClone.querySelectorAll("[id]").forEach(n => n.removeAttribute("id"));
      lensClone.style.cssText = "position:absolute!important;left:0!important;top:0!important;width:" + innerWidth + "px!important;max-width:none!important;margin:0!important;transform:scale(1.65)!important;transform-origin:top left!important;pointer-events:none!important;";
      lens.firstElementChild.replaceChildren(lensClone);
    }
    lensClone.style.left = (-(x * 1.65 - 76)) + "px";
    lensClone.style.top = (-(y * 1.65 - 76)) + "px";
  }
  function runGesture(kind) {
    closeSettings();
    if (kind === "left") {
      if (outerRing) outerRing.hidden = false; // reserved outer arc; does not open the existing five-button menu
    } else if (kind === "up") {
      hideOuterRing();
      if (settings) settings.hidden = false;
    } else if (kind === "down") {
      hideOuterRing();
      if (window.AICortexPetLimbo) window.AICortexPetLimbo.spawnFrom(ball);
    }
  }
  ball.addEventListener("pointerdown", event => {
    if (event.button !== undefined && event.button !== 0) return;
    active = { id: event.pointerId, x: event.clientX, y: event.clientY, kind: null, moved: false };
    try { ball.setPointerCapture(event.pointerId); } catch {}
  });
  ball.addEventListener("pointermove", event => {
    if (!active || active.id !== event.pointerId) return;
    const dx = event.clientX - active.x, dy = event.clientY - active.y;
    if (Math.hypot(dx, dy) < 14) return;
    active.moved = true;
    if (dx < -threshold && Math.abs(dx) > Math.abs(dy) * 1.15) active.kind = "left";
    else if (dy < -threshold && Math.abs(dy) > Math.abs(dx) * 1.15) active.kind = "up";
    else if (dy > threshold && Math.abs(dy) > Math.abs(dx) * 1.15) active.kind = "down";
    else active.kind = null;
    if (active.kind) clearLens(); else showLens(event.clientX, event.clientY);
  });
  function finish(event) {
    if (!active || (event.pointerId !== undefined && active.id !== event.pointerId)) return;
    const done = active; active = null; clearLens();
    if (done.moved) {
      suppressClickUntil = Date.now() + 500;
      if (done.kind) { event.preventDefault(); runGesture(done.kind); }
    }
  }
  ball.addEventListener("pointerup", finish);
  ball.addEventListener("pointercancel", finish);
  ball.addEventListener("lostpointercapture", event => { if (active && active.id === event.pointerId) finish(event); });
  // Keep the original click handler intact; only suppress the synthetic click after a drag.
  ball.addEventListener("click", event => {
    if (Date.now() < suppressClickUntil) { event.preventDefault(); event.stopImmediatePropagation(); }
  }, true);
  if (settings) {
    settings.querySelector("[data-close-ai-settings]")?.addEventListener("click", closeSettings);
    settings.addEventListener("pointerdown", event => { if (event.target === settings) closeSettings(); });
  }
  document.addEventListener("pointerdown", event => {
    if (outerRing && !outerRing.hidden && !outerRing.contains(event.target) && !ball.contains(event.target)) hideOuterRing();
  });
  document.addEventListener("keydown", event => { if (event.key === "Escape") { closeSettings(); hideOuterRing(); } });
})();
'''
