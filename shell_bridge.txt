printf '%s\n' '=== external Replit health endpoint ==='
domain="${REPLIT_DEV_DOMAIN:-}"
if [ -n "$domain" ]; then
  base="$domain"
  case "$base" in http://*|https://*) ;; *) base="https://$base" ;; esac
  for route in /health /heartbeat/health; do
    code="$(curl -sS --max-time 12 -o /tmp/ext-health-body -w '%{http_code}' "$base$route" || true)"
    echo "$route HTTP $code"
    cat /tmp/ext-health-body 2>/dev/null || true
    printf '\n'
  done
else
  echo 'REPLIT_DEV_DOMAIN missing'
fi
