const baseUrl = String(process.env.HEARTBEAT_BASE_URL || "").replace(/\/+$/, "");
const token = process.env.HEARTBEAT_SUPERVISOR_TOKEN || "";

async function request(path, method) {
  const response = await fetch(baseUrl + path, {
    method,
    headers: { Authorization: "Bearer " + token, Accept: "application/json" },
    signal: AbortSignal.timeout(15000)
  });
  const rawLog = await response.text();
  let data = null;
  try { data = rawLog ? JSON.parse(rawLog) : null; } catch {}
  return { ok: response.ok, status: response.status, rawLog, data };
}

async function main() {
  if (!baseUrl || !token) {
    throw new Error("Missing HEARTBEAT_BASE_URL or HEARTBEAT_SUPERVISOR_TOKEN environment variable.");
  }
  const health = await request("/heartbeat/health", "GET");
  if (health.ok && health.data && health.data.heartbeat === "running") {
    console.log("HeartBeat is running. PID=" + health.data.pid);
    return;
  }
  console.error("GET /heartbeat/health HTTP " + health.status + "\n" + health.rawLog);
  const start = await request("/heartbeat/start", "POST");
  if (!start.ok) {
    throw new Error("POST /heartbeat/start HTTP " + start.status + "\n" + start.rawLog);
  }
  console.log("POST /heartbeat/start HTTP " + start.status + "\n" + start.rawLog);
}

main().catch(error => {
  console.error(error && error.stack ? error.stack : String(error));
  process.exitCode = 1;
});

import fs from "node:fs";
import path from "node:path";
import https from "node:https";
import { fileURLToPath } from "node:url";

const HERE = path.dirname(fileURLToPath(import.meta.url));
const STATE_FILE = path.join(HERE, "supervisor-state.json");
const FAILURE_FILE = path.join(HERE, "feedback", "failure1.txt");
const SUPERVISOR_FILE = path.join(HERE, "supervisor.js");
const WORKFLOW_FILE = path.join(HERE, "supervisor-workflow.yml");
const CHECK_MS = 360000;
const REPOSITORIES = ["Appsusage/SiriusA", "Appsusage/SiriusB"];
let runningCycle = false;

function appendFailure(rawLog, label) {
  try {
    fs.mkdirSync(path.dirname(FAILURE_FILE), { recursive: true });
    fs.appendFileSync(FAILURE_FILE,
      "\n[" + new Date().toISOString() + "] " + label + "\n" +
      String(rawLog || "No raw error output was available.") + "\n", "utf8");
  } catch (error) {
    console.error("Could not save complete failure log:", error.stack || error, "\nOriginal log:", rawLog);
  }
}

function readState() {
  try {
    const value = JSON.parse(fs.readFileSync(STATE_FILE, "utf8"));
    return { activeRepo: REPOSITORIES.includes(value.activeRepo) ? value.activeRepo : REPOSITORIES[0] };
  } catch {
    return { activeRepo: REPOSITORIES[0] };
  }
}

function writeLocalState(state) {
  const temporary = STATE_FILE + ".tmp";
  fs.writeFileSync(temporary, JSON.stringify(state, null, 2) + "\n", "utf8");
  fs.renameSync(temporary, STATE_FILE);
}

function githubRequest(method, repo, remotePath, body) {
  const token = process.env.GITHUB_TOKEN || process.env.GH_TOKEN;
  if (!token) {
    return Promise.resolve({ ok: false, rawLog: "Missing GITHUB_TOKEN or GH_TOKEN environment variable." });
  }
  const encodedPath = remotePath.split("/").map(encodeURIComponent).join("/");
  const requestBody = body ? JSON.stringify(body) : "";
  return new Promise(resolve => {
    const request = https.request({
      hostname: "api.github.com",
      path: "/repos/" + repo + "/contents/" + encodedPath,
      method,
      headers: {
        "User-Agent": "ai-agent-supervisor-router",
        Authorization: "Bearer " + token,
        Accept: "application/vnd.github+json",
        ...(body ? { "Content-Type": "application/json", "Content-Length": Buffer.byteLength(requestBody) } : {})
      },
      timeout: 15000
    }, response => {
      const chunks = [];
      response.setEncoding("utf8");
      response.on("data", chunk => chunks.push(chunk));
      response.on("end", () => {
        const raw = chunks.join("");
        let data = null;
        try { data = raw ? JSON.parse(raw) : null; } catch {}
        resolve({
          ok: response.statusCode >= 200 && response.statusCode < 300,
          statusCode: response.statusCode,
          data,
          rawLog: "GitHub " + method + " " + repo + "/" + remotePath +
            " returned HTTP " + response.statusCode + "\n" + raw
        });
      });
    });
    request.on("error", error => resolve({ ok: false, rawLog: error.stack || String(error) }));
    request.on("timeout", () => request.destroy(new Error("GitHub request timed out")));
    if (body) request.write(requestBody);
    request.end();
  });
}

async function syncFile(repo, remotePath, localPath, contentOverride = null) {
  const content = contentOverride === null ? fs.readFileSync(localPath, "utf8") : contentOverride;
  const current = await githubRequest("GET", repo, remotePath);
  if (current.ok && current.data && typeof current.data.content === "string") {
    const remote = Buffer.from(current.data.content.replace(/\s/g, ""), "base64").toString("utf8");
    if (remote === content) return { ok: true };
  } else if (current.statusCode !== 404) {
    return { ok: false, rawLog: current.rawLog };
  }
  const body = {
    message: "Ensure external HeartBeat supervisor source",
    content: Buffer.from(content, "utf8").toString("base64")
  };
  if (current.ok && current.data && current.data.sha) body.sha = current.data.sha;
  const saved = await githubRequest("PUT", repo, remotePath, body);
  return saved.ok ? { ok: true } : { ok: false, rawLog: saved.rawLog };
}

async function syncSupervisor(repo, activeState) {
  for (const [remotePath, localPath] of [
    ["supervisor.js", SUPERVISOR_FILE],
    [".github/workflows/supervisor.yml", WORKFLOW_FILE]
  ]) {
    const result = await syncFile(repo, remotePath, localPath);
    if (!result.ok) return { ok: false, rawLog: result.rawLog };
  }
  const stateResult = await syncFile(repo, "supervisor-state.json", null,
    JSON.stringify({ activeRepo: activeState }, null, 2) + "\n");
  if (!stateResult.ok) return { ok: false, rawLog: stateResult.rawLog };
  return { ok: true };
}

async function runCycle() {
  if (runningCycle) return;
  runningCycle = true;
  try {
    const current = readState().activeRepo;
    const other = REPOSITORIES.find(repo => repo !== current);
    let selected = null;
    for (const repo of [current, other]) {
      const result = await syncSupervisor(repo, repo);
      if (result.ok) {
        selected = repo;
        break;
      }
      appendFailure(result.rawLog, "Supervisor create/repair failed in " + repo);
    }
    if (!selected) {
      appendFailure("Both Sirius repositories failed in this cycle; stopping without further retries.",
        "Supervisor rotation exhausted");
      return;
    }
    writeLocalState({ activeRepo: selected });
    const standby = REPOSITORIES.find(repo => repo !== selected);
    const standbyState = await syncFile(standby, "supervisor-state.json", null,
      JSON.stringify({ activeRepo: selected }, null, 2) + "\n");
    if (!standbyState.ok) {
      appendFailure(standbyState.rawLog, "Could not update standby repository routing state " + standby);
    }
  } catch (error) {
    appendFailure(error.stack || String(error), "Supervisor router cycle failed");
  } finally {
    runningCycle = false;
  }
}

if (!fs.existsSync(STATE_FILE)) writeLocalState({ activeRepo: REPOSITORIES[0] });
runCycle();
setInterval(() => { runCycle(); }, CHECK_MS);
EOF
