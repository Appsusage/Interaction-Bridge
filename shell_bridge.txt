const key=process.env.BROWSER_API_KEY;
if(!key){console.log('BLOCKED: BROWSER_API_KEY missing');process.exit(2);}
async function post(path,body){
 const r=await fetch('http://127.0.0.1:3000'+path,{method:'POST',headers:{'content-type':'application/json','x-browser-key':key},body:JSON.stringify(body)});
 const t=await r.text(); let j; try{j=JSON.parse(t)}catch{j={raw:t}};
 console.log('POST '+path+' status='+r.status+' response='+JSON.stringify(j).slice(0,3500));
 if(!r.ok) throw new Error(path+' returned HTTP '+r.status);
 return j;
}
const created=await post('/browser/session',{headless:false});
if(!created.sessionId) throw new Error('Session creation returned no sessionId');
const id=created.sessionId;
console.log('NEW_HEADED_SESSION='+id);
await post('/browser/open',{sessionId:id,url:'https://chatgpt.com/'});
await new Promise(r=>setTimeout(r,7000));
const read=await post('/browser/read',{sessionId:id});
console.log('FINAL_PAGE='+JSON.stringify({sessionId:id,url:read.url,title:read.title,text:String(read.text||'').slice(0,1600)}));
console.log('--- matching browser process launch flags ---');
const {execFileSync}=await import('node:child_process');
const ps=execFileSync('ps',['-eo','pid,ppid,args'],{encoding:'utf8'});
for(const line of ps.split('\n')) if(line.includes('chromium') && !line.includes('--type=') && !line.includes('crashpad')) console.log(line.slice(0,500));
