cd /home/runner/workspace/ai-agent
node --input-type=module <<'EOF'
import {
  createBrowserSession,
  getBrowserPage,
  openPage,
  readPage,
  clickElement,
  typeText,
  scrollPage,
  captureScreenshot,
  closeBrowserSession
} from "./backend/extension/browser/browser_service.js";
const { sessionId } = await createBrowserSession();
try {
  const page = getBrowserPage(sessionId);
  const opened = await openPage(page, "https://example.com");
  const read = await readPage(page);
  await scrollPage(page, 300);
  await page.setContent('<input id="entry"><button id="go" onclick="document.body.dataset.clicked=document.querySelector(\'#entry\').value">Go</button>');
  await typeText(page, "#entry", "browser-test");
  await clickElement(page, "#go");
  const clicked = await page.locator("body").getAttribute("data-clicked");
  const screenshotPath = "/tmp/ai-agent-browser-smoke.png";
  await captureScreenshot(page, screenshotPath);
  console.log(JSON.stringify({ openedTitle: opened.title, readTitle: read.title, bodyTextLength: read.text.length, typedAndClicked: clicked === "browser-test", screenshotExists: true }));
} finally {
  await closeBrowserSession(sessionId);
}
EOF
node --check backend/extension/browser/browser_service.js
node --check backend/server.js
