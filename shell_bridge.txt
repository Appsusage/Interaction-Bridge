cd /home/runner/workspace/ai-agent
printf '%s\n' '=== token presence only ==='
if [ -n "${HEARTBEAT_SUPERVISOR_TOKEN:-}" ]; then echo 'Shell token=SET'; else echo 'Shell token=MISSING'; fi
server_pid="$(ps -eo pid,args | awk '$0 ~ /node backend\/server\.js/ && $0 !~ /awk/ {print $1; exit}')"
if [ -n "$server_pid" ] && [ -r "/proc/$server_pid/environ" ]; then
 tr '\0' '\n' <"/proc/$server_pid/environ" | grep -q '^HEARTBEAT_SUPERVISOR_TOKEN=.' && echo 'server token=SET' || echo 'server token=MISSING'
fi
printf '%s\n' '=== latest SiriusA workflow runs ==='
gh_token="${GITHUB_TOKEN:-${GH_TOKEN:-}}"
if [ -n "$gh_token" ]; then
 curl -sS --max-time 15 -H "Authorization: Bearer $gh_token" -H 'Accept: application/vnd.github+json' 'https://api.github.com/repos/Appsusage/SiriusA/actions/workflows/supervisor.yml/runs?per_page=5' -o /tmp/hb-runs-auth
 node -e 'const x=JSON.parse(require("fs").readFileSync("/tmp/hb-runs-auth","utf8")); if(x.workflow_runs) console.log(JSON.stringify(x.workflow_runs.map(r=>({id:r.id,status:r.status,conclusion:r.conclusion,event:r.event,created_at:r.created_at,html_url:r.html_url})),null,2)); else console.log(JSON.stringify(x))'
 printf '%s\n' '=== Actions secret names only ==='
 curl -sS --max-time 15 -H "Authorization: Bearer $gh_token" -H 'Accept: application/vnd.github+json' 'https://api.github.com/repos/Appsusage/SiriusA/actions/secrets?per_page=100' -o /tmp/hb-secrets
 node -e 'const x=JSON.parse(require("fs").readFileSync("/tmp/hb-secrets","utf8")); if(x.secrets) console.log(x.secrets.map(s=>s.name).join("\n") || "(no secrets)"); else console.log(JSON.stringify(x))'
else echo 'GITHUB_TOKEN unavailable'; fi

ps -eo pid,args | grep -E '[n]ode backend/heartbeat/supervisor-router\.js' || true
