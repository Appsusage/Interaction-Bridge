#!/bin/sh
set +e
cd /home/runner/workspace/ai-agent || exit 90
echo "=== current server process ==="
pgrep -af 'node backend/server.js' || true
echo "=== server port 3000 ==="
if command -v curl >/dev/null 2>&1; then
  curl -sS -o /tmp/pet_cursor_probe.txt -w 'GET / status_http=%{http_code}\n' --max-time 3 http://127.0.0.1:3000/ 2>&1
  echo "=== protected route probe (no credentials; 401/403 expected) ==="
  for route in /browser/cursor/status /browser/recognition/inspect; do
    code=$(curl -sS -o /tmp/pet_cursor_probe.txt -w '%{http_code}' --max-time 3 -X POST "http://127.0.0.1:3000$route" -H 'Content-Type: application/json' --data '{}' 2>&1)
    echo "$route http=$code"
  done
else
  echo "curl unavailable"
fi
echo "=== source route/import checks ==="
grep -n -E 'acquirePetCursor|capturePetRecognitionFrame|/browser/cursor/|/browser/recognition/' backend/server.js
echo "=== no destructive browser actions executed ==="
echo PET_API_PROBE_COMPLETE
