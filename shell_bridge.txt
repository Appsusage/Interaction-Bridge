node --input-type=module <<'NODE'
import { chromium } from 'playwright';
const exe = process.env.PLAYWRIGHT_EXECUTABLE_PATH || '/repl/tools/bin/chromium';
let browser;
try {
  browser = await chromium.launch({headless:false, executablePath:exe, timeout:15000});
  const page = await browser.newPage();
  await page.goto('https://chatgpt.com/', {waitUntil:'domcontentloaded',timeout:30000});
  await page.waitForTimeout(6000);
  console.log('headful_launch=success');
  console.log('url='+page.url());
  console.log('title='+await page.title());
  console.log('body_excerpt='+(await page.locator('body').innerText().catch(()=>'' )).slice(0,1000));
} catch(e) {
  console.log('headful_launch=failed');
  console.log('error='+String(e.message).slice(0,2500));
  process.exitCode=1;
} finally { if(browser) await browser.close().catch(()=>{}); }
NODE
