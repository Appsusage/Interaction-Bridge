import { browserActivity, getBrowserPage, listBrowserSessions } from "./browser_service.js";

export function attachBrowserLive(server) {
  const wss = new WebSocketServer({ server, path: "/browser/live" });
  const clients = new Set();
  const streams = new Map();

  function send(client, message) {
    if (client.readyState === 1) client.send(JSON.stringify(message));
  }
  function broadcast(message) {
    for (const client of clients) send(client, message);
  }
  async function stopStream(client) {
    const state = client._browserLive;
    if (!state) return;
    client._browserLive = null;
    const stream = streams.get(state.sessionId);
    if (stream && stream.client === client) {
      streams.delete(state.sessionId);
      try { await stream.cdp.send("Page.stopScreencast"); } catch {}
      try { await stream.cdp.detach(); } catch {}
    }
  }
  async function attach(client, sessionId) {
    await stopStream(client);
    let page;
    try { page = getBrowserPage(sessionId); }
    catch (error) { send(client, { type: "error", message: error.message }); return; }
    const prior = streams.get(sessionId);
    if (prior) {
      try { await prior.cdp.send("Page.stopScreencast"); } catch {}
      try { await prior.cdp.detach(); } catch {}
      if (prior.client && prior.client !== client) {
        prior.client._browserLive = null;
        send(prior.client, { type: "status", state: "stream-replaced" });
      }
    }
    try {
      const cdp = await page.context().newCDPSession(page);
      const state = { sessionId, cdp };
      client._browserLive = state;
      streams.set(sessionId, { client, cdp });
      cdp.on("Page.screencastFrame", async frame => {
        send(client, { type: "frame", sessionId, data: frame.data, metadata: frame.metadata });
        try { await cdp.send("Page.screencastFrameAck", { sessionId: frame.sessionId }); } catch {}
      });
      cdp.on("Page.frameNavigated", event => {
        if (event.frame?.url) send(client, { type: "navigation", sessionId, url: event.frame.url });
      });
      await cdp.send("Page.enable");
      await cdp.send("Page.startScreencast", { format: "jpeg", quality: 65, maxWidth: 1100, maxHeight: 800, everyNthFrame: 1 });
      send(client, { type: "status", state: "streaming", sessionId, url: page.url(), title: await page.title() });
      send(client, { type: "activity", sessionId, action: "已连接实时画面", state: "进行中", at: Date.now() });
    } catch (error) {
      await stopStream(client);
      send(client, { type: "error", message: "无法启动 Chromium 实时画面：" + error.message });
    }
  }

  browserActivity.on("activity", event => broadcast({ type: "activity", ...event }));
  browserActivity.on("session-created", event => broadcast({ type: "session-created", ...event }));
  browserActivity.on("session-closed", event => broadcast({ type: "session-closed", ...event }));

  wss.on("connection", (client, request) => {
    const origin = request.headers.origin;
    const host = request.headers.host;
    if (origin && host) {
      try { if (new URL(origin).host !== host) { client.close(1008, "Origin not allowed"); return; } }
      catch { client.close(1008, "Invalid origin"); return; }
    }
    clients.add(client);
    send(client, { type: "sessions", sessions: listBrowserSessions() });
    client.on("message", raw => {
      let message;
      try { message = JSON.parse(raw.toString()); } catch { return; }
      if (message.type === "attach" && typeof message.sessionId === "string") attach(client, message.sessionId);
      if (message.type === "detach") stopStream(client);
    });
    client.on("close", () => { clients.delete(client); stopStream(client); });
    client.on("error", () => { clients.delete(client); stopStream(client); });
  });
  return wss;
}
''')
