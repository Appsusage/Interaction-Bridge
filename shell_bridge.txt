#!/bin/sh
set +e
cd /home/runner/workspace/ai-agent || exit 90
echo "=== process listening on 3000 ==="
if command -v ss >/dev/null 2>&1; then ss -ltnp 2>&1 | grep ':3000' || true; fi
if command -v lsof >/dev/null 2>&1; then lsof -nP -iTCP:3000 -sTCP:LISTEN 2>&1 || true; fi
echo "=== node processes ==="
ps -eo pid,ppid,args | grep '[n]ode' || true
echo "=== route declarations ==="
grep -n -E 'app\.post\("/browser/(cursor|recognition)/|requireBrowserKey' backend/server.js | tail -n 30
echo "=== package startup scripts ==="
node -e "const p=require('./package.json'); console.log(JSON.stringify({scripts:p.scripts,type:p.type},null,2))" 2>&1
echo "=== root response headers ==="
curl -sS -D - -o /tmp/pet_root_body.txt --max-time 5 http://127.0.0.1:3000/ 2>&1 | head -n 20
for route in /browser/cursor/status /browser/recognition/inspect; do
 echo "--- $route"
 curl -sS -i --max-time 5 -X POST "http://127.0.0.1:3000$route" -H 'Content-Type: application/json' --data '{}' 2>&1 | head -n 20
done
echo "=== restart log ==="
tail -n 60 /tmp/ai-agent-pet-api-restart.log 2>/dev/null || true
echo ROUTE_404_DIAG_COMPLETE
