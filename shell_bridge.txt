cd /home/runner/workspace/ai-agent && set -eu
KEY="${BROWSER_API_KEY:-}"
test -n "$KEY"
SESSION_ID='1792211391529-s2y2p73y89n'
printf '%s\n' '=== CAPTURE LIVE SERVER BROWSER SCREEN ==='
R=$(curl -fsS --max-time 30 -X POST http://127.0.0.1:3000/browser/screenshot -H "content-type: application/json" -H "x-browser-key: $KEY" -d "{\"sessionId\":\"$SESSION_ID\"}")
printf 'screenshot_api=%s\n' "$R"
P=$(printf '%s' "$R" | node -e 'let s="";process.stdin.on("data",d=>s+=d).on("end",()=>{try{process.stdout.write(JSON.parse(s).path||"")}catch{}})')
test -n "$P"
python3 - "$P" <<'PY'
import sys,base64
from PIL import Image
from io import BytesIO
im=Image.open(sys.argv[1]).convert("RGB")
im.thumbnail((900,650))
out=BytesIO()
im.save(out,format="JPEG",quality=38,optimize=True)
print("CHATGPT_SCREENSHOT_BASE64_BEGIN")
print(base64.b64encode(out.getvalue()).decode())
print("CHATGPT_SCREENSHOT_BASE64_END")
PY
