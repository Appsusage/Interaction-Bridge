#!/bin/sh
set +e
cd /home/runner/workspace/ai-agent || exit 90
OUT=/tmp/pet_next_step_inspection.txt
{
  echo "=== browser_live.js lines 1-260 ==="
  sed -n '1,260p' backend/extension/browser/browser_live.js
  echo
  echo "=== browser_service.js lines 1-240 ==="
  sed -n '1,240p' backend/extension/browser/browser_service.js
  echo
  echo "=== pet.js symbols and relevant sections ==="
  grep -nE 'setInterval|requestAnimationFrame|function |const .*=>|addEventListener|window\.|globalThis|module.exports|export ' backend/extension/pet/limbo/pet.js | head -220
  echo
  echo "=== pet.js lines 1-260 ==="
  sed -n '1,260p' backend/extension/pet/limbo/pet.js
  echo
  echo "=== package.json ==="
  cat package.json
  echo
  echo "=== server routes/imports relevant to browser/pet ==="
  grep -nE 'browser_live|browser_service|pet|express\.Router|app\.(get|post|use)' backend/server.js | head -180
  echo
  echo "=== frontend pet script refs ==="
  grep -nE 'pet\.js|pet\.css|browser' frontend/index.html | head -100
  echo
  echo "=== existing recognition / image APIs references ==="
  grep -RniE 'screenshot\(|page\.screenshot|vision|image.*recogn|recogn.*image|Input\.dispatchMouseEvent|connectOverCDP|newPage\(' backend/extension --include='*.js' | head -180
} > "$OUT" 2>&1
cat "$OUT"
