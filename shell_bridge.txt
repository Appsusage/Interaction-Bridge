python3 - <<'PY'
from pathlib import Path
p = Path("backend/extension/upgrade/tool_selector.js")
s = p.read_text()
anchor = '''export function selectToolChain(action){

    // =========================
    // Robot：直接执行'''
replacement = '''export function selectToolChain(action){

    // Browser：直接调用已注册的浏览器工具
    if (typeof action.type === "string" && action.type.startsWith("browser_")) {
        return {
            chain: [action.type],
            reason: { type: "direct_browser" }
        };
    }

    // =========================
    // Robot：直接执行'''
if 'type: "direct_browser"' in s:
    raise SystemExit("BROWSER_TOOL_ROUTING_ALREADY_EXISTS; no changes made")
if anchor not in s:
    raise SystemExit("TOOL_SELECTOR_ANCHOR_NOT_FOUND; no changes made")
p.write_text(s.replace(anchor, replacement, 1))
PY
