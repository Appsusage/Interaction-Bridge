python3 - <<'PY'
from pathlib import Path
p=Path("backend/extension/browser/chatgpt_live.js")
s=p.read_text()
s=s.replace('const getLoginEmail = () => process.env.CHATGPT_LOGIN_EMAIL || "frankii@robotomail.co";',
            'const getLoginEmail = () => String(process.env.CHATGPT_LOGIN_EMAIL || "").trim();',1)
start=s.index("async function beginEmailLogin(page) {")
end=s.index("/**\n * Persistent ChatGPT login playbook:", start)
replacement='''async function beginEmailLogin(page) {
  if (loginFlowActive) return;
  loginFlowActive = true;
  emitStatus("进行中", { step: "open-login" });
  try {
    if (!/auth\\\\.openai\\\\.com|\\\\/auth\\\\/login/i.test(page.url())) {
      await page.goto(CHATGPT_LOGIN, { waitUntil: "domcontentloaded", timeout: 30000 });
    }
    await wait(1200);
    const title = await page.title().catch(() => "");
    const text = await pageText(page);
    if (/just a moment|verify you are human|security check|unusual activity/i.test(title + " " + text)) {
      emitStatus("waiting", { step: "security-check", note: "ChatGPT 正在显示安全检查页面；不会绕过检查，请用户在实时浏览器中正常完成。" });
      return;
    }
    const emailField = page.locator('input[type="email"]').first();
    const loginEmail = getLoginEmail();
    if (await emailField.count()) {
      if (!loginEmail) {
        emitStatus("waiting", { step: "manual-login", note: "未配置 CHATGPT_LOGIN_EMAIL；请用户在实时浏览器输入自己的 ChatGPT 账号，不会代填其他邮箱。" });
        return;
      }
      await emailField.fill(loginEmail, { timeout: 10000 });
      const continueButton = page.getByRole("button", { name: /^continue$/i }).first();
      if (await continueButton.count()) await continueButton.click({ timeout: 10000 });
      await wait(1500);
    }
    const currentText = await pageText(page);
    if (/Check your inbox|verification code|Enter the code/i.test(currentText) || /email-verification/i.test(page.url())) {
      emitStatus("waiting", { step: "email-verification", email: loginEmail || null, note: "已打开验证码页面；当前版本不会读取或记录验证码。" });
      return;
    }
    if (/captcha|verify you are human|security check|unusual activity/i.test(currentText)) {
      emitStatus("waiting", { step: "security-check", note: "遇到安全验证，停止自动操作，等待用户在网页中完成。" });
      return;
    }
    emitStatus("waiting", { step: "login-form", email: loginEmail || null, note: "登录页面已打开；若出现额外资料或安全验证，需要用户完成。" });
  } catch (error) {
    emitStatus("失败", { step: "login-flow", error: error.message });
  }
}

async function monitorLoginState() {
  if (!pageRef) return;
  try {
    if (await isLoggedIn(pageRef)) {
      if (loginFlowActive) emitStatus("完成", { step: "logged-in", url: pageRef.url() });
      loginFlowActive = false;
      return;
    }
    if (!loginFlowActive) await beginEmailLogin(pageRef);
    else {
      const title = await pageRef.title().catch(() => "");
      const text = await pageText(pageRef);
      if (/just a moment|captcha|verify you are human|security check|unusual activity/i.test(title + " " + text)) {
        emitStatus("waiting", { step: "security-check", note: "ChatGPT 正在显示安全检查页面；不会绕过检查，请用户在实时浏览器中正常完成。" });
      }
    }
  } catch (error) {
    emitStatus("失败", { step: "state-check", error: error.message });
  }
}

'''
s=s[:start]+replacement+s[end:]
p.write_text(s)
PY
node --check backend/extension/browser/chatgpt_live.js
grep -nE 'getLoginEmail|just a moment|manual-login' backend/extension/browser/chatgpt_live.js
echo 'LOGIN_PATCH_OK'
