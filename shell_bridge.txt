import WebSocket from 'ws';
async function verify(path, expected) {
  await new Promise((resolve, reject) => {
    const ws = new WebSocket('ws://127.0.0.1:3000' + path);
    const seen = new Set();
    const timer = setTimeout(() => { ws.terminate(); reject(new Error(path + ' timeout; received=' + [...seen].join(','))); }, 3500);
    ws.on('message', raw => {
      let m;
      try { m = JSON.parse(raw.toString()); } catch { return; }
      if (expected.includes(m.type)) seen.add(m.type);
      if (expected.every(type => seen.has(type))) {
        clearTimeout(timer);
        console.log(path + ' messages=' + [...seen].join(','));
        ws.terminate();
        resolve();
      }
      if (m.type === 'error') { clearTimeout(timer); ws.terminate(); reject(new Error(path + ' error=' + m.message)); }
    });
    ws.on('unexpected-response', (_req, res) => { clearTimeout(timer); reject(new Error(path + ' HTTP ' + res.statusCode)); });
    ws.on('error', e => { clearTimeout(timer); reject(e); });
  });
}
await verify('/backend-console/live', ['snapshot-start','snapshot-end','status']);
