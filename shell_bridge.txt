#!/bin/sh
set +e
cd /home/runner/workspace/ai-agent || exit 90
echo "=== process listening on 3000 ==="
(ps -ef | grep -E '[n]ode|[n]pm|[r]eplit' | head -n 60) 2>&1
if command -v ss >/dev/null 2>&1; then ss -ltnp 2>&1 | grep -E ':3000\b|State' || true; fi
if command -v lsof >/dev/null 2>&1; then lsof -nP -iTCP:3000 -sTCP:LISTEN 2>&1 || true; fi
echo "=== project and package scripts ==="
pwd
node -p "JSON.stringify(require('./package.json').scripts || {})" 2>&1
echo "=== route registration source ==="
sed -n '45,90p' backend/server.js
sed -n '1345,1405p' backend/server.js
echo "=== server startup / route mount markers ==="
grep -n -E 'app\.listen|server\.listen|listen\(|requireBrowserKey|/browser/cursor/status|/browser/recognition/inspect' backend/server.js | tail -n 50
echo "=== HTTP response details ==="
for route in /browser/cursor/status /browser/recognition/inspect /browser/session; do
  echo "--- $route ---"
  curl -sS -i --max-time 5 -X POST "http://127.0.0.1:3000$route" -H 'Content-Type: application/json' --data '{}' 2>&1 | head -n 15
done
echo "=== existing backend log candidates ==="
for f in /tmp/ai-agent-pet-api-restart.log /tmp/ai-agent.log /tmp/server.log; do
  if [ -f "$f" ]; then echo "--- $f ---"; tail -n 30 "$f"; fi
done
echo DIAGNOSE_PET_API_404_COMPLETE
