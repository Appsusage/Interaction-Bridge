        const registry = JSON.parse(await readFile(registryPath, "utf8"));
        const entry = (registry.processes || []).find(item =>

        );

        if (!entry || !entry.pid || !entry.pidStartTime) {
            return res.status(503).json({ status: "error", heartbeat: "not_registered" });
        }

        const [stat, cmdline] = await Promise.all([
            readFile("/proc/" + entry.pid + "/stat", "utf8"),
            readFile("/proc/" + entry.pid + "/cmdline", "utf8")
        ]);
        const close = stat.lastIndexOf(")");
        const startTime = close >= 0 ? stat.slice(close + 2).trim().split(/\\s+/)[19] : null;
        const args = cmdline.split("\\0").filter(Boolean);

        if (String(startTime) !== String(entry.pidStartTime) || !args.includes(heartbeatPath)) {
            return res.status(503).json({ status: "error", heartbeat: "not_running" });
        }

        return res.json({ status: "ok", heartbeat: "running", pid: Number(entry.pid) });
    } catch {
        return res.status(503).json({ status: "error", heartbeat: "unavailable" });
    }
});
