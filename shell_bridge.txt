import { WebSocketServer } from "ws";
import {
  browserActivity,
  getBrowserPage,
  listBrowserSessions,
  resolveBrowserDialog
} from "./browser_service.js";

export function attachBrowserLive(server) {
  const wss = new WebSocketServer({ server, path: "/browser/live" });
  const clients = new Set();
  const streams = new Map();
  const pendingSessions = new Set();

  function send(client, message) {
    if (client.readyState === 1) client.send(JSON.stringify(message));
  }
  function broadcast(message) {
    for (const client of clients) send(client, message);
  }
  function stateKind(state) {
    const value = String(state || "").toLowerCase();
    if (["进行中", "running", "in-progress", "started"].includes(value)) return "running";
    if (["完成", "completed", "失败", "error", "failed", "waiting", "等待用户确认"].includes(value)) return value;
    return value;
  }
  async function stopStream(client) {
    const state = client._browserLive;
    if (!state) return;
    client._browserLive = null;
    const stream = streams.get(state.sessionId);
    if (stream && stream.client === client) {
      streams.delete(state.sessionId);
      try { await stream.cdp.send("Page.stopScreencast"); } catch {}
      try { await stream.cdp.detach(); } catch {}
    }
  }
  async function attach(client, sessionId) {
    await stopStream(client);
    let page;
    try { page = getBrowserPage(sessionId); }
    catch (error) { send(client, { type: "error", message: error.message }); return; }
    const prior = streams.get(sessionId);
    if (prior) {
      try { await prior.cdp.send("Page.stopScreencast"); } catch {}
      try { await prior.cdp.detach(); } catch {}
      if (prior.client && prior.client !== client) {
        prior.client._browserLive = null;
        send(prior.client, { type: "status", state: "stream-replaced" });
      }
    }
    try {
      const cdp = await page.context().newCDPSession(page);
      client._browserLive = { sessionId, cdp };
      streams.set(sessionId, { client, cdp });
      cdp.on("Page.screencastFrame", async frame => {
        send(client, { type: "frame", sessionId, data: frame.data, metadata: frame.metadata });
        try { await cdp.send("Page.screencastFrameAck", { sessionId: frame.sessionId }); } catch {}
      });
      cdp.on("Page.frameNavigated", event => {
        if (event.frame?.url) send(client, { type: "navigation", sessionId, url: event.frame.url });
      });
      await cdp.send("Page.enable");
      await cdp.send("Page.startScreencast", {
        format: "jpeg", quality: 65, maxWidth: 1100, maxHeight: 800, everyNthFrame: 1
      });
      send(client, { type: "status", state: "streaming", sessionId, url: page.url(), title: await page.title() });
    } catch (error) {
      await stopStream(client);
      send(client, { type: "error", message: "无法启动 Chromium 实时画面：" + error.message });
    }
  }

  async function dispatchUserInput(client, message) {
    const sessionId = String(message.sessionId || "");
    if (!sessionId || client._browserLive?.sessionId !== sessionId) return;
    if (pendingSessions.has(sessionId)) {
      send(client, { type: "interaction-blocked", sessionId, reason: "AI 任务执行中，用户操作暂时暂停" });
      return;
    }
    const stream = streams.get(sessionId);
    if (!stream || stream.client !== client) return;
    const page = getBrowserPage(sessionId);
    const viewport = await page.evaluate(() => ({ width: window.innerWidth, height: window.innerHeight }));
    const nx = Math.max(0, Math.min(1, Number(message.x) || 0));
    const ny = Math.max(0, Math.min(1, Number(message.y) || 0));
    const x = Math.max(0, Math.min(viewport.width - 1, nx * viewport.width));
    const y = Math.max(0, Math.min(viewport.height - 1, ny * viewport.height));
    if (message.type === "user-click") {
      await stream.cdp.send("Input.dispatchMouseEvent", { type: "mouseMoved", x, y, button: "none" });
      await stream.cdp.send("Input.dispatchMouseEvent", { type: "mousePressed", x, y, button: "left", clickCount: 1 });
      await stream.cdp.send("Input.dispatchMouseEvent", { type: "mouseReleased", x, y, button: "left", clickCount: 1 });
    } else if (message.type === "user-wheel") {
      const deltaX = Math.max(-1600, Math.min(1600, Number(message.deltaX) || 0));
      const deltaY = Math.max(-2400, Math.min(2400, Number(message.deltaY) || 0));
      await stream.cdp.send("Input.dispatchMouseEvent", { type: "mouseWheel", x, y, deltaX, deltaY });
    }
    // Intentionally no keyboard or text-input message type is implemented.
  }

  browserActivity.on("activity", event => {
    const kind = stateKind(event.state);
    if (kind === "running") pendingSessions.add(String(event.sessionId));
    else if (["完成", "completed", "失败", "error", "failed", "waiting", "等待用户确认"].includes(kind)) {
      pendingSessions.delete(String(event.sessionId));
    }
    broadcast({ type: "activity", ...event });
  });
  browserActivity.on("session-created", event => broadcast({ type: "session-created", ...event }));
  browserActivity.on("session-closed", event => {
    pendingSessions.delete(String(event.sessionId));
    broadcast({ type: "session-closed", ...event });
  });
  browserActivity.on("dialog", event => broadcast({ type: "dialog", ...event }));
  browserActivity.on("dialog-result", event => broadcast({ type: "dialog-result", ...event }));

  wss.on("connection", (client, request) => {
    const origin = request.headers.origin;
    const host = request.headers.host;
    if (origin && host) {
      try {
        if (new URL(origin).host !== host) { client.close(1008, "Origin not allowed"); return; }
      } catch { client.close(1008, "Invalid origin"); return; }
    }
    clients.add(client);
    send(client, { type: "sessions", sessions: listBrowserSessions() });
    client.on("message", raw => {
      let message;
      try { message = JSON.parse(raw.toString()); } catch { return; }
      if (message.type === "attach" && typeof message.sessionId === "string") attach(client, message.sessionId);
      else if (message.type === "detach") stopStream(client);
      else if (message.type === "user-click" || message.type === "user-wheel") {
        dispatchUserInput(client, message).catch(error => send(client, { type: "error", message: "网页操作失败：" + error.message }));
      } else if (message.type === "dialog-action" && typeof message.sessionId === "string") {
        resolveBrowserDialog(message.sessionId, message.accept === true)
          .catch(error => send(client, { type: "error", message: "处理网页提示失败：" + error.message }));
      }
    });
    client.on("close", () => { clients.delete(client); stopStream(client); });
    client.on("error", () => { clients.delete(client); stopStream(client); });
  });
  return wss;
}
