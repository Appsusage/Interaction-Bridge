cd /home/runner/workspace/ai-agent || exit 1
set +e
printf '%s\n' '=== FORCED DROPBOX TOKEN REFRESH ==='
node --input-type=module <<'NODE'
import { getDropboxAccessToken, dropboxFetch } from './bridge/shell/dropbox_token.js';
try {
  const token = await getDropboxAccessToken({ forceRefresh: true });
  console.log('REFRESH_RESULT=SUCCESS');
  console.log('ACCESS_TOKEN_PRESENT=' + Boolean(token));
  const response = await dropboxFetch('https://api.dropboxapi.com/2/users/get_current_account', {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: '{}'
  });
  const text = await response.text();
  console.log('ACCOUNT_API_HTTP=' + response.status);
  if (!response.ok) {
    try {
      const data = JSON.parse(text);
      console.log('ACCOUNT_API_ERROR=' + (data.error_summary || data.error || 'unknown'));
    } catch {
      console.log('ACCOUNT_API_ERROR=non_json_response');
    }
  } else {
    console.log('ACCOUNT_API_RESULT=SUCCESS');
  }
} catch (error) {
  console.log('REFRESH_RESULT=FAILED');
  console.log('ERROR=' + String(error?.message || error).replace(/[\r\n]+/g, ' ').slice(0, 300));
  process.exitCode = 1;
}
NODE
REFRESH_RC=$?
printf '%s\n' '=== SYNTAX CHECKS ==='
for f in bridge/shell/dropbox_token.js bridge/shell/github_adaptor.js bridge/shell/server.js; do
  node --check "$f" || exit 1
  printf 'SYNTAX_OK=%s\n' "$f"
done
printf '%s\n' '=== TOKEN CACHE METADATA ONLY ==='
if [ -f bridge/shell/.dropbox_token_cache.json ]; then
  stat -c 'CACHE_EXISTS=true MODE=%a BYTES=%s' bridge/shell/.dropbox_token_cache.json
else
  printf '%s\n' 'CACHE_EXISTS=false'
fi
exit "$REFRESH_RC"
