python3 - <<'PY'
from pathlib import Path
p=Path("backend/extension/browser/browser_service.js")
s=p.read_text()
if 'import path from "node:path";' not in s:
    s=s.replace('import { EventEmitter } from "node:events";','import { EventEmitter } from "node:events";\nimport path from "node:path";')
old='''export async function createBrowserSession(options = {}) {
  const browser = await launchBrowser({ headless: options.headless ?? true });
  const context = await browser.newContext();
  const page = await context.newPage();
  const sessionId = registerBrowserSession({ browser, context, page, options });
  return { sessionId };
}'''
new='''export async function createBrowserSession(options = {}) {
  let browser = null;
  let context;
  let page;
  if (options.persistent === true) {
    const executablePath = process.env.PLAYWRIGHT_EXECUTABLE_PATH || "/repl/tools/bin/chromium";
    const profileDir = path.resolve(process.cwd(), ".gmail-profile");
    context = await chromium.launchPersistentContext(profileDir, {
      headless: options.headless ?? false,
      executablePath
    });
    page = context.pages()[0] || await context.newPage();
  } else {
    browser = await launchBrowser({ headless: options.headless ?? true });
    context = await browser.newContext();
    page = await context.newPage();
  }
  const sessionId = registerBrowserSession({ browser, context, page, options });
  return { sessionId, persistent: options.persistent === true };
}'''
if old not in s:
    if 'launchPersistentContext(profileDir' not in s:
        raise SystemExit("Expected function not found; stopped safely.")
else:
    backup=Path("backend/extension/browser/browser_service.js.bak-gmail-persistent-20261018")
    if not backup.exists(): backup.write_text(s)
    p.write_text(s.replace(old,new))
PY
node --check backend/extension/browser/browser_service.js

const page=await opened.json();
console.log(JSON.stringify({stage:'gmail',createStatus:created.status,openStatus:opened.status,sessionId:session.sessionId,persistent:session.persistent===true,url:page.url||null,title:page.title||null,error:page.error||null}));
