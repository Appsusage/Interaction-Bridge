cd /home/runner/workspace/ai-agent
node --input-type=module <<'EOF'
import { launchBrowser, openPage, readPage } from "./backend/extension/browser/browser_service.js";
const browser = await launchBrowser();
try {
  const page = await browser.newPage();
  const opened = await openPage(page, "https://example.com");
  const result = await readPage(page);
  console.log(JSON.stringify({ opened, title: result.title, hasBody: result.text.includes("Example Domain") }));
} finally {
  await browser.close();
}
EOF
