cd /home/runner/workspace/ai-agent
python3 - <<'PY'
from pathlib import Path
p = Path("backend/tools/tool_manifest.js")
s = p.read_text()
old = '\n\n}\n\nbrowser: {'
new = '\n\n},\n\nbrowser: {'
if old not in s:
    raise SystemExit("BROWSER_COMMA_ANCHOR_NOT_FOUND; no changes made")
p.write_text(s.replace(old, new, 1))
PY
node --check backend/tools/tool_manifest.js
node --input-type=module <<'EOF'
import { loadToolPrompts } from "./backend/tool_prompt_loader.js";
const prompts = loadToolPrompts();
for (const token of ["browser_new_session", "browser_open", "browser_read", "browser_click", "browser_type", "browser_scroll", "browser_screenshot", "browser_close_session"]) {
  if (!prompts.includes(token)) throw new Error("Missing browser prompt token: " + token);
}
console.log("BROWSER_PROMPT_LOADED", prompts.includes("browser_new_session"), "LENGTH", prompts.length);
EOF
