import { chromium } from "playwright";
import fs from "node:fs/promises";
import path from "node:path";
import { registerBrowserSession, browserActivity } from "./browser_service.js";

const CHATGPT_HOME = "https://chatgpt.com/";
const CHATGPT_LOGIN = "https://chatgpt.com/auth/login";
const LOGIN_EMAIL = process.env.CHATGPT_LOGIN_EMAIL || "frankii@robotomail.co";
const PROFILE_DIR = process.env.CHATGPT_PROFILE_DIR || path.resolve(process.cwd(), ".chatgpt-profile");
const EXECUTABLE_PATH = process.env.PLAYWRIGHT_EXECUTABLE_PATH || "/repl/tools/bin/chromium";

let started = false;
let loginFlowActive = false;
let monitor = null;
let startupPromise = null;
let pageRef = null;
let sessionRef = null;

const wait = ms => new Promise(resolve => setTimeout(resolve, ms));

function emitStatus(state, detail = {}) {
  if (!sessionRef) return;
  browserActivity.emit("activity", {
    sessionId: sessionRef,
    action: "ChatGPT 自动登录",
    state,
    detail,
    importance: state === "waiting" || state === "失败" ? 100 : 30,
    requiresUserAction: state === "waiting",
    displayPolicy: state === "waiting" ? "foreground" : "background",
    at: Date.now()
  });
}

async function pageText(page) {
  try { return await page.locator("body").innerText({ timeout: 5000 }); }
  catch { return ""; }
}

async function isLoggedIn(page) {
  const url = page.url();
  const text = await pageText(page);
  if (/auth\.openai\.com|\/auth\/login|\/email-verification/i.test(url)) return false;
  if (/Log in or sign up|Continue with Google|Enter your email|Check your inbox|verification code/i.test(text)) return false;
  return /New chat|What’s on the agenda|What can I help with|Upgrade|Free/i.test(text);
}

async function beginEmailLogin(page) {
  if (loginFlowActive) return;
  loginFlowActive = true;
  emitStatus("进行中", { step: "open-login" });
  try {
    if (!/auth\.openai\.com|\/auth\/login/i.test(page.url())) {
      await page.goto(CHATGPT_LOGIN, { waitUntil: "domcontentloaded", timeout: 30000 });
    }
    await wait(1200);

    const emailField = page.locator('input[type="email"]').first();
    if (await emailField.count()) {
      await emailField.fill(LOGIN_EMAIL, { timeout: 10000 });
      const continueButton = page.getByRole("button", { name: /^continue$/i }).first();
      if (await continueButton.count()) await continueButton.click({ timeout: 10000 });
      await wait(1500);
    }

    const text = await pageText(page);
    if (/Check your inbox|verification code|Enter the code/i.test(text) || /email-verification/i.test(page.url())) {
      emitStatus("waiting", { step: "email-verification", email: LOGIN_EMAIL, note: "已打开验证码页面；当前版本不会读取或记录验证码。" });
      return;
    }
    if (/captcha|verify you are human|security check|unusual activity/i.test(text)) {
      emitStatus("waiting", { step: "security-check", note: "遇到安全验证，停止自动操作，等待用户在网页中完成。" });
      return;
    }
    emitStatus("waiting", { step: "login-form", email: LOGIN_EMAIL, note: "登录页面已打开；若出现额外资料或安全验证，需要用户完成。" });
  } catch (error) {
    emitStatus("失败", { step: "login-flow", error: error.message });
  }
}

async function monitorLoginState() {
  if (!pageRef) return;
  try {
    if (await isLoggedIn(pageRef)) {
      if (loginFlowActive) emitStatus("完成", { step: "logged-in", url: pageRef.url() });
      loginFlowActive = false;
      return;
    }
    if (!loginFlowActive) await beginEmailLogin(pageRef);
    else {
      const text = await pageText(pageRef);
      if (/captcha|verify you are human|security check|unusual activity/i.test(text)) {
        emitStatus("waiting", { step: "security-check", note: "遇到安全验证，停止自动操作，等待用户在网页中完成。" });
      }
    }
  } catch (error) {
    emitStatus("失败", { step: "state-check", error: error.message });
  }
}

/**
 * Persistent ChatGPT login playbook:
 * 1. Reuse the same on-disk Chromium profile at .chatgpt-profile.
 * 2. Open chatgpt.com when the server starts.
 * 3. If signed out, open the official ChatGPT login page and submit CHATGPT_LOGIN_EMAIL.
 * 4. The profile keeps browser cookies between normal server restarts; no plaintext password is stored.
 * 5. Email verification codes and security checks are not bypassed. This version pauses on the code page;
 *    the user must complete the code/security step in the live browser before the monitor resumes.
 */
export async function startChatGPTLive() {
  if (started) return { started: true, sessionId: sessionRef };
  if (startupPromise) return startupPromise;

  startupPromise = (async () => {
    await fs.mkdir(PROFILE_DIR, { recursive: true });
    const context = await chromium.launchPersistentContext(PROFILE_DIR, {
      headless: process.env.CHATGPT_BROWSER_HEADLESS !== "false",
      executablePath: EXECUTABLE_PATH,
      viewport: { width: 1280, height: 900 },
      args: ["--no-sandbox"]
    });
    pageRef = context.pages()[0] || await context.newPage();
    sessionRef = registerBrowserSession({
      browser: null,
      context,
      page: pageRef,
      options: { displayPolicy: "background", importance: 30 }
    });
    started = true;

    try {
      await pageRef.goto(CHATGPT_HOME, { waitUntil: "domcontentloaded", timeout: 30000 });
      await wait(1800);
      if (await isLoggedIn(pageRef)) {
        emitStatus("完成", { step: "existing-session", url: pageRef.url() });
      } else {
        await beginEmailLogin(pageRef);
      }
    } catch (error) {
      emitStatus("失败", { step: "startup-navigation", error: error.message });
    }

    monitor = setInterval(() => { void monitorLoginState(); }, 15000);
    monitor.unref?.();
    return { started: true, sessionId: sessionRef, profileDir: PROFILE_DIR };
  })();

  try { return await startupPromise; }
  finally { startupPromise = null; }
}
