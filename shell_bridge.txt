const browserSessions = new Map();

export async function createBrowserSession() {
  const browser = await launchBrowser();
  const context = await browser.newContext();
  const page = await context.newPage();
  const sessionId = String(Date.now()) + "-" + Math.random().toString(36).slice(2);
  browserSessions.set(sessionId, { browser, context, page });
  return { sessionId };
}

export function getBrowserPage(sessionId) {
  const session = browserSessions.get(String(sessionId));
  if (!session) throw new Error("Browser session not found");
  return session.page;
}

export async function closeBrowserSession(sessionId) {
  const session = browserSessions.get(String(sessionId));
  if (!session) throw new Error("Browser session not found");
  browserSessions.delete(String(sessionId));
  await session.context.close();
  await session.browser.close();
  return { closed: true };
}
EOF
