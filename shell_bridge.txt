cd /home/runner/workspace/ai-agent && time node --input-type=module <<'__NODE_EOF__'
import fs from 'node:fs';
const entries = fs.readdirSync('/proc').filter(x => /^\d+$/.test(x));
let count=0;
for (const pid of entries) {
  try {
    fs.readFileSync('/proc/'+pid+'/cmdline','utf8');
    fs.readFileSync('/proc/'+pid+'/stat','utf8');
    count++;
  } catch {}
}
console.log('PROC_COUNT=' + entries.length);
console.log('READ_COUNT=' + count);
__NODE_EOF__
