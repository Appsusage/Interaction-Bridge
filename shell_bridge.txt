cat > test/prompt-master-local-routing.test.mjs <<'EOF'
import test from "node:test";
import assert from "node:assert/strict";
import { optimizePrompt } from "../backend/extension/prompt_master/cortex.js";

test("Cortex answers supported simple conversation locally without requesting a model", async () => {
  const result = await optimizePrompt({ message: "你好" });
  assert.equal(result.masterResult.needsModel, false);
  assert.equal(result.masterResult.route, "local");
  assert.equal(typeof result.masterResult.localResponse, "string");
  assert.ok(result.masterResult.localResponse.length > 0);
});

test("Cortex escalates unsupported complex analysis to the existing external route", async () => {
  const result = await optimizePrompt({ message: "请分析这段复杂代码的并发竞态和潜在安全问题" });
  assert.equal(result.masterResult.needsModel, true);
  assert.equal(result.masterResult.route, "external_ai");
  assert.equal(result.masterResult.localResponse, null);
});
EOF
node --test test/prompt-master-local-routing.test.mjs; printf '\nTEST_EXIT=%s\n' "$?"
