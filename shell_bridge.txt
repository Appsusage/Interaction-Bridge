set -eu
cd /home/runner/workspace/ai-agent
node --input-type=module - <<'NODE'
import fs from 'node:fs';
import path from 'node:path';
import { execFileSync } from 'node:child_process';

const root = process.cwd();
const target = path.join(root, 'backend/extension/browser/3D_viewer.js');
const htmlPath = path.join(root, 'frontend/index.html');
const tag = '<script src="/extension/browser/3D_viewer.js" defer></script>';
const needle = '<script src="/browser-live-viewer.js" defer></script>';
const token = process.env.GITHUB_TOKEN || process.env.GH_TOKEN;
if (!token) throw new Error('Missing GitHub token for private Genesis source');
const response = await fetch('https://api.github.com/repos/Appsusage/Genesis/contents/backend/extension/browser/3D_viewer.js?ref=main', {
  headers: { Authorization: 'Bearer ' + token, Accept: 'application/vnd.github+json' }
});
if (!response.ok) throw new Error('Genesis source HTTP ' + response.status);
const data = await response.json();
const source = Buffer.from(String(data.content || '').replace(/\s/g, ''), 'base64').toString('utf8');
if (!source.includes('const REDUCTIONS = Object.freeze([0, 10, 18, 25, 31, 36, 40, 43, 45, 46])')) {
  throw new Error('Unexpected 3D module source; nothing changed');
}
const html = fs.readFileSync(htmlPath, 'utf8');
if (!html.includes(tag) && !html.includes(needle)) throw new Error('Browser viewer script hook not found; nothing changed');
if (fs.existsSync(target) && fs.readFileSync(target, 'utf8') !== source) {
  throw new Error('3D_viewer.js already exists with different content; preserving existing file');
}
if (!fs.existsSync(target)) {
  const tmp = target + '.' + process.pid + '.js';
  try {
    fs.writeFileSync(tmp, source, { flag: 'wx' });
    execFileSync(process.execPath, ['--check', tmp], { stdio: 'pipe' });
    fs.renameSync(tmp, target);
  } finally {
    if (fs.existsSync(tmp)) fs.unlinkSync(tmp);
  }
}
if (!html.includes(tag)) fs.writeFileSync(htmlPath, html.replace(needle, needle + '\n  ' + tag));
console.log('3D_VIEWER_INSTALLED', JSON.stringify({
  file: 'backend/extension/browser/3D_viewer.js',
  bytes: Buffer.byteLength(source),
  htmlHook: fs.readFileSync(htmlPath, 'utf8').includes(tag),
  levels: [10,18,25,31,36,40,43,45,46]
}));
NODE
