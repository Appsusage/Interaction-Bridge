#!/bin/bash
set +e
cd /home/runner/workspace/ai-agent || exit 1
echo "TASK=real_local_pet_mouse_linkage_test"
node --input-type=module <<'NODE'
import { createBrowserSession, getBrowserPage, closeBrowserSession } from './backend/extension/browser/browser_service.js';
import { acquirePetCursor, movePetCursor, releasePetCursor } from './backend/extension/pet/cursor/index.js';
const out = { test: 'local-ai-agent-only', steps: [] };
let sessionId;
try {
  const s = await createBrowserSession({headless:true});
  sessionId = s.sessionId;
  const page = getBrowserPage(sessionId);
  await page.goto('http://127.0.0.1:3000/', {waitUntil:'domcontentloaded', timeout:15000});
  await page.waitForTimeout(1500);
  out.page = {url:page.url(), title:await page.title()};
  const pre = await page.evaluate(() => ({
    api: !!window.AICortexPetLimbo,
    float: !!document.querySelector('#mainFloat'),
    petCount: document.querySelectorAll('.limbo-pet').length,
    viewport: {w:innerWidth,h:innerHeight},
    ball: document.querySelector('#mainFloat')?.getBoundingClientRect().toJSON()
  }));
  out.preflight=pre;
  if (!pre.api || !pre.ball) throw new Error('Pet API or floating ball absent; no gesture test run');
  if (!pre.petCount) {
    const b=pre.ball, x=b.left+b.width/2, y=b.top+b.height/2;
    await page.mouse.move(x,y); await page.mouse.down();
    await page.mouse.move(x,y+90,{steps:8}); await page.mouse.up();
    await page.waitForTimeout(1300);
  }
  out.spawn = await page.evaluate(() => ({
    count:document.querySelectorAll('.limbo-pet').length,
    rect:document.querySelector('.limbo-pet')?.getBoundingClientRect().toJSON()
  }));
  if (!out.spawn.count) throw new Error('UI downward-drag did not spawn pet');
  const acq=await acquirePetCursor(sessionId);
  out.acquire=acq;
  if (!acq || acq.unavailable || acq.active===false) throw new Error('Cursor acquire failed: '+JSON.stringify(acq));
  const v=pre.viewport;
  const targets=[
    {dir:'right',x:Math.round(v.w*0.72),y:Math.round(v.h*0.5)},
    {dir:'down',x:Math.round(v.w*0.5),y:Math.round(v.h*0.72)},
    {dir:'left',x:Math.round(v.w*0.28),y:Math.round(v.h*0.5)},
    {dir:'up',x:Math.round(v.w*0.5),y:Math.round(v.h*0.28)}
  ];
  out.moves=[];
  for(const t of targets){
    const m=await movePetCursor(sessionId,t.x,t.y);
    const measured=await page.evaluate(()=>{const p=document.querySelector('.limbo-pet');if(!p)return null;const r=p.getBoundingClientRect();return {x:r.left+r.width/2,y:r.top+r.height/2,gaze:p.dataset.gaze};});
    out.moves.push({target:t,reported:m,measured});
    await page.waitForTimeout(150);
  }
  out.pass=out.moves.length===4 && out.moves.every(m=>m.measured && Math.abs(m.measured.x-m.target.x)<3 && Math.abs(m.measured.y-m.target.y)<3);
} catch(e) { out.error=String(e?.stack||e); }
finally {
 try { if(sessionId) { await releasePetCursor(sessionId); out.release='attempted'; } } catch(e){out.releaseError=String(e);}
 try { if(sessionId) { await closeBrowserSession(sessionId); out.closed=true; } } catch(e){out.closeError=String(e);}
}
console.log(JSON.stringify(out));
NODE
