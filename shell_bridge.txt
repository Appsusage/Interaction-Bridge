node --input-type=module -e 'import {ShellCommand} from "./bridge/shell/multi-task/shell_command.js";
const w=new ShellCommand({pollMs:1000,maxConcurrent:1,logger:console});
console.log("NEW_WORKER_RUN_START");
console.log("GITHUB_CREDENTIAL_PRESENT",Boolean(process.env.GITHUB_TOKEN||process.env.GH_TOKEN));
try{
 await w.collect();
 console.log("COLLECT_OK","queued="+w.queued.size,"running="+w.running.size);
 console.log("QUEUE_ITEMS",[...w.queued.values()].map(x=>({name:x.name,folder:x.folder,source:x.source?.kind})));
 if(w.queued.size>0){
  w.schedule();const started=Date.now();
  while((w.queued.size||w.running.size)&&Date.now()-started<18000){await new Promise(r=>setTimeout(r,1000)); console.log("WORKER_PROGRESS","queued="+w.queued.size,"running="+w.running.size);}
  console.log("WORKER_AFTER_WAIT","queued="+w.queued.size,"running="+w.running.size);
 }
}catch(e){console.log("NEW_WORKER_RUN_ERROR",e.stack?.slice(0,600)||e.message);process.exitCode=1}
console.log("NEW_WORKER_RUN_END")'
