cd /home/runner/workspace/ai-agent
set +e
SERVER_PID="$(pgrep -f 'node backend/server.js' | head -1)"
if [ -z "$SERVER_PID" ]; then
  SERVER_PID="$(pgrep -f 'backend/server.js' | head -1)"
fi
if [ -z "$SERVER_PID" ] || [ ! -r "/proc/$SERVER_PID/environ" ]; then
  echo 'BLOCKED: cannot inspect running server process environment; no session created.'
  exit 0
fi
SERVER_HAS_KEY="$(tr '\0' '\n' < "/proc/$SERVER_PID/environ" | grep -c '^BROWSER_API_KEY=')"
echo "SERVER_PROCESS_FOUND=yes"
echo "SERVER_HAS_BROWSER_API_KEY=$([ "$SERVER_HAS_KEY" -gt 0 ] && echo yes || echo no)"
if [ "$SERVER_HAS_KEY" -eq 0 ]; then
  echo 'BLOCKED: running server process has no BROWSER_API_KEY; live stream test cannot authenticate.'
  exit 0
fi
export BROWSER_API_KEY="$(tr '\0' '\n' < "/proc/$SERVER_PID/environ" | sed -n 's/^BROWSER_API_KEY=//p' | head -1)"
node --input-type=module <<'JS'
import WebSocket from "ws";
const base="http://127.0.0.1:3000", key=process.env.BROWSER_API_KEY;
const headers={"content-type":"application/json","x-browser-key":key};
const post=async(route,body={})=>{const r=await fetch(base+route,{method:"POST",headers,body:JSON.stringify(body)});const t=await r.text();if(!r.ok)throw Error(route+" HTTP "+r.status+" "+t);return JSON.parse(t)};
let id="",frames=0,created=false;
const ws=new WebSocket("ws://127.0.0.1:3000/browser/live");
const result=await new Promise(resolve=>{
 const timer=setTimeout(()=>resolve("TIMEOUT"),12000);
 ws.on("open",async()=>{try{id=(await post("/browser/session")).sessionId;await post("/browser/open",{sessionId:id,url:"https://example.com"});}catch(e){clearTimeout(timer);resolve("ERROR "+e.message)}});
 ws.on("message",async raw=>{let m;try{m=JSON.parse(raw.toString())}catch{return}
  if(m.type==="session-created"&&m.sessionId===id){created=true;ws.send(JSON.stringify({type:"attach",sessionId:id}))}
  if(m.type==="frame"&&m.sessionId===id&&++frames>=3){clearTimeout(timer);resolve("PASS")}
  if(m.type==="error"){clearTimeout(timer);resolve("WS_ERROR "+m.message)}
 });
 ws.on("error",e=>{clearTimeout(timer);resolve("WS_ERROR "+e.message)});
});
console.log("SESSION_CREATED="+Boolean(id));
console.log("SESSION_EVENT_RECEIVED="+created);
console.log("FRAME_COUNT="+frames);
console.log("FRAME_TEST="+result);
if(id){try{console.log("SESSION_CLOSE="+JSON.stringify(await post("/browser/close",{sessionId:id})))}catch(e){console.log("SESSION_CLOSE_ERROR="+e.message)}}
try{ws.close()}catch{}
JS
echo '=== PUBLIC CHECKS ==='
curl -sS -o /dev/null -w 'health_http=%{http_code}\n' http://127.0.0.1:3000/health
curl -sS -o /dev/null -w 'viewer_js_http=%{http_code}\n' http://127.0.0.1:3000/browser-live-viewer.js
curl -sS -o /dev/null -w 'viewer_css_http=%{http_code}\n' http://127.0.0.1:3000/browser-live-viewer.css
