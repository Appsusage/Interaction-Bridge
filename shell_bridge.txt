#!/bin/bash
set +e
cd /home/runner/workspace/ai-agent || exit 1

mkdir -p backend/extension/pet/cursor
cat > backend/extension/pet/cursor/four_direction_follow_test_S1258.txt <<'EOF'
# 宠宝—鼠标四向联动测试详细记录（S1258）

## 测试目标
验证本地 ai-agent 浏览器页面中，宠宝在 cursor 控制模块接管后，是否能跟随鼠标目标完成右、下、左、上四向移动，并核验 CSS 动画过渡结束后的实际屏幕坐标。

## 测试环境与方法
- 项目目录：/home/runner/workspace/ai-agent
- 本地页面：http://127.0.0.1:3000/
- 页面标题：私人AI助手
- 浏览器：browser_service 创建的 Chromium 会话；本次为 headless（无可见 GUI 窗口）
- viewport：1280 × 800，devicePixelRatio=1
- 宠宝创建方式：在页面中直接调用 window.AICortexPetLimbo.spawnFrom(document.body, 640, 400)
- 控制模块：backend/extension/pet/cursor/index.js
- 调用：acquireControl(sessionId) → 对四个目标依次 moveTo(sessionId,x,y) → releaseControl(sessionId)
- 每次移动后等待 750ms；宠宝 CSS 的 left/top transition 为 650ms，先前立即读取坐标会因过渡尚未结束而误判。

## 四向结果
| 方向 | 鼠标目标中心 (x,y) | 实测宠宝中心 (x,y) | 误差 | 位置匹配 |
|---|---:|---:|---:|---|
| 右 | (900,400) | (900.35,398.29) | 约 1.75px | PASS |
| 下 | (640,600) | (639.94,599.93) | 约 0.09px | PASS |
| 左 | (350,400) | (349.91,397.93) | 约 2.07px | PASS |
| 上 | (640,200) | (640.54,198.77) | 约 1.34px | PASS |

## 其他观察
- acquireControl 返回 acquired=true，petControl.active=true、petPresent=true。
- 四次 moveTo 均返回 moved=true；实测 DOM 中宠宝中心与目标点误差均小于 4px。
- gaze 属性有方向响应（本测试观察到 right/left）；此轮重点判据是位置跟随，未将 gaze 的上下方向映射作为通过条件。
- releaseControl 返回 released=true；浏览器会话已关闭。
- 测试总判定：四个方向的位置匹配均通过（pass=true）。

## 重要边界 / 尚未验证
1. 这是通过页面 JS API 创建宠宝、通过 cursor 模块调用移动的自动化集成测试；不是用户在可见窗口里真实拖动前端浮球触发宠宝出生的测试。
2. 浏览器是 headless；本次没有验证可见窗口中的手势拖动、拖动阈值、pointer/mouse 事件链或实际视觉反馈。
3. 后续需要在 headed/可见浏览器窗口里从真实前端浮球开始，执行按住—拖动—释放，并记录宠宝是否生成、是否跟随以及截图/坐标证据。
EOF

echo "=== report saved ==="
ls -l backend/extension/pet/cursor/four_direction_follow_test_S1258.txt
echo "=== browser service implementation ==="
sed -n '1,240p' backend/extension/browser/browser_service.js
echo "=== available display/browser binaries ==="
printf 'DISPLAY=%s\n' "$DISPLAY"
command -v chromium || command -v chromium-browser || command -v google-chrome || true
command -v xvfb-run || true
echo "=== headed session launch probe (local page only) ==="
node --input-type=module <<'NODE'
import {createBrowserSession,getBrowserPage,closeBrowserSession} from './backend/extension/browser/browser_service.js';
let id;
try {
 const s=await createBrowserSession({headless:false}); id=s.sessionId;
 const p=getBrowserPage(id);
 await p.setViewportSize({width:900,height:700});
 await p.goto('http://127.0.0.1:3000/',{waitUntil:'domcontentloaded',timeout:12000});
 await p.waitForTimeout(500);
 console.log(JSON.stringify({headedSessionCreated:true,sessionId:id,title:await p.title(),url:p.url(),viewport:await p.evaluate(()=>({w:innerWidth,h:innerHeight})),headlessOption:false}));
} catch(e) { console.log(JSON.stringify({headedSessionCreated:false,error:String(e?.stack||e)})); }
finally { try { if(id) await closeBrowserSession(id); } catch(e) { console.log('cleanupError='+String(e)); } }
NODE
