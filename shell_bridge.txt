import fs from 'node:fs';
import path from 'node:path';
import { fileURLToPath } from 'node:url';

const dir = path.dirname(fileURLToPath(import.meta.url));
const storePath = path.join(dir, 'pid.json');
const now = () => new Date().toISOString();

function load() {
  if (!fs.existsSync(storePath)) return { version: 1, processes: [] };
  const data = JSON.parse(fs.readFileSync(storePath, 'utf8'));
  if (!Array.isArray(data.processes)) throw new Error('invalid PID store');
  return data;
}

function normalize(p) {
  if (!p?.name) throw new Error('process name is required');
  return {
    name: String(p.name),
    pid: Number.isInteger(Number(p.pid)) && Number(p.pid) > 0 ? Number(p.pid) : null,
    pidStartTime: p.pidStartTime ?? null,
    command: String(p.command ?? ''),
    cwd: String(p.cwd ?? dir),
    args: Array.isArray(p.args) ? p.args.map(String) : [],
    required: Boolean(p.required),
    protected: Boolean(p.protected),
    status: String(p.status ?? 'registered'),
    registeredAt: p.registeredAt ?? now(),
    lastStartTime: p.lastStartTime ?? null,
    lastCheckTime: p.lastCheckTime ?? null,
    lastFailureTime: p.lastFailureTime ?? null,
    lastFailureReason: p.lastFailureReason ?? null,
    restartCount: Number(p.restartCount ?? 0),
    consecutiveFailures: Number(p.consecutiveFailures ?? 0)
  };
}

function save(data) {
  const tmp = `${storePath}.tmp-${process.pid}`;
  fs.writeFileSync(tmp, JSON.stringify({ version: 1, processes: data.processes.map(normalize) }, null, 2) + '\n', { mode: 0o600 });
  fs.renameSync(tmp, storePath);
}

function cleanLegacyIds(data) {
  let changed = false;
  data.processes = data.processes.map(p => {
    if (Object.prototype.hasOwnProperty.call(p, 'id')) {
      const { id, ...rest } = p;
      changed = true;
      return rest;
    }
    return p;
  });
  if (changed) save(data);
  return data;
}

export const getStorePath = () => storePath;
export const list = () => cleanLegacyIds(load()).processes.map(normalize);
export const getByName = name => list().find(p => p.name === String(name)) ?? null;
export const getByPid = pid => list().find(p => p.pid === Number(pid)) ?? null;
export const findExact = ({ name, pid } = {}) => name != null ? getByName(name) : pid != null ? getByPid(pid) : null;

export function register(p) {
  const data = cleanLegacyIds(load());
  const item = normalize(p);
  if (data.processes.some(x => x.name === item.name)) throw new Error(`managed process already exists: ${item.name}`);
  data.processes.push(item);
  save(data);
  return item;
}

export function upsert(p) {
  const data = cleanLegacyIds(load());
  const item = normalize(p);
  const i = data.processes.findIndex(x => x.name === item.name);
  if (i < 0) data.processes.push(item);
  else data.processes[i] = normalize({ ...data.processes[i], ...item });
  save(data);
  return getByName(item.name);
}

export function update(name, patch) {
  const data = cleanLegacyIds(load());
  const i = data.processes.findIndex(x => x.name === String(name));
  if (i < 0) throw new Error(`managed process not found: ${name}`);
  data.processes[i] = normalize({ ...data.processes[i], ...patch });
  save(data);
  return data.processes[i];
}

export function remove(name) {
  const data = cleanLegacyIds(load());
  const before = data.processes.length;
  data.processes = data.processes.filter(x => x.name !== String(name));
  save(data);
  return before !== data.processes.length;
}

export const clear = () => save({ version: 1, processes: [] });

import fs from 'node:fs';
import path from 'node:path';
import { spawn } from 'node:child_process';
import { fileURLToPath } from 'node:url';
import * as pidStore from './pid.js';

const ROOT = path.resolve(path.dirname(fileURLToPath(import.meta.url)), '..');
const CHECK_MS = 3000;
const SCAN_MS = 5000;
const CONFIRM_MS = 500;
const GITHUB_OWNER = 'Appsusage';
const GITHUB_REPO = 'Interaction-Bridge';
const GITHUB_BRANCH = 'main';
const recoveryInProgress = new Set();
let stopping = false;

const REQUIRED = [
  { name: 'ai_server', command: 'node', args: ['backend/server.js'], cwd: ROOT, required: true, protected: false },
  { name: 'shell_bridge', command: 'node', args: ['bridge/shell/server.js'], cwd: ROOT, required: true, protected: false },
  { name: 'github_adaptor', command: 'node', args: ['bridge/shell/github_adaptor.js'], cwd: ROOT, required: true, protected: false },
  { name: 'heartbeat', command: 'node', args: ['backend/heartbeat.js'], cwd: ROOT, required: true, protected: true }
];

const sleep = ms => new Promise(resolve => setTimeout(resolve, ms));
const sameArgs = (a, b) => a.length === b.length && a.every((v, i) => v === b[i]);

function procStartTime(pid) {
  try {
    const raw = fs.readFileSync(`/proc/${pid}/stat`, 'utf8');
    const close = raw.lastIndexOf(')');
    return raw.slice(close + 2).trim().split(/\s+/)[19] ?? null;
  } catch { return null; }
}

function readProcess(pid) {
  try {
    const args = fs.readFileSync(`/proc/${pid}/cmdline`, 'utf8').split('\0').filter(Boolean);
    const cwd = fs.realpathSync(`/proc/${pid}/cwd`);
    return { pid: Number(pid), args, cwd, pidStartTime: procStartTime(pid) };
  } catch { return null; }
}

function alive(pid, expectedStartTime = null) {
  const n = Number(pid);
  if (!Number.isInteger(n) || n <= 0 || !fs.existsSync(`/proc/${n}`)) return false;
  if (expectedStartTime != null && String(procStartTime(n)) !== String(expectedStartTime)) return false;
  return true;
}

function findRunning(spec) {
  let entries;
  try { entries = fs.readdirSync('/proc').filter(x => /^\d+$/.test(x)); } catch { return null; }
  for (const pid of entries) {
    const p = readProcess(pid);
    if (!p || p.pid === process.pid || p.cwd !== spec.cwd) continue;
    if (p.args[0] !== spec.command || !sameArgs(p.args.slice(1), spec.args)) continue;
    return p;
  }
  return null;
}

function specForProcess(p) {
  if (!p || p.cwd !== ROOT) return null;
  return REQUIRED.find(spec => p.args[0] === spec.command && sameArgs(p.args.slice(1), spec.args)) ?? null;
}

function ensureRecord(spec, running) {
  const existing = pidStore.getByName(spec.name);
  const patch = {
    name: spec.name,
    command: spec.command,
    args: spec.args,
    cwd: spec.cwd,
    required: spec.required,
    protected: spec.protected
  };
  if (running) {
    patch.pid = running.pid;
    patch.pidStartTime = running.pidStartTime;
    patch.status = 'running';
    patch.lastCheckTime = new Date().toISOString();
    if (!existing || existing.pid !== running.pid) patch.lastStartTime = new Date().toISOString();
  } else if (!existing) {
    patch.pid = null;
    patch.pidStartTime = null;
    patch.status = 'missing';
  }
  return pidStore.upsert(existing ? { ...existing, ...patch } : patch);
}

async function scanAndRegister() {
  for (const spec of REQUIRED) ensureRecord(spec, findRunning(spec));

  let entries;
  try { entries = fs.readdirSync('/proc').filter(x => /^\d+$/.test(x)); } catch { return; }
  for (const pid of entries) {
    const p = readProcess(pid);
    const spec = specForProcess(p);
    if (spec) ensureRecord(spec, p);
  }
}

async function confirmedDead(record) {
  if (record.pid == null) return true;
  if (alive(record.pid, record.pidStartTime)) return false;
  await sleep(CONFIRM_MS);
  return !alive(record.pid, record.pidStartTime);
}

async function launch(record) {
  const logPath = path.join('/tmp', `heartbeat-restart-${record.name}-${Date.now()}.raw_log.txt`);
  const stream = fs.createWriteStream(logPath, { flags: 'w', mode: 0o600 });
  const child = spawn(record.command, record.args ?? [], {
    cwd: record.cwd,
    detached: true,
    stdio: ['ignore', stream, stream],
    env: process.env
  });
  const error = await new Promise(resolve => {
    let done = false;
    child.once('error', err => { if (!done) { done = true; resolve(err); } });
    child.once('spawn', () => { if (!done) { done = true; resolve(null); } });
  });
  if (error) {
    stream.end();
    return { ok: false, error: error.message, rawLog: fs.existsSync(logPath) ? fs.readFileSync(logPath, 'utf8') : '' };
  }
  child.unref();
  await sleep(1200);
  stream.end();
  if (!alive(child.pid)) {
    return { ok: false, error: `process exited immediately; pid=${child.pid}`, rawLog: fs.existsSync(logPath) ? fs.readFileSync(logPath, 'utf8') : '' };
  }
  return { ok: true, pid: child.pid, pidStartTime: procStartTime(child.pid), rawLog: fs.existsSync(logPath) ? fs.readFileSync(logPath, 'utf8') : '' };
}

async function uploadFeedback(record, reason, rawLog) {
  const token = process.env.GITHUB_TOKEN || process.env.GH_TOKEN;
  const stamp = new Date().toISOString().replace(/[:.]/g, '-');
  const pathName = `feedback/heartbeat-${record.name}-${stamp}.txt`;
  const content = [
    '=== HeartBeat restart failure ===',
    `time: ${new Date().toISOString()}`,
    `process: ${record.name}`,
    `pid: ${record.pid ?? ''}`,
    `command: ${record.command} ${(record.args ?? []).join(' ')}`,
    `cwd: ${record.cwd}`,
    `reason: ${reason}`,
    '',
    '=== raw_log.txt ===',
    rawLog || '(empty)',
    '=== END ===',
    ''
  ].join('\n');
  if (!token) throw new Error('GITHUB_TOKEN/GH_TOKEN is missing; cannot send HeartBeat feedback');
  const response = await fetch(`https://api.github.com/repos/${GITHUB_OWNER}/${GITHUB_REPO}/contents/${pathName}`, {
    method: 'PUT',
    headers: {
      Authorization: `Bearer ${token}`,
      Accept: 'application/vnd.github+json',
      'X-GitHub-Api-Version': '2022-11-28',
      'Content-Type': 'application/json'
    },
    body: JSON.stringify({
      message: `HeartBeat feedback: ${record.name} restart failed`,
      content: Buffer.from(content, 'utf8').toString('base64'),
      branch: GITHUB_BRANCH
    })
  });
  const text = await response.text();
  if (!response.ok) throw new Error(`GitHub feedback upload ${response.status}: ${text}`);
  return pathName;
}

async function recover(record, reason) {
  if (recoveryInProgress.has(record.name)) return;
  recoveryInProgress.add(record.name);
  try {
    const current = pidStore.getByName(record.name);
    if (!current) return;
    const started = await launch(current);
    if (started.ok) {
      pidStore.update(current.name, {
        pid: started.pid,
        pidStartTime: started.pidStartTime,
        status: 'running',
        lastStartTime: new Date().toISOString(),
        lastCheckTime: new Date().toISOString(),
        restartCount: Number(current.restartCount ?? 0) + 1,
        consecutiveFailures: 0,
        lastFailureTime: null,
        lastFailureReason: null
      });
      return;
    }

    pidStore.remove(current.name);
    try {
      await uploadFeedback(current, `${reason}; ${started.error}`, started.rawLog);
    } catch (feedbackError) {
      console.error(`[HeartBeat] restart failed for ${current.name}; feedback upload failed: ${feedbackError.message}`);
    }
  } catch (error) {
    try { pidStore.remove(record.name); } catch {}
    try { await uploadFeedback(record, `${reason}; ${error.message}`, ''); } catch (feedbackError) {
      console.error(`[HeartBeat] restart failed for ${record.name}; feedback upload failed: ${feedbackError.message}`);
    }
  } finally {
    recoveryInProgress.delete(record.name);
  }
}

async function check(record) {
  const current = pidStore.getByName(record.name);
  if (!current) return;
  if (current.pid == null) {
    if (current.required) recover(current, 'required process is missing');
    return;
  }
  if (alive(current.pid, current.pidStartTime)) {
    if (current.status !== 'running') pidStore.update(current.name, { status: 'running' });
    return;
  }
  if (!(await confirmedDead(current))) return;
  pidStore.update(current.name, {
    status: 'dead',
    lastFailureTime: new Date().toISOString(),
    lastFailureReason: 'process not alive or PID identity changed'
  });
  recover(current, 'confirmed_process_failure');
}

async function main() {
  await scanAndRegister();
  setInterval(() => { scanAndRegister().catch(error => console.error(`[HeartBeat] scan failed: ${error.message}`)); }, SCAN_MS);
  setInterval(() => { Promise.allSettled(pidStore.list().map(check)); }, CHECK_MS);
  while (!stopping) await sleep(60000);
}

process.on('SIGTERM', () => { stopping = true; });
process.on('SIGINT', () => { stopping = true; });
main().catch(error => { console.error(`[HeartBeat] fatal: ${error.stack || error.message}`); process.exitCode = 1; });
