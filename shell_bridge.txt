#!/bin/sh
set +e
cd /home/runner/workspace/ai-agent || exit 90
echo "=== restart backend to load updated routes ==="
old_pids=$(pgrep -f '^node backend/server.js$' || true)
if [ -n "$old_pids" ]; then
  echo "existing_pids=$old_pids"
  kill -TERM $old_pids 2>/dev/null
  sleep 2
fi
new_pids=$(pgrep -f '^node backend/server.js$' || true)
if [ -z "$new_pids" ]; then
  echo "No managed backend process remains; starting backend in background."
  nohup node backend/server.js >/tmp/ai-agent-pet-api-restart.log 2>&1 &
  sleep 3
else
  echo "backend process already restarted/replaced: $new_pids"
fi
echo "=== health and route auth probes ==="
pgrep -af '^node backend/server.js$' || true
curl -sS -o /tmp/pet_cursor_probe.txt -w 'GET / status_http=%{http_code}\n' --max-time 5 http://127.0.0.1:3000/ 2>&1
for route in /browser/cursor/status /browser/recognition/inspect; do
  code=$(curl -sS -o /tmp/pet_cursor_probe.txt -w '%{http_code}' --max-time 5 -X POST "http://127.0.0.1:3000$route" -H 'Content-Type: application/json' --data '{}' 2>&1)
  echo "$route http=$code"
done
echo "=== restart log tail (if available) ==="
tail -n 20 /tmp/ai-agent-pet-api-restart.log 2>/dev/null || true
echo PET_API_RESTART_PROBE_COMPLETE
