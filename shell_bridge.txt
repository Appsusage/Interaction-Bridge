node --input-type=module <<'RELOAD_V2_DEDUP'
import fs from 'node:fs';
const d='./bridge/shell/multi-task/';
const old=Number(fs.readFileSync(d+'.multi-task-server.pid','utf8'));
const supervisor=Number(fs.readFileSync(d+'.multi-task-supervisor.pid','utf8'));
if(!fs.readFileSync('/proc/'+old+'/cmdline','utf8').includes('multi-task/server.js'))throw Error('Server PID mismatch');
if(!fs.readFileSync('/proc/'+supervisor+'/cmdline','utf8').includes('multi-task/supervisor.js'))throw Error('Supervisor PID mismatch');
let oldStatus=await (await fetch('http://127.0.0.1:3102/health')).json();
if(oldStatus.active||oldStatus.queued)throw Error('Current V2 has active tasks; no restart');
process.kill(old,'SIGTERM');
let next=null;
for(let i=0;i<70;i++){
 await new Promise(r=>setTimeout(r,500));
 try{
  const r=await fetch('http://127.0.0.1:3102/health');
  const data=await r.json();
  if(data.pid!==old&&data.running){next=data;break;}
 }catch{}
}
if(!next)throw Error('V2 restart not confirmed');
console.log('V2_DEDUP_PATCH_LOADED',JSON.stringify({oldPid:old,newPid:next.pid,supervisor}));
RELOAD_V2_DEDUP
