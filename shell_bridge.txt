cd /home/runner/workspace/ai-agent || exit 1
printf '%s\n' '=== UTC ==='
date -u '+%Y-%m-%dT%H:%M:%SZ'
if [ -z "${HEARTBEAT_SUPERVISOR_TOKEN:-}" ]; then
  echo 'START_SKIPPED: HEARTBEAT_SUPERVISOR_TOKEN missing from shell environment'
  exit 2
fi
printf '%s\n' '=== request authenticated local HeartBeat start (token value hidden) ==='
curl -sS --max-time 20 -X POST \
  -H "Authorization: Bearer ${HEARTBEAT_SUPERVISOR_TOKEN}" \
  -H 'Content-Type: application/json' \
  -w '\nHTTP_STATUS:%{http_code}\n' \
  http://127.0.0.1:3000/heartbeat/start
sleep 2
printf '%s\n' '=== HeartBeat health after start ==='
curl -sS --max-time 5 -w '\nHTTP_STATUS:%{http_code}\n' http://127.0.0.1:3000/heartbeat/health
printf '%s\n' '=== matching HeartBeat process ==='
ps -eo pid,ppid,stat,args | awk '/node/ && /backend\/heartbeat\/heartbeat\.js/ && !/awk/ {print}'
