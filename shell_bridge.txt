set +e
BASE="http://127.0.0.1:3000"
KEY="${BROWSER_API_KEY:-}"
if [ -z "$KEY" ]; then echo "ERROR: BROWSER_API_KEY missing"; exit 2; fi
SID="1792241456256-v95pmtm84w"
echo "=== OPEN CHATGPT HOMEPAGE ==="
curl --silent --show-error --max-time 35 -i -X POST "$BASE/browser/open" -H "Content-Type: application/json" -H "x-browser-key: $KEY" --data "{\"sessionId\":\"$SID\",\"url\":\"https://chatgpt.com/\"}"
echo
sleep 3
echo "=== READ CHATGPT PAGE ==="
curl --silent --show-error --max-time 25 -i -X POST "$BASE/browser/read" -H "Content-Type: application/json" -H "x-browser-key: $KEY" --data "{\"sessionId\":\"$SID\"}"
echo
