cd /home/runner/workspace/ai-agent
set -eu
PID="$(ps -eo pid,args | awk '$3 == "backend/server.js" && $2 ~ /node$/ {print $1; exit}')"
if [ -z "$PID" ]; then
  echo 'FAIL: existing ai-agent server process not found; refusing to start a duplicate'
  exit 2
fi
echo "Restarting only node backend/server.js (PID $PID)"
kill -TERM "$PID"
for i in $(seq 1 20); do
  if ! kill -0 "$PID" 2>/dev/null; then break; fi
  sleep 0.25
done
npm start > /tmp/ai-agent-server-restart.log 2>&1 &
NEW_PID=$!
echo "new_launcher_pid=$NEW_PID"
for i in $(seq 1 30); do
  CODE="$(curl --max-time 2 -s -o /tmp/pet-limbo-http.txt -w '%{http_code}' http://127.0.0.1:3000/extension/pet/limbo/pet.js || true)"
  if [ "$CODE" = "200" ]; then
    echo "PASS: Pet Limbo extension asset served over HTTP (200)"
    break
  fi
  sleep 0.5
done
if [ "${CODE:-}" != "200" ]; then
  echo "FAIL: Pet Limbo asset HTTP status=${CODE:-none}"
  tail -40 /tmp/ai-agent-server-restart.log || true
  exit 3
fi
node --check frontend/app.js
node --check backend/extension/pet/limbo/pet.js
node --check backend/server.js
git diff --check -- frontend/index.html frontend/app.js frontend/style.css backend/server.js
echo 'PASS: syntax and diff checks'
