#!/bin/bash
set +e
cd /home/runner/workspace/ai-agent || exit 1
echo "=== floating UI selectors and pet gesture hooks ==="
grep -n -E 'mainFloat|spawnFrom|pointerdown|mousedown|pointermove|mousemove|pointerup|touchstart|touchmove|touchend|drag' frontend/app.js frontend/index.html | head -160
echo "=== visible headed browser: real pointer drag against actual frontend float ==="
node --input-type=module <<'NODE'
import {createBrowserSession,getBrowserPage,closeBrowserSession} from './backend/extension/browser/browser_service.js';
let id; const out={};
try{
 const s=await createBrowserSession({headless:false});id=s.sessionId;const p=getBrowserPage(id);
 await p.setViewportSize({width:900,height:700});
 await p.goto('http://127.0.0.1:3000/',{waitUntil:'domcontentloaded',timeout:15000});await p.waitForTimeout(1400);
 out.title=await p.title();out.url=p.url();out.viewport=await p.evaluate(()=>({w:innerWidth,h:innerHeight,dpr:devicePixelRatio}));
 out.float=await p.evaluate(()=>{const e=document.querySelector('#mainFloat');if(!e)return null;const r=e.getBoundingClientRect();return {tag:e.tagName,id:e.id,cls:e.className,rect:{x:r.x,y:r.y,width:r.width,height:r.height},display:getComputedStyle(e).display,visibility:getComputedStyle(e).visibility,pointerEvents:getComputedStyle(e).pointerEvents,html:e.outerHTML.slice(0,600)}});
 out.petBefore=await p.locator('.limbo-pet').count();
 out.screenshotBefore='backend/extension/pet/cursor/visible_drag_before_S1260.png';
 await p.screenshot({path:out.screenshotBefore});
 if(out.float && out.float.rect.width>0 && out.float.rect.height>0){
  const x=out.float.rect.x+out.float.rect.width/2,y=out.float.rect.y+out.float.rect.height/2;
  await p.mouse.move(x,y);await p.mouse.down();await p.waitForTimeout(350);
  await p.mouse.move(x+140,y+110,{steps:12});await p.waitForTimeout(500);
  await p.mouse.up();await p.waitForTimeout(1800);
  out.drag={from:{x,y},to:{x:x+140,y:y+110},petCountAfter:await p.locator('.limbo-pet').count(),floatAfter:await p.evaluate(()=>{const e=document.querySelector('#mainFloat');if(!e)return null;const r=e.getBoundingClientRect();return{x:r.x,y:r.y,w:r.width,h:r.height}})};
  out.petAfter=await p.evaluate(()=>[...document.querySelectorAll('.limbo-pet')].map(e=>{const r=e.getBoundingClientRect();return {center:{x:r.x+r.width/2,y:r.y+r.height/2},gaze:e.dataset.gaze,cls:e.className}}));
  out.screenshotAfter='backend/extension/pet/cursor/visible_drag_after_S1260.png';
  await p.screenshot({path:out.screenshotAfter});
 }else out.dragSkipped='actual frontend #mainFloat is missing or has zero dimensions';
 out.sessionId=id;out.headed=true;
}catch(e){out.error=String(e?.stack||e)}
finally{try{if(id){await closeBrowserSession(id);out.closed=true}}catch(e){out.cleanupError=String(e)}}
console.log(JSON.stringify(out));
NODE
