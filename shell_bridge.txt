node --input-type=module <<'JS'
import WebSocket from "ws";
const base = "http://127.0.0.1:3000";
const key = process.env.BROWSER_API_KEY;
const headers = {"content-type":"application/json","x-browser-key":key};
const post = async (route, body={}) => {
  const r = await fetch(base+route,{method:"POST",headers,body:JSON.stringify(body)});
  const text = await r.text();
  if (!r.ok) throw new Error(route+" HTTP "+r.status+": "+text);
  try { return JSON.parse(text); } catch { return text; }
};
let sessionId = "";
let frameCount = 0;
let created = false;
const ws = new WebSocket("ws://127.0.0.1:3000/browser/live");
const framePromise = new Promise((resolve,reject)=>{
  const timeout=setTimeout(()=>resolve({timeout:true,frameCount}),12000);
  ws.on("error",reject);
  ws.on("open",async()=>{
    try {
      const s=await post("/browser/session");
      sessionId=s.sessionId;
      console.log("SESSION_CREATED="+Boolean(sessionId));
      await post("/browser/open",{sessionId,url:"https://example.com"});
    } catch(e) { clearTimeout(timeout); reject(e); }
  });
  ws.on("message",raw=>{
    let m; try {m=JSON.parse(raw.toString())} catch{return}
    if(m.type==="session-created" && m.sessionId===sessionId){created=true;ws.send(JSON.stringify({type:"attach",sessionId}));}
    if(m.type==="frame" && m.sessionId===sessionId){frameCount++;if(frameCount>=3){clearTimeout(timeout);resolve({timeout:false,frameCount});}}
    if(m.type==="error") console.log("WS_ERROR="+m.message);
  });
});
try {
  const result=await framePromise;
  console.log("SESSION_EVENT_RECEIVED="+created);
  console.log("FRAME_COUNT="+result.frameCount);
  console.log("FRAME_TEST="+(!result.timeout && result.frameCount>=3 ? "PASS":"NO_FRAMES"));
  const page=await post("/browser/read",{sessionId});
  console.log("PAGE_TITLE="+page.title);
  console.log("PAGE_URL="+page.url);
} catch(e) {
  console.log("INTEGRATION_ERROR="+e.message);
} finally {
  if(sessionId) { try { console.log("SESSION_CLOSE="+JSON.stringify(await post("/browser/close",{sessionId}))); } catch(e) { console.log("SESSION_CLOSE_ERROR="+e.message); } }
  try {ws.close()} catch{}
}
JS
printf '%s\n' '=== FINAL HTTP ASSET CHECK ==='
