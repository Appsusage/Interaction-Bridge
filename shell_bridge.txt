cat > backend/heartbeat/pid.js <<'EOF'
import fs from "node:fs";
import path from "node:path";
import { fileURLToPath } from "node:url";

const HERE = path.dirname(fileURLToPath(import.meta.url));
export const PID_FILE = path.join(HERE, "pid.json");

export function emptyRegistry() {
  return { version: 1, processes: [], failedTargets: [] };
}

function normalizeRegistry(value) {
  if (!value || typeof value !== "object") return emptyRegistry();
  return {
    version: 1,
    processes: Array.isArray(value.processes)
      ? value.processes.filter(
          item =>
            item &&
            typeof item === "object" &&
            Number.isInteger(Number(item.pid)) &&
            item.name &&
            item.command
        )
      : [],
    failedTargets: Array.isArray(value.failedTargets)
      ? value.failedTargets.filter(
          item => item && typeof item === "object" && typeof item.targetKey === "string"
        )
      : []
  };
}

export function readRegistry() {
  try {
    return normalizeRegistry(JSON.parse(fs.readFileSync(PID_FILE, "utf8")));
  } catch {
    return emptyRegistry();
  }
}

export function writeRegistry(value) {
  const registry = normalizeRegistry(value);
  fs.mkdirSync(HERE, { recursive: true });
  const temporary = PID_FILE + ".tmp";
  fs.writeFileSync(temporary, JSON.stringify(registry, null, 2) + "\n", "utf8");
  fs.renameSync(temporary, PID_FILE);
}

export function ensureRegistryFile() {
  if (!fs.existsSync(PID_FILE)) writeRegistry(emptyRegistry());
}
EOF
