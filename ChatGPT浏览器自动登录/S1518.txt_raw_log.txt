// Runs in the remote ChatGPT page before application scripts. The phone browser
// supplies the microphone track over WebRTC; ChatGPT's playback is mixed back
// into the same WebRTC audio connection.
function installRemoteAudioBridge() {
  if (window.__aiAudioBridgeInstalled) return;
  const mediaDevices = navigator.mediaDevices;
  const originalGetUserMedia = mediaDevices && mediaDevices.getUserMedia
    ? mediaDevices.getUserMedia.bind(mediaDevices) : null;
  let peer = null;
  let audioContext = null;
  let destination = null;
  const connectedElements = new WeakSet();
  window.__aiAudioBridgeMicStream = null;

  function ensureAudioGraph() {
    if (!audioContext) {
      audioContext = new AudioContext();
      destination = audioContext.createMediaStreamDestination();
    }
    document.querySelectorAll("audio,video").forEach(element => {
      if (connectedElements.has(element)) return;
      try {
        const source = audioContext.createMediaElementSource(element);
        source.connect(audioContext.destination);
        source.connect(destination);
        connectedElements.add(element);
      } catch {}
    });
    if (audioContext.state === "suspended") audioContext.resume().catch(() => {});
    return destination;
  }

  const observer = new MutationObserver(() => {
    try { ensureAudioGraph(); } catch {}
  });
  function observeDocument() {
    if (document.documentElement) {
      try { observer.observe(document.documentElement, { childList: true, subtree: true }); } catch {}
    }
  }
  observeDocument();
  document.addEventListener("DOMContentLoaded", () => {
    observeDocument();
    try { ensureAudioGraph(); } catch {}
  }, { once: true });
  document.addEventListener("click", () => {
    ensureAudioGraph();
    if (audioContext && audioContext.state === "suspended") audioContext.resume().catch(() => {});
  }, true);

  if (mediaDevices && originalGetUserMedia) {
    try {
      Object.defineProperty(mediaDevices, "getUserMedia", {
        configurable: true,
        writable: true,
        value: async constraints => {
          const remoteMic = window.__aiAudioBridgeMicStream;
          if (constraints && constraints.audio && remoteMic && remoteMic.getAudioTracks().length) {
            const tracks = remoteMic.getAudioTracks().map(track => track.clone());
            if (constraints.video) {
              const extra = await originalGetUserMedia({ audio: false, video: constraints.video });
              tracks.push(...extra.getVideoTracks());
            }
            return new MediaStream(tracks);
          }
          return originalGetUserMedia(constraints);
        }
      });
    } catch {}
  }

  function waitForIce(pc) {
    if (pc.iceGatheringState === "complete") return Promise.resolve();
    return new Promise(resolve => {
      const timer = setTimeout(resolve, 12000);
      pc.addEventListener("icegatheringstatechange", () => {
        if (pc.iceGatheringState === "complete") {
          clearTimeout(timer);
          resolve();
        }
      });
    });
  }

  window.__aiAudioBridge = {
    async acceptOffer(offer) {
      if (peer) { try { peer.close(); } catch {} }
      const graph = ensureAudioGraph();
      peer = new RTCPeerConnection({ iceServers: [{ urls: "stun:stun.l.google.com:19302" }] });
      peer.ontrack = event => {
        window.__aiAudioBridgeMicStream = event.streams[0] || new MediaStream([event.track]);
      };
      await peer.setRemoteDescription(offer);
      let transceiver = peer.getTransceivers().find(item => item.receiver && item.receiver.track && item.receiver.track.kind === "audio");
      if (!transceiver) transceiver = peer.addTransceiver("audio", { direction: "sendrecv" });
      transceiver.direction = "sendrecv";
      const outputTrack = graph.stream.getAudioTracks()[0];
      if (outputTrack) await transceiver.sender.replaceTrack(outputTrack);
      await peer.setLocalDescription(await peer.createAnswer());
      await waitForIce(peer);
      return { type: peer.localDescription.type, sdp: peer.localDescription.sdp };
    },
    close() {
      if (peer) { try { peer.close(); } catch {} peer = null; }
      window.__aiAudioBridgeMicStream = null;
    }
  };
  window.__aiAudioBridgeInstalled = true;
}

async function isLoggedIn(page) {
  const url = page.url();
  const text = await pageText(page);
  if (/auth\.openai\.com|\/auth\/login|\/email-verification/i.test(url)) return false;

  // The signed-out ChatGPT homepage also contains "New chat" and other
  // navigation labels. Check explicit signed-out UI first to avoid treating
  // the public landing page as an authenticated session.
  if (/Log in to get answers|Log in or sign up|Sign up for free|Continue with Google|Continue with Microsoft|Enter your email|Check your inbox|verification code/i.test(text)) return false;

  return /What’s on the agenda|What can I help with|Upgrade|Free/i.test(text);
}

        emitStatus("waiting", { step: "homepage-login-button", title, url: page.url(), note: "ChatGPT 首页还没有显示登录按钮；保留当前会话，不直接跳入 /auth/login。" });

    });
    let state = await readState();
    if (/just a moment|verify you are human|security check|unusual activity|captcha/i.test(state.title + " " + state.text)) {
      emitStatus("waiting", { step: "security-check", note: "ChatGPT 正在显示安全检查页面；不会绕过检查，请用户在实时浏览器中正常完成。" });
      return;

      return;

    if (/accounts\.google\.com/i.test(state.url) || /password|enter your password|choose an account|verify it's you/i.test(state.text)) {

      return;
    }
    if (/auth\.openai\.com|\/auth\/login/i.test(state.url) || /Log in or sign up|Continue with Google|Continue with Microsoft|Enter your email/i.test(state.text)) {
      emitStatus("waiting", { step: "login-form", email: loginEmail || null, url: state.url, note: "登录页面仍未完成自动识别；已停止重复提交，请在实时浏览器中选择登录方式或完成页面提示。" });
      return;
    }
    emitStatus("waiting", { step: "manual-login", email: loginEmail || null, url: state.url, note: "无法确认登录状态；已停止自动重试，请在实时浏览器检查当前页面。" });
  } catch (error) {
    emitStatus("失败", { step: "login-flow", error: error.message, note: "自动登录步骤已停止，不会持续重试。" });

      }
    }

    emitStatus("失败", { step: "state-check", error: error.message });
  }
