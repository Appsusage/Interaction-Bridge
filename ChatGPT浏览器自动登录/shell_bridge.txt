cd /home/runner/workspace/ai-agent
node --input-type=module -e '
import fs from "node:fs";
const pf=process.env.CHATGPT_PROFILE_DIR || ".chatgpt-profile";
console.log("CHATGPT_PROFILE_EXISTS",fs.existsSync(pf));
console.log("CHATGPT_MODULE_EXISTS",fs.existsSync("backend/extension/browser/chatgpt_live.js"));
let done=false;
const ws=new WebSocket("ws://127.0.0.1:3000/browser/live");
const timer=setTimeout(()=>{if(!done){console.log("LIVE_STATUS_TIMEOUT");done=true;ws.close();}},9000);
function finish(){done=true;clearTimeout(timer);ws.close();}
ws.addEventListener("message",ev=>{
try{
const m=JSON.parse(ev.data);
if(m.type==="sessions"){
const a=(m.sessions||[]).map(s=>({sessionId:s.sessionId,url:s.url}));
console.log("BROWSER_SESSIONS",JSON.stringify(a));
const target=a.find(s=>/chatgpt\.com|auth\.openai\.com/.test(s.url||""));
if(target)ws.send(JSON.stringify({type:"attach",sessionId:target.sessionId}));
else finish();
}else if(m.type==="status"){console.log("CHATGPT_LIVE_STATUS",JSON.stringify({state:m.state,url:m.url,title:m.title}));finish();}
else if(m.type==="error"){console.log("CHATGPT_LIVE_ERROR",m.message);finish();}
}catch(e){console.log("READ_ERROR",e.message);finish();}
});
ws.addEventListener("error",()=>{console.log("LIVE_SOCKET_ERROR");finish();});
ws.addEventListener("close",()=>{if(!done){console.log("LIVE_SOCKET_CLOSED");clearTimeout(timer);done=true;}});
'
