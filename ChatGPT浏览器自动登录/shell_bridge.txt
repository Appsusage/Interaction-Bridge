cd /home/runner/workspace/ai-agent
node --input-type=module <<'ALL_SESSIONS'
const ws=new WebSocket("ws://127.0.0.1:3000/browser/live");
const timer=setTimeout(()=>{console.log("SESSIONS_TIMEOUT");ws.close();},8000);
ws.addEventListener("message",async event=>{
 const m=JSON.parse(event.data);
 if(m.type!=="sessions")return;
 clearTimeout(timer);ws.close();
 console.log("SESSION_TOTAL",(m.sessions||[]).length);
 for(const s of m.sessions||[]){
  let u;try{u=new URL(s.url);}catch{continue;}
  const isChat=/chatgpt\.com$/.test(u.hostname);
  const safePath=u.pathname.startsWith("/c/")?"/c/[conversation]":u.pathname;
  const item={sessionId:s.sessionId,origin:u.origin,path:safePath};
  if(isChat&&process.env.BROWSER_API_KEY){
   try{
    const r=await fetch("http://127.0.0.1:3000/browser/recognition/inspect",{
      method:"POST",headers:{"content-type":"application/json","x-browser-key":process.env.BROWSER_API_KEY},
      body:JSON.stringify({sessionId:s.sessionId}),signal:AbortSignal.timeout(10000)
    });
    const d=await r.json();
    item.title=d.title;item.challengeDetected=d.challengeDetected;
    item.loginCandidates=(d.candidates||[]).filter(x=>/log.?in|sign.?in|登录/i.test(x.text||"")).slice(0,8).map(x=>({tag:x.tag,text:x.text,rect:x.rect}));
   }catch(e){item.error=e.message;}
  }
  console.log("SESSION_INFO",JSON.stringify(item));
 }
});
ws.addEventListener("error",()=>{clearTimeout(timer);console.log("SESSION_CONNECTION_FAILED");});
ALL_SESSIONS
