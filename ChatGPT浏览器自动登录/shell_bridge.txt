node --input-type=module <<'NODE'
import fs from 'node:fs';
const listen=[];
for(let p of ['/proc/net/tcp','/proc/net/tcp6'])try{const lines=fs.readFileSync(p,'utf8').trim().split('\n').slice(1);for(const line of lines){const parts=line.trim().split(/\s+/);if(parts[3]==='0A')listen.push({port:parseInt(parts[1].split(':').at(-1),16),inode:parts[9]})}}catch{}
for(let id of [113980,161990])try{
let env=fs.readFileSync('/proc/'+id+'/environ','utf8').split('\u0000').find(x=>x.startsWith('PORT='))||'PORT=3000(default)';
let fd=fs.readdirSync('/proc/'+id+'/fd').map(n=>{try{return fs.readlinkSync('/proc/'+id+'/fd/'+n)}catch{return ''}});
console.log('SERVER_LISTENING',JSON.stringify({pid:id,portConfig:env,listeningPorts:listen.filter(x=>fd.includes('socket:['+x.inode+']')).map(x=>x.port)}));
}catch(e){console.log('SERVER_PROC_FAIL',id,e.message)}
NODE
