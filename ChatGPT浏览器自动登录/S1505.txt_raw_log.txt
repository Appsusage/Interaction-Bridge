async function beginEmailLogin(page) {
  if (loginFlowActive) return;
  loginFlowActive = true;
  emitStatus("进行中", { step: "open-login" });
  try {
    if (!/auth\.openai\.com|\/auth\/login/i.test(page.url())) {
      await page.goto(CHATGPT_LOGIN, { waitUntil: "domcontentloaded", timeout: 30000 });
    }
    await wait(1800);
    const loginEmail = getLoginEmail();
    const readState = async () => ({
      title: await page.title().catch(() => ""),
      text: await pageText(page),
      url: page.url()
    });
    let state = await readState();
    if (/just a moment|verify you are human|security check|unusual activity|captcha/i.test(state.title + " " + state.text)) {
      emitStatus("waiting", { step: "security-check", note: "ChatGPT 正在显示安全检查页面；不会绕过检查，请用户在实时浏览器中正常完成。" });
      return;
    }

    // Prefer the normal email flow. If the login page offers no email field but
    // the configured account is Gmail, try the official Google sign-in button once.
    let emailField = page.locator('input[type="email"], input[name="email"], input[autocomplete="username"]').first();
    if (!(await emailField.count())) {
      // Some current auth screens first show a Log in / Sign in choice.
      // Open only an explicit login entry, then re-detect the form.
      const loginEntryButton = page.getByRole("button", { name: /^(log in|sign in|log in or sign up)$/i }).first();
      const loginEntryLink = page.getByRole("link", { name: /^(log in|sign in)$/i }).first();
      if (await loginEntryButton.count()) {
        await loginEntryButton.click({ timeout: 8000 }).catch(() => {});
        await wait(1200);
        state = await readState();
      } else if (await loginEntryLink.count()) {
        await loginEntryLink.click({ timeout: 8000 }).catch(() => {});
        await wait(1200);
        state = await readState();
      }
      emailField = page.locator('input[type="email"], input[name="email"], input[autocomplete="username"]').first();
    }
    if (!(await emailField.count())) {
      const continueEmail = page.getByRole("button", { name: /continue with email|email address/i }).first();
      if (await continueEmail.count()) {
        await continueEmail.click({ timeout: 8000 }).catch(() => {});
        await wait(1000);
        emailField = page.locator('input[type="email"], input[name="email"], input[autocomplete="username"]').first();
      }
    }
    if (await emailField.count()) {
      if (!loginEmail) {
        emitStatus("waiting", { step: "manual-login", note: "未配置 CHATGPT_LOGIN_EMAIL；请用户在实时浏览器输入自己的 ChatGPT 账号。" });
        return;
      }
      await emailField.fill(loginEmail, { timeout: 10000 });
      const continueButton = page.getByRole("button", { name: /^continue$|^next$/i }).first();
      if (await continueButton.count()) {
        await continueButton.click({ timeout: 10000 });
      } else {
        await emailField.press("Enter").catch(() => {});
      }
      await wait(1800);
    } else if (/gmail\.com$/i.test(loginEmail)) {
      const googleButton = page.getByRole("button", { name: /continue with google|google/i }).first();
      if (await googleButton.count()) {
        await googleButton.click({ timeout: 10000 });
        await wait(1800);
        state = await readState();
        // Select only the exact configured Google account when the account chooser offers it.
        const exactAccount = page.getByText(loginEmail, { exact: true }).first();
        if (await exactAccount.count()) {
          await exactAccount.click({ timeout: 8000 }).catch(() => {});
          await wait(1500);
        }
      }
    }

    state = await readState();

async function monitorLoginState() {
  if (!pageRef) return;
  try {
    if (await isLoggedIn(pageRef)) {
      if (loginFlowActive) emitStatus("完成", { step: "logged-in", url: pageRef.url() });
      loginFlowActive = false;
      return;
    }
    if (!loginFlowActive) await beginEmailLogin(pageRef);
    else {
      const title = await pageRef.title().catch(() => "");
      const text = await pageText(pageRef);
      const url = pageRef.url();
      if (/just a moment|captcha|verify you are human|security check|unusual activity/i.test(title + " " + text)) {
        emitStatus("waiting", { step: "security-check", note: "ChatGPT 正在显示安全检查页面；不会绕过检查，请用户在实时浏览器中正常完成。" });
      } else if (/Check your inbox|verification code|Enter the code|email-verification/i.test(text + " " + url)) {
        emitStatus("waiting", { step: "email-verification", email: getLoginEmail() || null, note: "等待用户在实时浏览器完成验证码；程序不会读取验证码。" });
      } else if (/accounts\.google\.com/i.test(url) || /password|choose an account|verify it's you/i.test(text)) {
        emitStatus("waiting", { step: "provider-auth", provider: /accounts\.google\.com/i.test(url) ? "google" : "unknown", note: "等待用户完成账户选择、密码或 MFA 验证；不会自动输入密码或绕过验证。" });
      }
    }
  } catch (error) {
    emitStatus("失败", { step: "state-check", error: error.message });
  }
}
