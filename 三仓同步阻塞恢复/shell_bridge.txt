set -u
printf 'TRI_REPO_VERIFY_AT=%s\n' "$(date -u '+%Y-%m-%dT%H:%M:%SZ')"
node backend/sync/tri_repo_supervisor.js status
for kind in .gen-sync-cache .migration-sync-cache; do
  folder="/home/runner/workspace/$kind"
  printf 'CACHE=%s\n' "$kind"
  if [ -d "$folder/.git" ]; then
    printf 'GIT_STATE=initialized\n'
    git -C "$folder" rev-parse --verify HEAD 2>&1 || true
  elif [ -d "$folder" ]; then
    printf 'GIT_STATE=directory_without_git\n'
  elif [ -e "$folder" ]; then
    printf 'GIT_STATE=non_directory\n'
  else
    printf 'GIT_STATE=absent\n'
  fi
done
