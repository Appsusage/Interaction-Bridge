set -u
printf 'GIT_CACHE_INTEGRITY_AT=%s\n' "$(date -u '+%Y-%m-%dT%H:%M:%SZ')"
node backend/sync/tri_repo_supervisor.js status
for name in .gen-sync-cache .migration-sync-cache; do
  d="/home/runner/workspace/$name"
  printf 'CACHE=%s\n' "$name"
  if [ ! -d "$d/.git" ]; then
    printf 'NO_VALID_GIT_DIR\n'
    continue
  fi
  printf 'HEAD='
  git -C "$d" rev-parse --short HEAD 2>&1 || true
  printf 'REMOTE_MAIN='
  git -C "$d" rev-parse --short refs/remotes/origin/main 2>&1 || true
  printf 'TRACKED_COUNT='
  git -C "$d" ls-files | wc -l
  printf 'MISSING_TRACKED_COUNT='
  git -C "$d" ls-files --deleted | wc -l
  printf 'UNSTAGED_MODIFIED_COUNT='
  git -C "$d" diff --name-only | wc -l
  printf 'GIT_STATUS_SUMMARY='
  git -C "$d" status --short --untracked-files=no 2>&1 | awk 'BEGIN{n=0} {n++} END{print n}'
done
printf 'GIT_CACHE_INTEGRITY_DONE\n'
