set -u
d="/home/runner/workspace/.gen-sync-cache"
printf 'GEN_LOCK_AUDIT_AT=%s\n' "$(date -u '+%Y-%m-%dT%H:%M:%SZ')"
printf 'GIT_RUNNING_PROCESSES\n'
ps -eo pid,ppid,etime,stat,comm | awk 'NR==1 || $5=="git" || $5=="git-lfs" || $5=="git-remote-ht"'
printf 'LOCK_STAT\n'
if [ -f "$d/.git/index.lock" ]; then
  stat -c 'INDEX_LOCK_MTIME=%y SIZE=%s' "$d/.git/index.lock"
else
  printf 'NO_INDEX_LOCK\n'
fi
printf 'HEAD_TREE_COUNT='
git -C "$d" ls-tree -r --name-only HEAD | wc -l
printf 'INDEX_ENTRY_COUNT='
git -C "$d" ls-files | wc -l
printf 'STAGED_DELETE_COUNT='
git -C "$d" diff --cached --diff-filter=D --name-only HEAD | wc -l
printf 'UNTRACKED_COUNT='
git -C "$d" ls-files --others --exclude-standard | wc -l
printf 'HEAD_AND_ORIGIN_EQUAL='
a="$(git -C "$d" rev-parse HEAD)"
b="$(git -C "$d" rev-parse origin/main)"
if [ "$a" = "$b" ]; then printf 'yes\n'; else printf 'no\n'; fi
printf 'GEN_LOCK_AUDIT_DONE\n'
