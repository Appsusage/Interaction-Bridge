set -u
cd /home/runner/workspace/ai-agent || exit 11
printf 'TRI_REPO_DIAG_AT=%s\n' "$(date -u '+%Y-%m-%dT%H:%M:%SZ')"
node backend/sync/tri_repo_supervisor.js status
printf 'ACTIVE_GIT_PROCESSES\n'
ps -eo pid,ppid,etime,stat,comm | awk 'NR==1 || $5=="git" || $5=="git-lfs"'
for name in gen migration; do
  d="/home/runner/workspace/.$name-sync-cache"
  printf '%s_CACHE\n' "$name"
  if [ -d "$d/.git" ]; then
    printf 'GIT_DIR_VALID=yes\n'
    if [ -f "$d/.git/index.lock" ]; then stat -c 'INDEX_LOCK=%y SIZE=%s' "$d/.git/index.lock"; else printf 'INDEX_LOCK=no\n'; fi
    printf 'HEAD='
    git -C "$d" rev-parse --short HEAD || true
    printf 'INDEX_COUNT='
    git -C "$d" ls-files | wc -l
  else printf 'GIT_DIR_VALID=no\n'; fi
done
printf 'TRI_REPO_DIAG_DONE\n'
