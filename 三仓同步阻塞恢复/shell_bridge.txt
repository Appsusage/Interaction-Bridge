set -eu
umask 077
cd /home/runner/workspace/ai-agent
supervisor="backend/sync/tri_repo_supervisor.js"
cache="/home/runner/workspace/.gen-sync-cache"
runtime="/home/runner/workspace/.ai-agent-tri-repo-runtime"
printf 'GEN_BULK_REPAIR_BEGIN=%s\n' "$(date -u '+%Y-%m-%dT%H:%M:%SZ')"
restart_required=0
ensure_restart() {
  if [ "$restart_required" -eq 1 ]; then
    printf 'RESTORE_SUPERVISOR_AFTER_REPAIR\n'
    node "$supervisor" start || true
    restart_required=0
  fi
}
trap ensure_restart EXIT
if [ ! -d "$cache/.git" ]; then printf 'ABORT_INVALID_CACHE\n'; exit 31; fi
head="$(git -C "$cache" rev-parse HEAD)"
remote="$(git -C "$cache" rev-parse origin/main)"
printf 'CACHE_HEAD=%s REMOTE_HEAD=%s\n' "$head" "$remote"
if [ "$head" != "$remote" ]; then printf 'ABORT_UNPUSHED_COMMIT\n'; exit 32; fi
state="$(node "$supervisor" status)"
pid="$(printf '%s' "$state" | node -e 'let s="";process.stdin.on("data",x=>s+=x).on("end",()=>{try{const j=JSON.parse(s);process.stdout.write(String(j.status==="running"?j.pid:""));}catch{process.exitCode=1}})')"
if [ -n "$pid" ]; then
  case "$pid" in *[!0-9]*|'') printf 'ABORT_INVALID_SUPERVISOR_PID\n'; exit 33;; esac
  if [ ! -r "/proc/$pid/cmdline" ] || ! tr '\000' ' ' < "/proc/$pid/cmdline" | grep -q 'tri_repo_supervisor.js'; then
    printf 'ABORT_SUPERVISOR_OWNERSHIP_UNVERIFIED\n'; exit 34
  fi
  printf 'STOP_ONLY_TRI_REPO_SUPERVISOR_PID=%s\n' "$pid"
  node "$supervisor" stop
  restart_required=1
  stopped=0
  for i in $(seq 1 20); do
    if ! test -d "/proc/$pid" || ! tr '\000' ' ' < "/proc/$pid/cmdline" | grep -q 'tri_repo_supervisor.js'; then stopped=1; break; fi
    sleep 1
  done
  if [ "$stopped" -ne 1 ]; then
    if ! tr '\000' ' ' < "/proc/$pid/cmdline" | grep -q 'tri_repo_supervisor.js'; then printf 'ABORT_PID_CHANGED\n'; exit 35; fi
    printf 'FORCE_STOP_ONLY_VERIFIED_SYNC_SUPERVISOR_PID=%s\n' "$pid"
    kill -KILL "$pid"
  fi
fi
git_idle=0
for i in $(seq 1 120); do
  if ! ps -eo comm= | grep -Eq '^(git|git-lfs)$'; then git_idle=1; break; fi
  sleep 1
done
if [ "$git_idle" -ne 1 ]; then printf 'ABORT_ACTIVE_GIT_PROCESS_NO_CACHE_CHANGE\n'; exit 36; fi
if [ -f "$cache/.git/index.lock" ]; then
  printf 'ABORT_INDEX_LOCK_AFTER_QUIESCENCE\n'
  exit 37
fi
head_now="$(git -C "$cache" rev-parse HEAD)"
remote_now="$(git -C "$cache" rev-parse origin/main)"
if [ "$head_now" != "$head" ] || [ "$remote_now" != "$remote" ]; then
  printf 'ABORT_CACHE_HEAD_CHANGED\n'; exit 38
fi
size_kb="$(du -sk "$cache" | awk '{print $1}')"
free_kb="$(df -Pk "$cache" | awk 'END{print $4}')"
printf 'CACHE_KB=%s FREE_KB=%s\n' "$size_kb" "$free_kb"
if [ "$free_kb" -le "$((size_kb * 2))" ]; then printf 'ABORT_INSUFFICIENT_BACKUP_SPACE\n'; exit 39; fi
mkdir -p "$runtime"
backup="$(mktemp -d "$runtime/preserved-gen-cache-bulk-XXXXXXXX")"
printf 'BACKUP_BEGIN=%s\n' "$backup"
cp -a "$cache" "$backup/gen-sync-cache"
backup_head="$(git -C "$backup/gen-sync-cache" rev-parse HEAD)"
if [ "$backup_head" != "$head" ] || [ ! -f "$backup/gen-sync-cache/.git/index" ]; then
  printf 'ABORT_BACKUP_VERIFICATION_FAILED\n'; exit 40
fi
printf 'BACKUP_VERIFIED=%s\n' "$backup"
printf 'BULK_REBUILD_INDEX\n'
git -C "$cache" read-tree --reset HEAD
printf 'BULK_RESTORE_TRACKED_FILES\n'
git -C "$cache" checkout-index --all --force
printf 'PRESERVED_UNTRACKED_IN_BACKUP_CLEAN_MIRROR\n'
git -C "$cache" clean -fdq
index_count="$(git -C "$cache" ls-files | wc -l | tr -d ' ')"
tree_count="$(git -C "$cache" ls-tree -r --name-only HEAD | wc -l | tr -d ' ')"
dirty_count="$(git -C "$cache" status --porcelain --untracked-files=all | wc -l | tr -d ' ')"
printf 'INDEX_COUNT=%s TREE_COUNT=%s DIRTY_COUNT=%s\n' "$index_count" "$tree_count" "$dirty_count"
if [ "$index_count" -ne "$tree_count" ] || [ "$dirty_count" -ne 0 ]; then
  printf 'INCOMPLETE_CACHE_REPAIR_BACKUP_PRESERVED\n'; exit 41
fi
printf 'GEN_CACHE_CLEAN_VERIFIED\n'
ensure_restart
node "$supervisor" status
printf 'GEN_BULK_REPAIR_DONE=%s\n' "$(date -u '+%Y-%m-%dT%H:%M:%SZ')"
