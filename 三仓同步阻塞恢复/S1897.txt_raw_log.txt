    printf 'CACHE_KIND=symlink\n'
  elif [ -d "$d" ]; then
    printf 'CACHE_KIND=directory\n'

    if [ -d "$d/.git" ]; then
      printf 'GIT_KIND=directory\n'
      git -C "$d" status --porcelain=v1 --untracked-files=no 2>&1 | head -n 12
      git -C "$d" rev-parse --verify HEAD 2>&1 || true
    elif [ -f "$d/.git" ]; then
      printf 'GIT_KIND=file\n'
    else
      printf 'GIT_KIND=missing\n'
    fi
    printf 'IMMEDIATE_ENTRY_COUNT='
    find "$d" -mindepth 1 -maxdepth 1 -print 2>/dev/null | wc -l
  elif [ -e "$d" ]; then
    printf 'CACHE_KIND=non_directory\n'
    stat -c 'CACHE_MTIME=%y CACHE_MODE=%a CACHE_OWNER=%U' "$d" 2>&1 || true
  else
    printf 'CACHE_KIND=absent\n'
  fi
done
printf 'TRI_REPO_DIAG_END\n'
