 child.unref();
 console.log('STARTED_REPLACEMENT',child.pid);

 console.log('REPLACEMENT_ALREADY_RUNNING',JSON.stringify(current));

 await wait(900);

 if(health===200&&privateSite===404&&privateShared===404&&ps.length===1&&ps[0]!==oldPid){

 }

}
if(!successful)process.exitCode=2;
